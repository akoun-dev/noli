   311→/* ──────── Comparison Modal ──────── */
   312→
   313→function ComparisonModal({
   314→  offers,
   315→  open,
   316→  onClose,
   317→}: {
   318→  offers: InsurerOffer[];
   319→  open: boolean;
   320→  onClose: () => void;
   321→}) {
   322→  // Collect all unique features across all offers (before early return for hooks rules)
   323→  const allFeatures = useMemo(() => {
   324→    const featureSet = new Set<string>();
   325→    offers.forEach((o) => o.features.forEach((f) => featureSet.add(f)));
   326→    return Array.from(featureSet);
   327→  }, [offers]);
   328→
   329→  if (!open || offers.length < 2) return null;
   330→
   331→  const formatMonthlyPrice = (price: number) => {
   332→    const val = Math.round(price);
   333→    if (val % 1 !== 0) return new Intl.NumberFormat("fr-FR", { minimumFractionDigits: 0, maximumFractionDigits: 3 }).format(val) + " FCFA";
   334→    return formatFCFA(val);
   335→  };
   336→
   337→  return (
   338→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   339→      <DialogContent className="sm:max-w-[900px] max-h-[90vh] overflow-y-auto p-0">
   340→        {/* Header */}
   341→        <div className="flex items-center justify-between px-6 py-4 border-b border-border/60 bg-muted/30 sticky top-0 z-10">
   342→          <DialogTitle className="text-lg font-bold text-foreground">
   343→            Comparer les garanties
   344→          </DialogTitle>
   345→          <DialogDescription className="sr-only">
   346→            Comparaison détaillée de {offers.length} offres d&apos;assurance
   347→          </DialogDescription>
   348→        </div>
   349→
   350→        {/* Comparison Table */}
   351→        <div className="overflow-x-auto">
   352→          <table className="w-full min-w-[500px]">
   353→            {/* Column Headers: Insurers */}
   354→            <thead>
   355→              <tr>
   356→                <th className="text-left p-4 w-48 bg-muted/30 sticky left-0 z-10 border-b border-border/60">
   357→                  <span className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   358→                    Critères
   359→                  </span>
   360→                </th>
   361→                {offers.map((offer) => (
   362→                  <th
   363→                    key={offer.id}
   364→                    className="text-center p-4 bg-muted/30 border-b border-border/60"
   365→                  >
   366→                    <div className="flex flex-col items-center gap-2">
   367→                      <div className="flex h-12 w-12 items-center justify-center rounded-full bg-primary/10 border-2 border-primary/20">
   368→                        <Shield className="size-6 text-primary" />
   369→                      </div>
   370→                      <span className="text-sm font-bold text-foreground leading-tight">
   371→                        {offer.insurerName}
   372→                      </span>
   373→                      <span className="text-xs text-muted-foreground capitalize">
   374→                        {offer.coverageType.replace("_", " ")}
   375→                      </span>
   376→                    </div>
   377→                  </th>
   378→                ))}
   379→              </tr>
   380→            </thead>
   381→
   382→            <tbody>
   383→              {/* Prix mensuel */}
   384→              <tr className="bg-card">
   385→                <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-card border-b border-border/30">
   386→                  Prix mensuel
   387→                </td>
   388→                {offers.map((offer) => (
   389→                  <td
   390→                    key={offer.id}
   391→                    className="p-4 text-center text-sm font-bold text-foreground border-b border-border/30"
   392→                  >
   393→                    {formatMonthlyPrice(offer.monthlyPrice)}
   394→                  </td>
   395→                ))}
   396→              </tr>
   397→
   398→              {/* Franchise */}
   399→              <tr className="bg-muted/20">
   400→                <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-muted/20 border-b border-border/30">
   401→                  Franchise
   402→                </td>
   403→                {offers.map((offer) => (
   404→                  <td
   405→                    key={offer.id}
   406→                    className="p-4 text-center text-sm text-foreground border-b border-border/30"
   407→                  >
   408→                    {formatFCFA(offer.deductible)}
   409→                  </td>
   410→                ))}
   411→              </tr>
   412→
   413→              {/* Note / Rating */}
   414→              <tr className="bg-card">
   415→                <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-card border-b border-border/30">
   416→                  Note assureur
   417→                </td>
   418→                {offers.map((offer) => (
   419→                  <td
   420→                    key={offer.id}
   421→                    className="p-4 text-center border-b border-border/30"
   422→                  >
   423→                    <div className="flex items-center justify-center">
   424→                      <StarRating rating={offer.insurerRating} />
   425→                    </div>
   426→                  </td>
   427→                ))}
   428→              </tr>
   429→
   430→              {/* Features */}
   431→              {allFeatures.map((feature, idx) => {
   432→                const isZebra = idx % 2 === 1;
   433→                return (
   434→                  <tr
   435→                    key={feature}
   436→                    className={isZebra ? "bg-muted/20" : "bg-card"}
   437→                  >
   438→                    <td
   439→                      className={`p-4 text-sm text-foreground sticky left-0 border-b border-border/30 ${
   440→                        isZebra ? "bg-muted/20" : "bg-card"
   441→                      }`}
   442→                    >
   443→                      {feature}
   444→                    </td>
   445→                    {offers.map((offer) => {
   446→                      const hasFeature = offer.features.includes(feature);
   447→                      return (
   448→                        <td
   449→                          key={offer.id}
   450→                          className="p-4 text-center border-b border-border/30"
   451→                        >
   452→                          {hasFeature ? (
   453→                            <div className="flex items-center justify-center">
   454→                              <div className="flex h-6 w-6 items-center justify-center rounded-full bg-green-100 dark:bg-green-900/30">

   457→                            </div>
   458→                          ) : (
   459→                            <div className="flex items-center justify-center">
   460→                              <div className="flex h-6 w-6 items-center justify-center rounded-full bg-red-100 dark:bg-red-900/30">
   461→                                <X className="size-3.5 text-red-500 dark:text-red-400" />
   462→                              </div>
   463→                            </div>
   464→                          )}
   465→                        </td>
   466→                      );
   467→                    })}
   468→                  </tr>
   469→                );
   470→              })}
   471→
   472→              {/* Conditions */}
   473→              {offers.some((o) => o.conditions) && (
   474→                <tr className="bg-muted/20">
   475→                  <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-muted/20">
   476→                    Conditions
   477→                  </td>
   478→                  {offers.map((offer) => (
   479→                    <td
   480→                      key={offer.id}
   481→                      className="p-4 text-center text-xs text-muted-foreground max-w-[200px]"
   482→                    >
   483→                      <span className="line-clamp-2">{offer.conditions || "—"}</span>
   484→                    </td>
   485→                  ))}
   486→                </tr>
   487→              )}
   488→            </tbody>
   489→          </table>
   490→        </div>
   491→      </DialogContent>
   492→    </Dialog>
   493→  );
   494→}

   923→        next.delete(name);

   925→        next.add(name);

   936→    }

   940→    }

   952→        results.sort((a, b) => b.insurerRating - a.insurerRating);
