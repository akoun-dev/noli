     1→"use client";
     2→
     3→import { useEffect, useState, useCallback } from "react";
     4→import { useToast } from "@/hooks/use-toast";
     5→import {
     6→  Plus, Pencil, Trash2, Search, Loader2, ShieldCheck, MoreHorizontal, Eye, Check,
     7→} from "lucide-react";
     8→import { Button } from "@/components/ui/button";
     9→import { Input } from "@/components/ui/input";
    10→import { Label } from "@/components/ui/label";
    11→import { Badge } from "@/components/ui/badge";
    12→import { Switch } from "@/components/ui/switch";
    13→import { Textarea } from "@/components/ui/textarea";
    14→import { Skeleton } from "@/components/ui/skeleton";
    15→import { Card, CardContent } from "@/components/ui/card";
    16→import { Separator } from "@/components/ui/separator";
    17→import {
    18→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    19→} from "@/components/ui/table";
    20→import {
    21→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription,
    22→} from "@/components/ui/dialog";
    23→import {
    24→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    25→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    26→} from "@/components/ui/alert-dialog";
    27→import {
    28→  DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuTrigger,
    29→} from "@/components/ui/dropdown-menu";
    30→import {
    31→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    32→} from "@/components/ui/select";
    33→import { Checkbox } from "@/components/ui/checkbox";
    34→import { ScrollArea } from "@/components/ui/scroll-area";
    35→
    36→// ── Types ────────────────────────────────────────────────────────
    37→interface Offer {
    38→  id: string; insurerId: string; name: string; coverageType: string;
    39→  description: string | null; basePrice: number; annualPrice: number | null;
    40→  deductible: number | null; maxCoverage: number | null; features: string[];
    41→  conditions: string | null; isActive: boolean; createdAt: string;
    42→  insurer: { id: string; name: string; logo: string | null };
    43→  guaranteeLinks: { id: string; guaranteeId: string; isIncluded: boolean; guarantee: { id: string; name: string; calcMethod: string } }[];
    44→}
    45→
    46→interface InsurerMini { id: string; name: string; }
    47→interface GuaranteeMini { id: string; name: string; calcMethod: string; }
    48→
    49→const fmtPrice = (n: number) => new Intl.NumberFormat("fr-FR").format(n) + " FCFA";
    50→
    51→const coverageLabels: Record<string, string> = { tiers: "Tiers", tiers_plus: "Tiers+", tous_risques: "Tous Risques" };
    52→const coverageColors: Record<string, string> = {
    53→  tiers: "bg-slate-100 text-slate-800",
    54→  tiers_plus: "bg-amber-100 text-amber-800",
    55→  tous_risques: "bg-emerald-100 text-emerald-800",
    56→};
    57→
    58→const CalcBadge = ({ method }: { method: string }) => {
    59→  const colors: Record<string, string> = { FREE: "bg-emerald-100 text-emerald-800", FIXED_AMOUNT: "bg-blue-100 text-blue-800", VARIABLE_BASED: "bg-orange-100 text-orange-800", MATRIX_BASED: "bg-purple-100 text-purple-800" };
    60→  const labels: Record<string, string> = { FREE: "Gratuit", FIXED_AMOUNT: "Fixe", VARIABLE_BASED: "Variable", MATRIX_BASED: "Matrice" };
    61→  return <span className={`inline-block rounded-full px-2 py-0.5 text-[10px] font-semibold ${colors[method] || "bg-gray-100"}`}>{labels[method] || method}</span>;
    62→};
    63→
    64→const emptyOffer = {
    65→  insurerId: "", name: "", coverageType: "tiers" as string, description: "",
    66→  basePrice: 0, annualPrice: null as number | null, deductible: null as number | null,
    67→  maxCoverage: null as number | null, features: "[]" as string, conditions: "", isActive: true,
    68→};
    69→
    70→export function OffresTab() {
    71→  const { toast } = useToast();
    72→  const [offers, setOffers] = useState<Offer[]>([]);
    73→  const [insurers, setInsurers] = useState<InsurerMini[]>([]);
    74→  const [loading, setLoading] = useState(true);
    75→  const [search, setSearch] = useState("");
    76→  const [filterInsurer, setFilterInsurer] = useState("");
    77→  const [filterType, setFilterType] = useState("");
    78→  const [formOpen, setFormOpen] = useState(false);
    79→  const [editing, setEditing] = useState<Offer | null>(null);
    80→  const [form, setForm] = useState(emptyOffer);
    81→  const [saving, setSaving] = useState(false);
    82→  const [deleteId, setDeleteId] = useState<string | null>(null);
    83→  const [deleting, setDeleting] = useState(false);
    84→
    85→  // Guarantee dialog
    86→  const [guaranteeDialogOpen, setGuaranteeDialogOpen] = useState(false);
    87→  const [selectedOffer, setSelectedOffer] = useState<Offer | null>(null);
    88→  const [allGuarantees, setAllGuarantees] = useState<GuaranteeMini[]>([]);
    89→  const [guaranteeChecks, setGuaranteeChecks] = useState<{ guaranteeId: string; isIncluded: boolean }[]>([]);
    90→  const [savingGuarantees, setSavingGuarantees] = useState(false);
    91→
    92→  // Detail view
    93→  const [detailOffer, setDetailOffer] = useState<Offer | null>(null);
    94→  const [detailLoading, setDetailLoading] = useState(false);
    95→
    96→  const fetchOffers = useCallback(async () => {
    97→    try {
    98→      const params = new URLSearchParams();
    99→      if (search) params.set("search", search);
   100→      if (filterInsurer) params.set("insurerId", filterInsurer);
   101→      if (filterType) params.set("coverageType", filterType);
   102→      const res = await fetch(`/api/admin/offers?${params}`);
   103→      if (res.ok) setOffers(await res.json());
   104→    } finally { setLoading(false); }
   105→  }, [search, filterInsurer, filterType]);
   106→
   107→  const fetchInsurers = useCallback(async () => {
   108→    const res = await fetch("/api/admin/insurers?active=true");
   109→    if (res.ok) setInsurers(await res.json());
   110→  }, []);
   111→
   112→  useEffect(() => { fetchOffers(); fetchInsurers(); }, [fetchOffers, fetchInsurers]);
   113→
   114→  const filtered = offers;
   115→
   116→  const openCreate = () => { setEditing(null); setForm(emptyOffer); setFormOpen(true); };
   117→  const openEdit = (o: Offer) => {
   118→    setEditing(o);
   119→    setForm({
   120→      insurerId: o.insurerId, name: o.name, coverageType: o.coverageType,
   121→      description: o.description || "", basePrice: o.basePrice,
   122→      annualPrice: o.annualPrice, deductible: o.deductible, maxCoverage: o.maxCoverage,
   123→      features: JSON.stringify(o.features), conditions: o.conditions || "", isActive: o.isActive,
   124→    });
   125→    setFormOpen(true);
   126→  };
   127→
   128→  const handleSave = async () => {
   129→    if (!form.insurerId || !form.name.trim()) { toast({ title: "Assureur et nom obligatoires", variant: "destructive" }); return; }
   130→    setSaving(true);
   131→    try {
   132→      const url = editing ? `/api/admin/offers/${editing.id}` : "/api/admin/offers";
   133→      const res = await fetch(url, {
   134→        method: editing ? "PUT" : "POST",
   135→        headers: { "Content-Type": "application/json" },
   136→        body: JSON.stringify({ ...form, features: JSON.parse(form.features || "[]") }),
   137→      });
   138→      if (!res.ok) { const e = await res.json(); throw new Error(e.error || "Erreur"); }
   139→      toast({ title: editing ? "Offre modifiée" : "Offre créée" });
   140→      setFormOpen(false);
   141→      fetchOffers();
   142→    } catch (err) { toast({ title: (err as Error).message, variant: "destructive" }); }
   143→    finally { setSaving(false); }
   144→  };
   145→
   146→  const handleDelete = async () => {
   147→    if (!deleteId) return;
   148→    setDeleting(true);
   149→    try {
   150→      const res = await fetch(`/api/admin/offers/${deleteId}`, { method: "DELETE" });
   151→      if (!res.ok) throw new Error();
   152→      toast({ title: "Offre supprimée" });
   153→      setDeleteId(null); fetchOffers();
   154→    } catch { toast({ title: "Erreur de suppression", variant: "destructive" }); }
   155→    finally { setDeleting(false); }
   156→  };
   157→
   158→  const handleToggleActive = async (o: Offer) => {
   159→    try {
   160→      await fetch(`/api/admin/offers/${o.id}`, {
   161→        method: "PUT", headers: { "Content-Type": "application/json" },
   162→        body: JSON.stringify({ isActive: !o.isActive }),
   163→      });
   164→      fetchOffers();
   165→    } catch { toast({ title: "Erreur", variant: "destructive" }); }
   166→  };
   167→
   168→  // ── Detail view ────────────────────────────────────────────────
   169→  const openDetail = async (o: Offer) => {
   170→    setDetailLoading(true);
   171→    setDetailOffer(null);
   172→    try {
   173→      const res = await fetch(`/api/admin/offers/${o.id}`);
   174→      if (res.ok) {
   175→        setDetailOffer(await res.json());
   176→      } else throw new Error();
   177→    } catch { toast({ title: "Erreur de chargement", variant: "destructive" }); }
   178→    finally { setDetailLoading(false); }
   179→  };
   180→
   181→  // ── Guarantee dialog ────────────────────────────────────────────
   182→  const openGuaranteeDialog = async (offer: Offer) => {
   183→    try {
   184→      const [offerRes, guaranteesRes] = await Promise.all([
   185→        fetch(`/api/admin/offers/${offer.id}`),
   186→        fetch("/api/admin/guarantees"),
   187→      ]);
   188→      if (!offerRes.ok || !guaranteesRes.ok) throw new Error();
   189→      const offerFull: Offer = await offerRes.json();
   190→      const guarantees: GuaranteeMini[] = await guaranteesRes.json();
   191→      setSelectedOffer(offerFull);
   192→      setAllGuarantees(guarantees);
   193→      const linkMap = new Map(offerFull.guaranteeLinks.map((l) => [l.guaranteeId, l]));
   194→      setGuaranteeChecks(guarantees.map((g) => ({ guaranteeId: g.id, isIncluded: linkMap.get(g.id)?.isIncluded ?? false })));
   195→      setGuaranteeDialogOpen(true);
   196→    } catch { toast({ title: "Erreur de chargement", variant: "destructive" }); }
   197→  };
   198→
   199→  const toggleCheck = (idx: number) => {
   200→    setGuaranteeChecks((prev) => prev.map((c, i) => i === idx ? { ...c, isIncluded: !c.isIncluded } : c));
   201→  };
   202→
   203→  const saveGuarantees = async () => {
   204→    if (!selectedOffer) return;
   205→    setSavingGuarantees(true);
   206→    try {
   207→      const res = await fetch(`/api/admin/offers/${selectedOffer.id}/guarantees`, {
   208→        method: "PUT", headers: { "Content-Type": "application/json" },
   209→        body: JSON.stringify({ guarantees: guaranteeChecks }),
   210→      });
   211→      if (!res.ok) throw new Error();
   212→      toast({ title: "Garanties mises à jour" });
   213→      setGuaranteeDialogOpen(false);
   214→      fetchOffers();
   215→    } catch { toast({ title: "Erreur de sauvegarde", variant: "destructive" }); }
   216→    finally { setSavingGuarantees(false); }
   217→  };
   218→
   219→  // Group guarantees by calcMethod
   220→  const groupedGuarantees = allGuarantees.reduce<Record<string, GuaranteeMini[]>>((acc, g) => {
   221→    (acc[g.calcMethod] ??= []).push(g);
   222→    return acc;
   223→  }, {});
   224→
   225→  const groupLabels: Record<string, string> = {
   226→    FREE: "🟢 Gratuites", FIXED_AMOUNT: "🔵 Montant fixe",
   227→    VARIABLE_BASED: "🟠 Taux variable", MATRIX_BASED: "🟣 Matrice",
   228→  };
   229→
   230→  // ── Render ─────────────────────────────────────────────────────
   231→  if (loading) return <div className="space-y-4">{Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12 rounded-lg" />)}</div>;
   232→
   233→  return (
   234→    <div>
   235→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 mb-6">
   236→        <h1 className="text-2xl font-bold">Offres</h1>
   237→        <Button onClick={openCreate} className="bg-[#B9E54D] text-black hover:bg-[#a5d044]"><Plus className="h-4 w-4 mr-1" /> Ajouter</Button>
   238→      </div>
   239→
   240→      {/* Filters */}
   241→      <div className="flex flex-col sm:flex-row gap-3 mb-4">
   242→        <div className="relative flex-1">
   243→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   244→          <Input placeholder="Rechercher..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-9" />
   245→        </div>
   246→        <Select value={filterInsurer} onValueChange={(v) => setFilterInsurer(v === "all" ? "" : v)}>
   247→          <SelectTrigger className="w-full sm:w-48"><SelectValue placeholder="Assureur" /></SelectTrigger>
   248→          <SelectContent>
   249→            <SelectItem value="all">Tous les assureurs</SelectItem>
   250→            {insurers.map((i) => <SelectItem key={i.id} value={i.id}>{i.name}</SelectItem>)}
   251→          </SelectContent>
   252→        </Select>
   253→        <Select value={filterType} onValueChange={(v) => setFilterType(v === "all" ? "" : v)}>
   254→          <SelectTrigger className="w-full sm:w-40"><SelectValue placeholder="Type" /></SelectTrigger>
   255→          <SelectContent>
   256→            <SelectItem value="all">Tous les types</SelectItem>
   257→            <SelectItem value="tiers">Tiers</SelectItem>
   258→            <SelectItem value="tiers_plus">Tiers+</SelectItem>
   259→            <SelectItem value="tous_risques">Tous Risques</SelectItem>
   260→          </SelectContent>
   261→        </Select>
   262→      </div>
   263→
   264→      {/* Desktop table */}
   265→      <div className="hidden md:block rounded-xl border bg-card overflow-hidden">
   266→        <Table>
   267→          <TableHeader>
   268→            <TableRow className="bg-muted/50">
   269→              <TableHead>Nom</TableHead>
   270→              <TableHead>Assureur</TableHead>
   271→              <TableHead>Type</TableHead>
   272→              <TableHead>Prix de base</TableHead>
   273→              <TableHead>Franchise</TableHead>
   274→              <TableHead className="text-center">Garanties</TableHead>
   275→              <TableHead className="text-center">Statut</TableHead>
   276→              <TableHead className="w-10"></TableHead>
   277→            </TableRow>
   278→          </TableHeader>
   279→          <TableBody>
   280→            {filtered.map((o) => (
   281→              <TableRow key={o.id}>
   282→                <TableCell className="font-medium">{o.name}</TableCell>
   283→                <TableCell className="text-sm">{o.insurer.name}</TableCell>
   284→                <TableCell><Badge className={coverageColors[o.coverageType] || ""}>{coverageLabels[o.coverageType] || o.coverageType}</Badge></TableCell>
   285→                <TableCell className="text-sm font-mono">{fmtPrice(o.basePrice)}</TableCell>
   286→                <TableCell className="text-sm">{o.deductible ? fmtPrice(o.deductible) : "—"}</TableCell>
   287→                <TableCell className="text-center">
   288→                  <Button variant="ghost" size="sm" className="text-xs" onClick={() => openGuaranteeDialog(o)}>
   289→                    <ShieldCheck className="h-3.5 w-3.5 mr-1" />{o.guaranteeLinks.filter((l) => l.isIncluded).length}
   290→                  </Button>
   291→                </TableCell>
   292→                <TableCell className="text-center"><Switch checked={o.isActive} onCheckedChange={() => handleToggleActive(o)} /></TableCell>
   293→                <TableCell>
   294→                  <DropdownMenu>
   295→                    <DropdownMenuTrigger asChild><Button variant="ghost" size="icon" className="h-8 w-8"><MoreHorizontal className="h-4 w-4" /></Button></DropdownMenuTrigger>
   296→                    <DropdownMenuContent align="end">
   297→                      <DropdownMenuItem onClick={() => openDetail(o)}><Eye className="h-4 w-4 mr-2" /> Voir</DropdownMenuItem>
   298→                      <DropdownMenuItem onClick={() => openGuaranteeDialog(o)}><ShieldCheck className="h-4 w-4 mr-2" /> Gérer les garanties</DropdownMenuItem>
   299→                      <DropdownMenuItem onClick={() => openEdit(o)}><Pencil className="h-4 w-4 mr-2" /> Modifier</DropdownMenuItem>
   300→                      <DropdownMenuItem className="text-destructive" onClick={() => setDeleteId(o.id)}><Trash2 className="h-4 w-4 mr-2" /> Supprimer</DropdownMenuItem>
   301→                    </DropdownMenuContent>
   302→                  </DropdownMenu>
   303→                </TableCell>
   304→              </TableRow>
   305→            ))}
   306→          </TableBody>
   307→        </Table>
   308→        {filtered.length === 0 && <p className="text-center py-8 text-muted-foreground text-sm">Aucune offre trouvée.</p>}
   309→      </div>
   310→
   311→      {/* Mobile cards */}
   312→      <div className="md:hidden space-y-3">
   313→        {filtered.map((o) => (
   314→          <Card key={o.id} className="p-4">
   315→            <div className="flex items-start justify-between mb-2">
   316→              <div>
   317→                <p className="font-semibold">{o.name}</p>
   318→                <p className="text-xs text-muted-foreground">{o.insurer.name}</p>
   319→              </div>
   320→              <Switch checked={o.isActive} onCheckedChange={() => handleToggleActive(o)} />
   321→            </div>
   322→            <div className="flex flex-wrap gap-2 text-xs mb-2">
   323→              <Badge className={coverageColors[o.coverageType] || ""}>{coverageLabels[o.coverageType]}</Badge>
   324→              <Badge variant="outline">{o.guaranteeLinks.filter((l) => l.isIncluded).length} garanties</Badge>
   325→            </div>
   326→            <p className="text-sm font-mono font-medium mb-3">{fmtPrice(o.basePrice)}</p>
   327→            <div className="flex gap-2">
   328→              <Button variant="outline" size="sm" className="flex-1" onClick={() => openDetail(o)}><Eye className="h-3.5 w-3.5 mr-1" /> Voir</Button>
   329→              <Button variant="outline" size="sm" onClick={() => openEdit(o)}><Pencil className="h-3.5 w-3.5" /></Button>
   330→              <Button variant="outline" size="sm" className="text-destructive" onClick={() => setDeleteId(o.id)}><Trash2 className="h-3.5 w-3.5" /></Button>
   331→            </div>
   332→          </Card>
   333→        ))}
   334→      </div>
   335→
   336→      {/* ── Detail Dialog ──────────────────────────────────────────── */}
   337→      <Dialog open={!!detailOffer || detailLoading} onOpenChange={() => setDetailOffer(null)}>
   338→        <DialogContent className="max-w-2xl max-h-[90vh]">
   339→          {detailLoading ? (
   340→            <div className="flex items-center justify-center py-12"><Loader2 className="h-6 w-6 animate-spin" /></div>
   341→          ) : detailOffer ? (
   342→            <>
   343→              <DialogHeader>
   344→                <div className="flex items-center gap-3 flex-wrap">
   345→                  <DialogTitle className="text-xl">{detailOffer.name}</DialogTitle>
   346→                  <Badge className={coverageColors[detailOffer.coverageType] || ""}>
   347→                    {coverageLabels[detailOffer.coverageType] || detailOffer.coverageType}
   348→                  </Badge>
   349→                  <Badge className="bg-[#B9E54D]/20 text-black">{detailOffer.insurer.name}</Badge>
   350→                </div>
   351→                {detailOffer.description && (
   352→                  <DialogDescription>{detailOffer.description}</DialogDescription>
   353→                )}
   354→              </DialogHeader>
   355→
   356→              <ScrollArea className="max-h-[65vh] pr-2">
   357→                <div className="space-y-6 pb-4">
   358→                  {/* Pricing */}
   359→                  <div className="space-y-2">
   360→                    <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide">Tarification</h3>
   361→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   362→                      <Card><CardContent className="p-3">
   363→                        <p className="text-[10px] text-muted-foreground uppercase">Prix de base</p>
   364→                        <p className="font-bold text-sm font-mono">{fmtPrice(detailOffer.basePrice)}</p>
   365→                      </CardContent></Card>
   366→                      <Card><CardContent className="p-3">
   367→                        <p className="text-[10px] text-muted-foreground uppercase">Prix annuel</p>
   368→                        <p className="font-bold text-sm font-mono">{detailOffer.annualPrice ? fmtPrice(detailOffer.annualPrice) : "—"}</p>
   369→                      </CardContent></Card>
   370→                      <Card><CardContent className="p-3">
   371→                        <p className="text-[10px] text-muted-foreground uppercase">Franchise</p>
   372→                        <p className="font-bold text-sm font-mono">{detailOffer.deductible ? fmtPrice(detailOffer.deductible) : "—"}</p>
   373→                      </CardContent></Card>
   374→                      <Card><CardContent className="p-3">
   375→                        <p className="text-[10px] text-muted-foreground uppercase">Couverture max</p>
   376→                        <p className="font-bold text-sm font-mono">{detailOffer.maxCoverage ? fmtPrice(detailOffer.maxCoverage) : "—"}</p>
   377→                      </CardContent></Card>
   378→                    </div>
   379→                  </div>
   380→
   381→                  {/* Features */}
   382→                  {detailOffer.features.length > 0 && (
   383→                    <div className="space-y-2">
   384→                      <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide">Caractéristiques</h3>
   385→                      <div className="space-y-1">
   386→                        {detailOffer.features.map((f, idx) => (
   387→                          <div key={idx} className="flex items-center gap-2 text-sm">
   388→                            <Check className="h-3.5 w-3.5 text-emerald-600 shrink-0" />
   389→                            <span>{f}</span>
   390→                          </div>
   391→                        ))}
   392→                      </div>
   393→                    </div>
   394→                  )}
   395→
   396→                  {/* Conditions */}
   397→                  {detailOffer.conditions && (
   398→                    <div className="space-y-2">
   399→                      <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide">Conditions</h3>
   400→                      <p className="text-sm text-muted-foreground whitespace-pre-wrap">{detailOffer.conditions}</p>
   401→                    </div>
   402→                  )}
   403→
   404→                  <Separator />
   405→
   406→                  {/* Included guarantees */}
   407→                  <div className="space-y-2">
   408→                    <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide">
   409→                      Garanties incluses ({detailOffer.guaranteeLinks.filter((l) => l.isIncluded).length})
   410→                    </h3>
   411→                    {detailOffer.guaranteeLinks.filter((l) => l.isIncluded).length > 0 ? (
   412→                      <div className="rounded-xl border overflow-hidden">
   413→                        <Table>
   414→                          <TableHeader>
   415→                            <TableRow className="bg-muted/50">
   416→                              <TableHead>Garantie</TableHead>
   417→                              <TableHead>Méthode</TableHead>
   418→                            </TableRow>
   419→                          </TableHeader>
   420→                          <TableBody>
   421→                            {detailOffer.guaranteeLinks.filter((l) => l.isIncluded).map((gl) => (
   422→                              <TableRow key={gl.id}>
   423→                                <TableCell className="font-medium text-sm">{gl.guarantee.name}</TableCell>
   424→                                <TableCell><CalcBadge method={gl.guarantee.calcMethod} /></TableCell>
   425→                              </TableRow>
   426→                            ))}
   427→                          </TableBody>
   428→                        </Table>
   429→                      </div>
   430→                    ) : (
   431→                      <p className="text-sm text-muted-foreground">Aucune garantie incluse.</p>
   432→                    )}
   433→                  </div>
   434→                </div>
   435→              </ScrollArea>
   436→
   437→              <Separator />
   438→              <DialogFooter>
   439→                <Button variant="outline" onClick={() => { setDetailOffer(null); openEdit(detailOffer); }}>
   440→                  <Pencil className="h-4 w-4 mr-1" /> Modifier
   441→                </Button>
   442→              </DialogFooter>
   443→            </>
   444→          ) : null}
   445→        </DialogContent>
   446→      </Dialog>
   447→
   448→      {/* ── Form Dialog ──────────────────────────────────────────── */}
   449→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   450→        <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   451→          <DialogHeader><DialogTitle>{editing ? "Modifier l'offre" : "Nouvelle offre"}</DialogTitle><DialogDescription>Remplissez les informations de l'offre.</DialogDescription></DialogHeader>
   452→          <div className="grid gap-4 py-4">
   453→            <div className="grid gap-2">
   454→              <Label>Assureur *</Label>
   455→              <Select value={form.insurerId} onValueChange={(v) => setForm({ ...form, insurerId: v })}>
   456→                <SelectTrigger><SelectValue placeholder="Sélectionner..." /></SelectTrigger>
   457→                <SelectContent>{insurers.map((i) => <SelectItem key={i.id} value={i.id}>{i.name}</SelectItem>)}</SelectContent>
   458→              </Select>
   459→            </div>
   460→            <div className="grid grid-cols-2 gap-4">
   461→              <div className="grid gap-2"><Label>Nom *</Label><Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} /></div>
   462→              <div className="grid gap-2">
   463→                <Label>Type de couverture *</Label>
   464→                <Select value={form.coverageType} onValueChange={(v) => setForm({ ...form, coverageType: v })}>
   465→                  <SelectTrigger><SelectValue /></SelectTrigger>
   466→                  <SelectContent>
   467→                    <SelectItem value="tiers">Tiers</SelectItem>
   468→                    <SelectItem value="tiers_plus">Tiers+</SelectItem>
   469→                    <SelectItem value="tous_risques">Tous Risques</SelectItem>
   470→                  </SelectContent>
   471→                </Select>
   472→              </div>
   473→            </div>
   474→            <div className="grid gap-2"><Label>Description</Label><Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={2} /></div>
   475→            <div className="grid grid-cols-2 gap-4">
   476→              <div className="grid gap-2"><Label>Prix de base (FCFA/mois) *</Label><Input type="number" value={form.basePrice} onChange={(e) => setForm({ ...form, basePrice: parseFloat(e.target.value) || 0 })} /></div>
   477→              <div className="grid gap-2"><Label>Prix annuel (FCFA)</Label><Input type="number" value={form.annualPrice ?? ""} onChange={(e) => setForm({ ...form, annualPrice: e.target.value ? parseFloat(e.target.value) : null })} /></div>
   478→            </div>
   479→            <div className="grid grid-cols-2 gap-4">
   480→              <div className="grid gap-2"><Label>Franchise (FCFA)</Label><Input type="number" value={form.deductible ?? ""} onChange={(e) => setForm({ ...form, deductible: e.target.value ? parseFloat(e.target.value) : null })} /></div>
   481→              <div className="grid gap-2"><Label>Couverture max (FCFA)</Label><Input type="number" value={form.maxCoverage ?? ""} onChange={(e) => setForm({ ...form, maxCoverage: e.target.value ? parseFloat(e.target.value) : null })} /></div>
   482→            </div>
   483→            <div className="grid gap-2"><Label>Caractéristiques (JSON)</Label><Textarea value={form.features} onChange={(e) => setForm({ ...form, features: e.target.value })} rows={3} className="font-mono text-xs" /></div>
   484→            <div className="grid gap-2"><Label>Conditions</Label><Textarea value={form.conditions} onChange={(e) => setForm({ ...form, conditions: e.target.value })} rows={2} /></div>
   485→            <div className="flex items-center justify-between">
   486→              <Label>Active</Label>
   487→              <Switch checked={form.isActive} onCheckedChange={(v) => setForm({ ...form, isActive: v })} />
   488→            </div>
   489→          </div>
   490→          <DialogFooter>
   491→            <Button variant="outline" onClick={() => setFormOpen(false)}>Annuler</Button>
   492→            <Button onClick={handleSave} disabled={saving} className="bg-[#B9E54D] text-black hover:bg-[#a5d044]">{saving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}{editing ? "Modifier" : "Créer"}</Button>
   493→          </DialogFooter>
   494→        </DialogContent>
   495→      </Dialog>
   496→
   497→      {/* ── Delete Dialog ────────────────────────────────────────── */}
   498→      <AlertDialog open={!!deleteId} onOpenChange={() => setDeleteId(null)}>
   499→        <AlertDialogContent>
   500→          <AlertDialogHeader><AlertDialogTitle>Supprimer cette offre ?</AlertDialogTitle><AlertDialogDescription>Cette action est irréversible et supprimera les devis associés.</AlertDialogDescription></AlertDialogHeader>
   501→          <AlertDialogFooter>
   502→            <AlertDialogCancel>Annuler</AlertDialogCancel>
   503→            <AlertDialogAction onClick={handleDelete} disabled={deleting} className="bg-destructive text-destructive-foreground">{deleting && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}Supprimer</AlertDialogAction>
   504→          </AlertDialogFooter>
   505→        </AlertDialogContent>
   506→      </AlertDialog>
   507→
   508→      {/* ── Guarantees Dialog ────────────────────────────────────── */}
   509→      <Dialog open={guaranteeDialogOpen} onOpenChange={setGuaranteeDialogOpen}>
   510→        <DialogContent className="max-w-2xl max-h-[90vh]">
   511→          <DialogHeader>
   512→            <DialogTitle>Garanties de « {selectedOffer?.name} »</DialogTitle>
   513→            <DialogDescription>{selectedOffer?.insurer.name} — {coverageLabels[selectedOffer?.coverageType || ""]}</DialogDescription>
   514→          </DialogHeader>
   515→          <ScrollArea className="h-[60vh] pr-4">
   516→            <div className="space-y-4">
   517→              {Object.entries(groupedGuarantees)
   518→                .sort(([a], [b]) => (a === "FREE" ? -1 : b === "FREE" ? 1 : 0))
   519→                .map(([method, guarantees]) => (
   520→                  <div key={method}>
   521→                    <p className="text-xs font-semibold text-muted-foreground uppercase mb-2">{groupLabels[method] || method} ({guarantees.length})</p>
   522→                    <div className="space-y-1">
   523→                      {guarantees.map((g) => {
   524→                        const idx = guaranteeChecks.findIndex((c) => c.guaranteeId === g.id);
   525→                        const checked = idx >= 0 ? guaranteeChecks[idx].isIncluded : false;
   526→                        return (
   527→                          <label key={g.id} className={`flex items-center gap-3 rounded-lg border p-3 cursor-pointer transition-colors ${checked ? "bg-card border-primary/30" : "bg-muted/30"}`}>
   528→                            <Checkbox checked={checked} onCheckedChange={() => idx >= 0 && toggleCheck(idx)} />
   529→                            <span className="text-sm flex-1">{g.name}</span>
   530→                            <CalcBadge method={g.calcMethod} />
   531→                          </label>
   532→                        );
   533→                      })}
   534→                    </div>
   535→                  </div>
   536→                ))}
   537→            </div>
   538→          </ScrollArea>
   539→          <Separator />
   540→          <DialogFooter>
   541→            <Button variant="outline" onClick={() => setGuaranteeDialogOpen(false)}>Annuler</Button>
   542→            <Button onClick={saveGuarantees} disabled={savingGuarantees} className="bg-[#B9E54D] text-black hover:bg-[#a5d044]">
   543→              {savingGuarantees && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}Enregistrer
   544→            </Button>
   545→          </DialogFooter>
   546→        </DialogContent>
   547→      </Dialog>
   548→    </div>
   549→  );
