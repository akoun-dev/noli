   129→   129→// ── Helpers ──────────────────────────────────────────────────────
   130→   130→function displayRateOrPrice(g: Guarantee): string {
   131→   131→  if (g.calcMethod === "FREE") return "Gratuit";
   132→   132→  if (g.calcMethod === "FIXED_AMOUNT") return g.fixedPrice ? fmtPrice(g.fixedPrice) : "—";
   133→   133→  if (g.rateConditions && g.rateConditions.length > 0) {
   134→   134→    return g.rateConditions.map((rc) => `${rc.condition}: ${rc.rate}%`).join(" / ");
   135→   135→  }
   136→   136→  if (g.rate != null) return `${g.rate} %`;
   137→   137→  return "—";
   138→   138→}
   139→   139→
   140→   140→function displayCapital(g: Guarantee): string {
   141→   141→  if (!g.capital) return "—";
   142→   142→  return Object.entries(g.capital).map(([k, v]) => `${k}: ${v}`).join(", ");
   143→   143→}
   144→   144→
   145→   145→function displayFranchise(g: Guarantee): string {
   146→   146→  if (!g.franchise) return "—";
   147→   147→  if (g.franchise.description) return g.franchise.description as string;
   148→   148→  const parts: string[] = [];
   149→   149→  if (g.franchise.percent) parts.push(`${g.franchise.percent}%`);
   150→   150→  if (g.franchise.min) parts.push(`min ${fmtPrice(g.franchise.min as number)}`);
   151→   151→  return parts.join(", ") || "—";
   152→   152→}
