   138→/* ── Component ─────────────────────────────────────────────── */
   139→export function CoveragesTab() {
   140→  const { toast } = useToast();
   141→  const [items, setItems] = useState<Coverage[]>([]);
   142→  const [insurers, setInsurers] = useState<InsurerOption[]>([]);
   143→  const [categories, setCategories] = useState<CatOption[]>([]);
   144→  const [insuranceCategories, setInsuranceCategories] = useState<InsCatOption[]>([]);
   145→  const [search, setSearch] = useState("");
   146→  const [filterInsurer, setFilterInsurer] = useState("");
   147→  const [filterCategory, setFilterCategory] = useState("");
   148→  const [filterCalcType, setFilterCalcType] = useState("");
   149→  const [loading, setLoading] = useState(true);
   150→  const [dialogOpen, setDialogOpen] = useState(false);
   151→  const [editing, setEditing] = useState<Coverage | null>(null);
   152→  const [step, setStep] = useState(1);
   153→  const [step1, setStep1] = useState<Step1Data>(emptyStep1);
   154→  const [calcType, setCalcType] = useState<CalculationType | "">("");
   155→  const [metadata, setMetadata] = useState<Record<string, unknown>>({});
   156→  const [saving, setSaving] = useState(false);
   157→  const [deleteId, setDeleteId] = useState<string | null>(null);
   158→
   159→  // Tariff rules panel
   160→  const [selectedId, setSelectedId] = useState<string | null>(null);
   161→  const [tariffRules, setTariffRules] = useState<TariffRule[]>([]);
   162→  const [trLoading, setTrLoading] = useState(false);
   163→  const [trDialogOpen, setTrDialogOpen] = useState(false);
   164→  const [trEditing, setTrEditing] = useState<TariffRule | null>(null);
   165→  const [trDeleteId, setTrDeleteId] = useState<string | null>(null);
   166→  const [trForm, setTrForm] = useState({
   167→    fuelType: "", minFiscalPower: "", maxFiscalPower: "", baseRate: "", fixedAmount: "", minAmount: "", maxAmount: "", formulaName: "",
   168→  });
   169→  const [trSaving, setTrSaving] = useState(false);
   170→
   171→  const fetchItems = useCallback(async () => {
   172→    try {
   173→      const params = new URLSearchParams();
   174→      if (search) params.set("search", search);
   175→      if (filterInsurer) params.set("insurerId", filterInsurer);
   176→      if (filterCategory) params.set("categoryId", filterCategory);
   177→      if (filterCalcType) params.set("calculationType", filterCalcType);
   178→      const res = await fetch(`/api/admin/coverages?${params}`);
   179→      if (!res.ok) throw new Error();
   180→      setItems(await res.json());
   181→    } catch {
   182→      toast({ title: "Erreur", description: "Impossible de charger les garanties", variant: "destructive" });
   183→    } finally {
   184→      setLoading(false);
   185→    }
   186→  }, [search, filterInsurer, filterCategory, filterCalcType, toast]);
   187→
   188→  const fetchLookups = useCallback(async () => {
   189→    try {
   190→      const [insRes, catRes, insCatRes] = await Promise.all([
   191→        fetch("/api/admin/insurers"),
   192→        fetch("/api/admin/coverage-categories"),
   193→        fetch("/api/admin/insurance-categories"),
   194→      ]);
   195→      if (insRes.ok) setInsurers(await insRes.json());
   196→      if (catRes.ok) setCategories(await catRes.json());
   197→      if (insCatRes.ok) setInsuranceCategories(await insCatRes.json());
   198→    } catch { /* silent */ }
   199→  }, []);
   200→
   201→  useEffect(() => { fetchLookups(); }, [fetchLookups]);
   202→  useEffect(() => { setLoading(true); fetchItems(); }, [fetchItems]);
   203→
   204→  const fetchTariffRules = useCallback(async (coverageId: string) => {
   205→    setTrLoading(true);
   206→    try {
   207→      const res = await fetch(`/api/admin/coverages/${coverageId}/tariff-rules`);
   208→      if (!res.ok) throw new Error();
   209→      setTariffRules(await res.json());
   210→    } catch {
   211→      toast({ title: "Erreur", description: "Impossible de charger les règles", variant: "destructive" });
   212→    } finally {
   213→      setTrLoading(false);
   214→    }
   215→  }, [toast]);
   216→
   217→  const handleRowClick = (id: string) => {
   218→    if (selectedId === id) {
   219→      setSelectedId(null);
   220→      return;
   221→    }
   222→    setSelectedId(id);
   223→    fetchTariffRules(id);
   224→  };
   225→
   226→  /* ── Wizard ─────────────────────────────────────────────── */
   227→  const openCreate = () => {
   228→    setEditing(null);
   229→    setStep1(emptyStep1);
   230→    setCalcType("");
   231→    setMetadata({});
   232→    setStep(1);
   233→    setDialogOpen(true);
   234→  };
   235→
   236→  const openEdit = (item: Coverage) => {
   237→    setEditing(item);
   238→    setStep1({
   239→      name: item.name,
   240→      insuranceCategoryId: "", // will be resolved after insuranceCategories load
   241→      description: item.description ?? "",
   242→      insurerId: item.insurerId,
   243→      categoryId: item.categoryId ?? "",
   244→      isMandatory: item.isMandatory,
   245→      displayOrder: item.displayOrder.toString(),
   246→      isActive: item.isActive,
   247→    });
   248→    setCalcType(item.calculationType as CalculationType);
   249→    setMetadata(item.metadata);
   250→    setStep(1);
   251→    setDialogOpen(true);
   252→  };
   253→
   254→  const buildMetadataFromStep3 = (): Record<string, unknown> => {
   255→    if (calcType === "FREE") return {};
   256→    if (calcType === "FIXED_AMOUNT") {
   257→      return { fixedAmount: (metadata.fixedAmount as number) || 0 };
   258→    }
   259→    if (calcType === "VARIABLE_BASED") {
   260→      const m: Record<string, unknown> = { variable: (metadata.variable as string) || "VN" };
   261→      if (metadata.hasConditional) {
   262→        m.hasConditional = true;
   263→        m.threshold = Number(metadata.threshold) || 0;
   264→        m.rateBelow = Number(metadata.rateBelow) || 0;
   265→        m.rateAbove = Number(metadata.rateAbove) || 0;
   266→      } else {
   267→        m.rate = Number(metadata.rate) || 0;
   268→      }
   269→      if (metadata.franchiseEnabled) {
   270→        m.franchiseEnabled = true;
   271→        m.franchisePercent = Number(metadata.franchisePercent) || 0;
   272→        m.franchiseMin = Number(metadata.franchiseMin) || 0;
   273→      }
   274→      return m;
   275→    }
   276→    if (calcType === "MATRIX_BASED") {
   277→      return { matrixType: (metadata.matrixType as string) || "FISCAL_POWER" };
   278→    }
   279→    return {};
   280→  };
   281→
   282→  const handleSave = async () => {
   283→    if (!step1.name.trim() || !step1.insurerId || !calcType) {
   284→      toast({ title: "Erreur", description: "Veuillez remplir tous les champs requis", variant: "destructive" });
   285→      return;
   286→    }
   287→    setSaving(true);
   288→    try {
   289→      const url = editing ? `/api/admin/coverages/${editing.id}` : "/api/admin/coverages";
   290→      const method = editing ? "PUT" : "POST";
   291→      // Resolve insurance category name for the `type` field
   292→      const insCat = insuranceCategories.find((c) => c.id === step1.insuranceCategoryId);
   293→      const body = {
   294→        name: step1.name,
   295→        type: insCat?.name || "",
   296→        description: step1.description,
   297→        insurerId: step1.insurerId,
   298→        categoryId: step1.categoryId || null,
   299→        isMandatory: step1.isMandatory,
   300→        displayOrder: parseInt(step1.displayOrder, 10) || 0,
   301→        isActive: step1.isActive,
   302→        calculationType: calcType,
   303→        metadata: buildMetadataFromStep3(),
   304→      };
   305→      const res = await fetch(url, { method, headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
   306→      if (!res.ok) {
   307→        const data = await res.json();
   308→        throw new Error(data.error || "Erreur");
   309→      }
   310→      toast({ title: editing ? "Garantie modifiée" : "Garantie créée" });
   311→      setDialogOpen(false);
   312→      fetchItems();
   313→    } catch (e) {
   314→      toast({ title: "Erreur", description: e instanceof Error ? e.message : "Erreur", variant: "destructive" });
   315→    } finally {
   316→      setSaving(false);
   317→    }
   318→  };
   319→
   320→  const handleDelete = async () => {
   321→    if (!deleteId) return;
   322→    try {
   323→      const res = await fetch(`/api/admin/coverages/${deleteId}`, { method: "DELETE" });
   324→      if (!res.ok) throw new Error();
   325→      toast({ title: "Garantie supprimée" });
   326→      setDeleteId(null);
   327→      if (selectedId === deleteId) setSelectedId(null);
   328→      fetchItems();
   329→    } catch {
   330→      toast({ title: "Erreur", variant: "destructive" });
   331→    }
   332→  };
   333→
   334→  /* ── Tariff Rule CRUD ────────────────────────────────────── */
   335→  const openTrCreate = () => {
   336→    setTrEditing(null);
   337→    setTrForm({ fuelType: "", minFiscalPower: "", maxFiscalPower: "", baseRate: "", fixedAmount: "", minAmount: "", maxAmount: "", formulaName: "" });
   338→    setTrDialogOpen(true);
   339→  };
   340→
   341→  const openTrEdit = (rule: TariffRule) => {
   342→    setTrEditing(rule);
   343→    setTrForm({
   344→      fuelType: rule.fuelType ?? "", minFiscalPower: rule.minFiscalPower?.toString() ?? "", maxFiscalPower: rule.maxFiscalPower?.toString() ?? "",
   345→      baseRate: rule.baseRate?.toString() ?? "", fixedAmount: rule.fixedAmount?.toString() ?? "", minAmount: rule.minAmount?.toString() ?? "",
   346→      maxAmount: rule.maxAmount?.toString() ?? "", formulaName: rule.formulaName ?? "",
   347→    });
   348→    setTrDialogOpen(true);
   349→  };
   350→
   351→  const saveTr = async () => {
   352→    if (!selectedId) return;
   353→    setTrSaving(true);
   354→    try {
   355→      const body = {
   356→        fuelType: trForm.fuelType || null,
   357→        minFiscalPower: trForm.minFiscalPower ? parseInt(trForm.minFiscalPower, 10) : null,
   358→        maxFiscalPower: trForm.maxFiscalPower ? parseInt(trForm.maxFiscalPower, 10) : null,
   359→        baseRate: trForm.baseRate ? parseFloat(trForm.baseRate) : null,
   360→        fixedAmount: trForm.fixedAmount ? parseFloat(trForm.fixedAmount) : null,
   361→        minAmount: trForm.minAmount ? parseFloat(trForm.minAmount) : null,
   362→        maxAmount: trForm.maxAmount ? parseFloat(trForm.maxAmount) : null,
   363→        formulaName: trForm.formulaName || null,
   364→      };
   365→      const res = await fetch(`/api/admin/coverages/${selectedId}/tariff-rules`, {
   366→        method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body),
   367→      });
   368→      if (!res.ok) throw new Error();
   369→      toast({ title: "Règle ajoutée" });
   370→      setTrDialogOpen(false);
   371→      fetchTariffRules(selectedId);
   372→    } catch {
   373→      toast({ title: "Erreur", variant: "destructive" });
   374→    } finally {
   375→      setTrSaving(false);
   376→    }
   377→  };
   378→
   379→  const handleTrDelete = async () => {
   380→    if (!trDeleteId || !selectedId) return;
   381→    try {
   382→      const res = await fetch(`/api/admin/tariff-rules/${trDeleteId}`, { method: "DELETE" });
   383→      if (!res.ok) throw new Error();
   384→      toast({ title: "Règle tarifaire supprimée" });
   385→      setTrDeleteId(null);
   386→      fetchTariffRules(selectedId);
   387→    } catch {
   388→      toast({ title: "Erreur", variant: "destructive" });
   389→    }
   390→  };
   391→
   392→  /* ── Render Step 3 ──────────────────────────────────────── */
   393→  const renderStep3 = () => {
   394→    if (calcType === "FREE") {
   395→      return (
   396→        <div className="rounded-lg bg-emerald-50 border border-emerald-200 p-6 text-center">
   397→          <p className="text-emerald-700 font-medium text-lg">🟢 Gratuit</p>
   398→          <p className="text-emerald-600 text-sm mt-1">Aucune configuration supplémentaire requise.</p>
   399→        </div>
   400→      );
   401→    }
   402→    if (calcType === "FIXED_AMOUNT") {
   403→      return (
   404→        <div className="space-y-4">
   405→          <Label>Montant fixe (FCFA) *</Label>
   406→          <Input className="w-full" type="number" value={(metadata.fixedAmount as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, fixedAmount: parseFloat(e.target.value) || 0 })} placeholder="Ex: 50000" />
   407→        </div>
   408→      );
   409→    }
   410→    if (calcType === "VARIABLE_BASED") {
   411→      return (
   412→        <div className="space-y-4">
   413→          <div>
   414→            <Label>Variable</Label>
   415→            <Select value={(metadata.variable as string) || "VN"} onValueChange={(v) => setMetadata({ ...metadata, variable: v })}>
   416→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   417→              <SelectContent>
   418→                <SelectItem value="VN">Valeur Neuve (VN)</SelectItem>
   419→                <SelectItem value="VN_REPLACEMENT_VALUE">VN / Valeur de remplacement</SelectItem>
   420→              </SelectContent>
   421→            </Select>
   422→          </div>
   423→          <div className="flex items-center gap-2">
   424→            <Checkbox id="conditional" checked={!!metadata.hasConditional} onCheckedChange={(v) => setMetadata({ ...metadata, hasConditional: !!v })} />
   425→            <Label htmlFor="conditional">Taux conditionné par seuil</Label>
   426→          </div>
   427→          {metadata.hasConditional ? (
   428→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 rounded-lg border p-4 bg-muted/50">
   429→              <div><Label className="text-xs">Seuil (FCFA)</Label><Input className="w-full" type="number" value={(metadata.threshold as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, threshold: parseFloat(e.target.value) || 0 })} /></div>
   430→              <div><Label className="text-xs">Taux sous (%)</Label><Input className="w-full" type="number" step="0.01" value={(metadata.rateBelow as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rateBelow: parseFloat(e.target.value) || 0 })} /></div>
   431→              <div><Label className="text-xs">Taux au-dessus (%)</Label><Input className="w-full" type="number" step="0.01" value={(metadata.rateAbove as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rateAbove: parseFloat(e.target.value) || 0 })} /></div>
   432→            </div>
   433→          ) : (
   434→            <div className="w-full"><Label>Taux (%)</Label><Input className="w-full" type="number" step="0.01" value={(metadata.rate as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rate: parseFloat(e.target.value) || 0 })} /></div>
   435→          )}
   436→          <Separator />
   437→          <div className="flex items-center gap-2">
   438→            <Checkbox id="franchise" checked={!!metadata.franchiseEnabled} onCheckedChange={(v) => setMetadata({ ...metadata, franchiseEnabled: !!v })} />
   439→            <Label htmlFor="franchise">Appliquer une franchise</Label>
   440→          </div>
   441→          {metadata.franchiseEnabled && (
   442→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 rounded-lg border p-4 bg-muted/50">
   443→              <div><Label className="text-xs">Franchise (%)</Label><Input className="w-full" type="number" step="0.01" value={(metadata.franchisePercent as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, franchisePercent: parseFloat(e.target.value) || 0 })} /></div>
   444→              <div><Label className="text-xs">Montant min (FCFA)</Label><Input className="w-full" type="number" value={(metadata.franchiseMin as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, franchiseMin: parseFloat(e.target.value) || 0 })} /></div>
   445→            </div>
   446→          )}
   447→        </div>
   448→      );
   449→    }
   450→    if (calcType === "MATRIX_BASED") {
   451→      return (
   452→        <div className="space-y-4">
   453→          <div>
   454→            <Label>Type de matrice</Label>
   455→            <Select value={(metadata.matrixType as string) || "FISCAL_POWER"} onValueChange={(v) => setMetadata({ ...metadata, matrixType: v })}>
   456→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   457→              <SelectContent>
   458→                <SelectItem value="FISCAL_POWER">Puissance fiscale</SelectItem>
   459→                <SelectItem value="FORMULA">Formule (IC/IPT)</SelectItem>
   460→                <SelectItem value="TIERCE_COMPLETE">Tierce complète</SelectItem>
   461→                <SelectItem value="TIERCE_COLLISION">Tierce collision</SelectItem>
   462→                <SelectItem value="FUEL_TYPE">Type carburant</SelectItem>
   463→                <SelectItem value="SEATS">Nombre de places</SelectItem>
   464→              </SelectContent>
   465→            </Select>
   466→          </div>
   467→          <p className="text-sm text-muted-foreground">
   468→            Les règles tarifaires associées définissent les valeurs de la matrice. Utilisez la section ci-dessous pour les gérer.
   469→          </p>
   470→        </div>
   471→      );
   472→    }
   473→    return null;
   474→  };
   475→
   476→  const selectedCoverage = items.find((c) => c.id === selectedId);
   477→
   478→  return (
