    36→    36→// ── Types ────────────────────────────────────────────────────────
    37→    37→interface Offer {

    53→    53→  tiers: "bg-slate-100 text-slate-800",
    54→    54→  tiers_plus: "bg-amber-100 text-amber-800",

    56→    56→};

    62→    62→};
    63→    63→
    64→    64→const emptyOffer = {
    65→    65→  insurerId: "", name: "", coverageType: "tiers" as string, description: "",
    66→    66→  basePrice: 0, annualPrice: null as number | null, deductible: null as number | null,
    67→    67→  maxCoverage: null as number | null, features: "[]" as string, conditions: "", isActive: true,
    68→    68→};
    69→    69→
    70→    70→export function OffresTab() {
    71→    71→  const { toast } = useToast();
    72→    72→  const [offers, setOffers] = useState<Offer[]>([]);
    73→    73→  const [insurers, setInsurers] = useState<InsurerMini[]>([]);
    74→    74→  const [loading, setLoading] = useState(true);
    75→    75→  const [search, setSearch] = useState("");
    76→    76→  const [filterInsurer, setFilterInsurer] = useState("");
    77→    77→  const [filterType, setFilterType] = useState("");
    78→    78→  const [formOpen, setFormOpen] = useState(false);
    79→    79→  const [editing, setEditing] = useState<Offer | null>(null);
    80→    80→  const [form, setForm] = useState(emptyOffer);
    81→    81→  const [saving, setSaving] = useState(false);
    82→    82→  const [deleteId, setDeleteId] = useState<string | null>(null);
    83→    83→  const [deleting, setDeleting] = useState(false);
    84→    84→
    85→    85→  // Guarantee dialog
    86→    86→  const [guaranteeDialogOpen, setGuaranteeDialogOpen] = useState(false);
    87→    87→  const [selectedOffer, setSelectedOffer] = useState<Offer | null>(null);
    88→    88→  const [allGuarantees, setAllGuarantees] = useState<GuaranteeMini[]>([]);
    89→    89→  const [guaranteeChecks, setGuaranteeChecks] = useState<{ guaranteeId: string; isIncluded: boolean }[]>([]);
    90→    90→  const [savingGuarantees, setSavingGuarantees] = useState(false);
    91→    91→
    92→    92→  // Detail view
    93→    93→  const [detailOffer, setDetailOffer] = useState<Offer | null>(null);
    94→    94→  const [detailLoading, setDetailLoading] = useState(false);
    95→    95→
    96→    96→  const fetchOffers = useCallback(async () => {
    97→    97→    try {
    98→    98→      const params = new URLSearchParams();
    99→    99→      if (search) params.set("search", search);
   100→   100→      if (filterInsurer) params.set("insurerId", filterInsurer);
   101→   101→      if (filterType) params.set("coverageType", filterType);
   102→   102→      const res = await fetch(`/api/admin/offers?${params}`);
   103→   103→      if (res.ok) setOffers(await res.json());
   104→   104→    } finally { setLoading(false); }
   105→   105→  }, [search, filterInsurer, filterType]);
   106→   106→
   107→   107→  const fetchInsurers = useCallback(async () => {
   108→   108→    const res = await fetch("/api/admin/insurers?active=true");
   109→   109→    if (res.ok) setInsurers(await res.json());
   110→   110→  }, []);
   111→   111→
   112→   112→  useEffect(() => { fetchOffers(); fetchInsurers(); }, [fetchOffers, fetchInsurers]);
   113→   113→
   114→   114→  const filtered = offers;
   115→   115→
   116→   116→  const openCreate = () => { setEditing(null); setForm(emptyOffer); setFormOpen(true); };
   117→   117→  const openEdit = (o: Offer) => {
   118→   118→    setEditing(o);
   119→   119→    setForm({
   120→   120→      insurerId: o.insurerId, name: o.name, coverageType: o.coverageType,
   121→   121→      description: o.description || "", basePrice: o.basePrice,
   122→   122→      annualPrice: o.annualPrice, deductible: o.deductible, maxCoverage: o.maxCoverage,
   123→   123→      features: JSON.stringify(o.features), conditions: o.conditions || "", isActive: o.isActive,
   124→   124→    });
   125→   125→    setFormOpen(true);
   126→   126→  };
   127→   127→
   128→   128→  const handleSave = async () => {
   129→   129→    if (!form.insurerId || !form.name.trim()) { toast({ title: "Assureur et nom obligatoires", variant: "destructive" }); return; }
   130→   130→    setSaving(true);
   131→   131→    try {
   132→   132→      const url = editing ? `/api/admin/offers/${editing.id}` : "/api/admin/offers";
   133→   133→      const res = await fetch(url, {
   134→   134→        method: editing ? "PUT" : "POST",
   135→   135→        headers: { "Content-Type": "application/json" },
   136→   136→        body: JSON.stringify({ ...form, features: JSON.parse(form.features || "[]") }),
   137→   137→      });
   138→   138→      if (!res.ok) { const e = await res.json(); throw new Error(e.error || "Erreur"); }
   139→   139→      toast({ title: editing ? "Offre modifiée" : "Offre créée" });
   140→   140→      setFormOpen(false);
   141→   141→      fetchOffers();
   142→   142→    } catch (err) { toast({ title: (err as Error).message, variant: "destructive" }); }
   143→   143→    finally { setSaving(false); }
   144→   144→  };
   145→   145→
   146→   146→  const handleDelete = async () => {
   147→   147→    if (!deleteId) return;
   148→   148→    setDeleting(true);
   149→   149→    try {
   150→   150→      const res = await fetch(`/api/admin/offers/${deleteId}`, { method: "DELETE" });
   151→   151→      if (!res.ok) throw new Error();
   152→   152→      toast({ title: "Offre supprimée" });
   153→   153→      setDeleteId(null); fetchOffers();
   154→   154→    } catch { toast({ title: "Erreur de suppression", variant: "destructive" }); }
   155→   155→    finally { setDeleting(false); }
   156→   156→  };
   157→   157→
   158→   158→  const handleToggleActive = async (o: Offer) => {
   159→   159→    try {
   160→   160→      await fetch(`/api/admin/offers/${o.id}`, {
   161→   161→        method: "PUT", headers: { "Content-Type": "application/json" },
   162→   162→        body: JSON.stringify({ isActive: !o.isActive }),
   163→   163→      });
   164→   164→      fetchOffers();
   165→   165→    } catch { toast({ title: "Erreur", variant: "destructive" }); }
   166→   166→  };
   167→   167→
   168→   168→  // ── Detail view ────────────────────────────────────────────────
   169→   169→  const openDetail = async (o: Offer) => {
   170→   170→    setDetailLoading(true);
   171→   171→    setDetailOffer(null);
   172→   172→    try {
   173→   173→      const res = await fetch(`/api/admin/offers/${o.id}`);
   174→   174→      if (res.ok) {
   175→   175→        setDetailOffer(await res.json());
   176→   176→      } else throw new Error();
   177→   177→    } catch { toast({ title: "Erreur de chargement", variant: "destructive" }); }
   178→   178→    finally { setDetailLoading(false); }
   179→   179→  };
   180→   180→
   181→   181→  // ── Guarantee dialog ────────────────────────────────────────────
   182→   182→  const openGuaranteeDialog = async (offer: Offer) => {
   183→   183→    try {
   184→   184→      const [offerRes, guaranteesRes] = await Promise.all([
   185→   185→        fetch(`/api/admin/offers/${offer.id}`),
   186→   186→        fetch("/api/admin/guarantees"),
   187→   187→      ]);
   188→   188→      if (!offerRes.ok || !guaranteesRes.ok) throw new Error();
   189→   189→      const offerFull: Offer = await offerRes.json();
   190→   190→      const guarantees: GuaranteeMini[] = await guaranteesRes.json();
   191→   191→      setSelectedOffer(offerFull);
   192→   192→      setAllGuarantees(guarantees);
   193→   193→      const linkMap = new Map(offerFull.guaranteeLinks.map((l) => [l.guaranteeId, l]));
   194→   194→      setGuaranteeChecks(guarantees.map((g) => ({ guaranteeId: g.id, isIncluded: linkMap.get(g.id)?.isIncluded ?? false })));
   195→   195→      setGuaranteeDialogOpen(true);
   196→   196→    } catch { toast({ title: "Erreur de chargement", variant: "destructive" }); }
   197→   197→  };
   198→   198→
   199→   199→  const toggleCheck = (idx: number) => {
   200→   200→    setGuaranteeChecks((prev) => prev.map((c, i) => i === idx ? { ...c, isIncluded: !c.isIncluded } : c));
   201→   201→  };
   202→   202→
   203→   203→  const saveGuarantees = async () => {
   204→   204→    if (!selectedOffer) return;
   205→   205→    setSavingGuarantees(true);
   206→   206→    try {
   207→   207→      const res = await fetch(`/api/admin/offers/${selectedOffer.id}/guarantees`, {
   208→   208→        method: "PUT", headers: { "Content-Type": "application/json" },
   209→   209→        body: JSON.stringify({ guarantees: guaranteeChecks }),
   210→   210→      });
   211→   211→      if (!res.ok) throw new Error();
   212→   212→      toast({ title: "Garanties mises à jour" });
   213→   213→      setGuaranteeDialogOpen(false);
   214→   214→      fetchOffers();
   215→   215→    } catch { toast({ title: "Erreur de sauvegarde", variant: "destructive" }); }
   216→   216→    finally { setSavingGuarantees(false); }
   217→   217→  };
   218→   218→
   219→   219→  // Group guarantees by calcMethod
   220→   220→  const groupedGuarantees = allGuarantees.reduce<Record<string, GuaranteeMini[]>>((acc, g) => {
   221→   221→    (acc[g.calcMethod] ??= []).push(g);
   222→   222→    return acc;
   223→   223→  }, {});
   224→   224→
   225→   225→  const groupLabels: Record<string, string> = {
   226→   226→    FREE: "🟢 Gratuites", FIXED_AMOUNT: "🔵 Montant fixe",
   227→   227→    VARIABLE_BASED: "🟠 Taux variable", MATRIX_BASED: "🟣 Matrice",
   228→   228→  };
   229→   229→
   230→   230→  // ── Render ─────────────────────────────────────────────────────
   231→   231→  if (loading) return <div className="space-y-4">{Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12 rounded-lg" />)}</div>;
   232→   232→
   233→   233→  return (
   234→   234→    <div>
   235→   235→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 mb-6">
   236→   236→        <h1 className="text-2xl font-bold">Offres</h1>
   237→   237→        <Button onClick={openCreate} className="bg-[#B9E54D] text-black hover:bg-[#a5d044]"><Plus className="h-4 w-4 mr-1" /> Ajouter</Button>
   238→   238→      </div>
   239→   239→
   240→   240→      {/* Filters */}
   241→   241→      <div className="flex flex-col sm:flex-row gap-3 mb-4">
   242→   242→        <div className="relative flex-1">
   243→   243→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   244→   244→          <Input placeholder="Rechercher..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-9" />
   245→   245→        </div>
   246→   246→        <Select value={filterInsurer} onValueChange={(v) => setFilterInsurer(v === "all" ? "" : v)}>
   247→   247→          <SelectTrigger className="w-full sm:w-48"><SelectValue placeholder="Assureur" /></SelectTrigger>
   248→   248→          <SelectContent>
   249→   249→            <SelectItem value="all">Tous les assureurs</SelectItem>
   250→   250→            {insurers.map((i) => <SelectItem key={i.id} value={i.id}>{i.name}</SelectItem>)}
   251→   251→          </SelectContent>
   252→   252→        </Select>
   253→   253→        <Select value={filterType} onValueChange={(v) => setFilterType(v === "all" ? "" : v)}>
   254→   254→          <SelectTrigger className="w-full sm:w-40"><SelectValue placeholder="Type" /></SelectTrigger>
   255→   255→          <SelectContent>
   256→   256→            <SelectItem value="all">Tous les types</SelectItem>
   257→   257→            <SelectItem value="tiers">Tiers</SelectItem>
   258→   258→            <SelectItem value="tiers_plus">Tiers+</SelectItem>
   259→   259→            <SelectItem value="tous_risques">Tous Risques</SelectItem>
   260→   260→          </SelectContent>
   261→   261→        </Select>
   262→   262→      </div>
   263→   263→
   264→   264→      {/* Desktop table */}
   265→   265→      <div className="hidden md:block rounded-xl border bg-card overflow-hidden">
   266→   266→        <Table>
   267→   267→          <TableHeader>
   268→   268→            <TableRow className="bg-muted/50">
   269→   269→              <TableHead>Nom</TableHead>
   270→   270→              <TableHead>Assureur</TableHead>
   271→   271→              <TableHead>Type</TableHead>
   272→   272→              <TableHead>Prix de base</TableHead>
   273→   273→              <TableHead>Franchise</TableHead>
   274→   274→              <TableHead className="text-center">Garanties</TableHead>
   275→   275→              <TableHead className="text-center">Statut</TableHead>
   276→   276→              <TableHead className="w-10"></TableHead>
   277→   277→            </TableRow>
   278→   278→          </TableHeader>
   279→   279→          <TableBody>
   280→   280→            {filtered.map((o) => (
   281→   281→              <TableRow key={o.id}>
   282→   282→                <TableCell className="font-medium">{o.name}</TableCell>
   283→   283→                <TableCell className="text-sm">{o.insurer.name}</TableCell>
   284→   284→                <TableCell><Badge className={coverageColors[o.coverageType] || ""}>{coverageLabels[o.coverageType] || o.coverageType}</Badge></TableCell>
   285→   285→                <TableCell className="text-sm font-mono">{fmtPrice(o.basePrice)}</TableCell>
   286→   286→                <TableCell className="text-sm">{o.deductible ? fmtPrice(o.deductible) : "—"}</TableCell>
   287→   287→                <TableCell className="text-center">
   288→   288→                  <Button variant="ghost" size="sm" className="text-xs" onClick={() => openGuaranteeDialog(o)}>
   289→   289→                    <ShieldCheck className="h-3.5 w-3.5 mr-1" />{o.guaranteeLinks.filter((l) => l.isIncluded).length}
   290→   290→                  </Button>
   291→   291→                </TableCell>
   292→   292→                <TableCell className="text-center"><Switch checked={o.isActive} onCheckedChange={() => handleToggleActive(o)} /></TableCell>
   293→   293→                <TableCell>
   294→   294→                  <DropdownMenu>
   295→   295→                    <DropdownMenuTrigger asChild><Button variant="ghost" size="icon" className="h-8 w-8"><MoreHorizontal className="h-4 w-4" /></Button></DropdownMenuTrigger>
   296→   296→                    <DropdownMenuContent align="end">
   297→   297→                      <DropdownMenuItem onClick={() => openDetail(o)}><Eye className="h-4 w-4 mr-2" /> Voir</DropdownMenuItem>
   298→   298→                      <DropdownMenuItem onClick={() => openGuaranteeDialog(o)}><ShieldCheck className="h-4 w-4 mr-2" /> Gérer les garanties</DropdownMenuItem>
   299→   299→                      <DropdownMenuItem onClick={() => openEdit(o)}><Pencil className="h-4 w-4 mr-2" /> Modifier</DropdownMenuItem>
   300→   300→                      <DropdownMenuItem className="text-destructive" onClick={() => setDeleteId(o.id)}><Trash2 className="h-4 w-4 mr-2" /> Supprimer</DropdownMenuItem>
   301→   301→                    </DropdownMenuContent>
   302→   302→                  </DropdownMenu>
   303→   303→                </TableCell>
   304→   304→              </TableRow>
   305→   305→            ))}
   306→   306→          </TableBody>
   307→   307→        </Table>
   308→   308→        {filtered.length === 0 && <p className="text-center py-8 text-muted-foreground text-sm">Aucune offre trouvée.</p>}
   309→   309→      </div>
   310→   310→
   311→   311→      {/* Mobile cards */}
   312→   312→      <div className="md:hidden space-y-3">
   313→   313→        {filtered.map((o) => (
   314→   314→          <Card key={o.id} className="p-4">
   315→   315→            <div className="flex items-start justify-between mb-2">
   316→   316→              <div>
   317→   317→                <p className="font-semibold">{o.name}</p>
   318→   318→                <p className="text-xs text-muted-foreground">{o.insurer.name}</p>
   319→   319→              </div>
   320→   320→              <Switch checked={o.isActive} onCheckedChange={() => handleToggleActive(o)} />
   321→   321→            </div>
   322→   322→            <div className="flex flex-wrap gap-2 text-xs mb-2">
   323→   323→              <Badge className={coverageColors[o.coverageType] || ""}>{coverageLabels[o.coverageType]}</Badge>
   324→   324→              <Badge variant="outline">{o.guaranteeLinks.filter((l) => l.isIncluded).length} garanties</Badge>
   325→   325→            </div>
   326→   326→            <p className="text-sm font-mono font-medium mb-3">{fmtPrice(o.basePrice)}</p>
   327→   327→            <div className="flex gap-2">
   328→   328→              <Button variant="outline" size="sm" className="flex-1" onClick={() => openDetail(o)}><Eye className="h-3.5 w-3.5 mr-1" /> Voir</Button>
   329→   329→              <Button variant="outline" size="sm" onClick={() => openEdit(o)}><Pencil className="h-3.5 w-3.5" /></Button>
   330→   330→              <Button variant="outline" size="sm" className="text-destructive" onClick={() => setDeleteId(o.id)}><Trash2 className="h-3.5 w-3.5" /></Button>
   331→   331→            </div>
   332→   332→          </Card>
   333→   333→        ))}
   334→   334→      </div>
   335→   335→
   336→   336→      {/* ── Detail Dialog ──────────────────────────────────────────── */}
   337→   337→      <Dialog open={!!detailOffer || detailLoading} onOpenChange={() => setDetailOffer(null)}>
   338→   338→        <DialogContent className="max-w-2xl max-h-[90vh]">
   339→   339→          {detailLoading ? (
   340→   340→            <div className="flex items-center justify-center py-12"><Loader2 className="h-6 w-6 animate-spin" /></div>
   341→   341→          ) : detailOffer ? (
   342→   342→            <>
   343→   343→              <DialogHeader>
   344→   344→                <div className="flex items-center gap-3 flex-wrap">
   345→   345→                  <DialogTitle className="text-xl">{detailOffer.name}</DialogTitle>
   346→   346→                  <Badge className={coverageColors[detailOffer.coverageType] || ""}>
   347→   347→                    {coverageLabels[detailOffer.coverageType] || detailOffer.coverageType}
   348→   348→                  </Badge>
   349→   349→                  <Badge className="bg-[#B9E54D]/20 text-black">{detailOffer.insurer.name}</Badge>
   350→   350→                </div>
   351→   351→                {detailOffer.description && (
   352→   352→                  <DialogDescription>{detailOffer.description}</DialogDescription>
   353→   353→                )}
   354→   354→              </DialogHeader>
   355→   355→
   356→   356→              <ScrollArea className="max-h-[65vh] pr-2">
   357→   357→                <div className="space-y-6 pb-4">
   358→   358→                  {/* Pricing */}
   359→   359→                  <div className="space-y-2">
   360→   360→                    <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide">Tarification</h3>
   361→   361→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
