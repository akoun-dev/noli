   104→// ─── Component ──────────────────────────────────────────────────────
   105→export function ComparisonForm() {
   106→  const {
   107→    comparisonStep, setComparisonStep,
   108→    personalInfo, setPersonalInfo,
   109→    vehicleInfo, setVehicleInfo,
   110→    coverageNeeds, setCoverageNeeds,
   111→    setView, setIsComparing, setComparisonResults,
   112→    isComparing, user,
   113→  } = useAppStore();
   114→  const { toast } = useToast();
   115→  const [direction, setDirection] = useState(0);
   116→  const [errors, setErrors] = useState<Record<string, string>>({});
   117→
   118→  // Dynamic coverage categories from DB
   119→  const [dbCategories, setDbCategories] = useState<DBCoverageCategory[]>([]);
   120→  const [categoriesLoading, setCategoriesLoading] = useState(false);
   121→
   122→  // Fetch coverage categories when component mounts (or step reaches 3)
   123→  useEffect(() => {
   124→    if (dbCategories.length > 0) return;
   125→    setCategoriesLoading(true);
   126→    fetch("/api/coverage-categories")
   127→      .then((res) => res.json())
   128→      .then((data) => {
   129→        if (Array.isArray(data)) setDbCategories(data);
   130→      })
   131→      .catch(() => {
   132→        // Fallback: empty — user will see error on submit
   133→      })
   134→      .finally(() => setCategoriesLoading(false));
   135→  }, [dbCategories.length]);
   136→
   137→  const goNext = useCallback(() => {
   138→    if (comparisonStep < 3) {
   139→      setDirection(1);
   140→      setComparisonStep(comparisonStep + 1);
   141→      setErrors({});
   142→    }
   143→  }, [comparisonStep, setComparisonStep]);
   144→
   145→  const goBack = useCallback(() => {
   146→    if (comparisonStep === 1) { setView("landing"); return; }
   147→    setDirection(-1);
   148→    setComparisonStep(comparisonStep - 1);
   149→    setErrors({});
   150→  }, [comparisonStep, setComparisonStep, setView]);
   151→
   152→  // ─── Validation ─────────────────────────────────────────────────
   153→  const validateStep1 = (): boolean => {
   154→    const e: Record<string, string> = {};
   155→    if (!personalInfo.lastName.trim()) e.lastName = "Le nom est requis";
   156→    if (!personalInfo.firstName.trim()) e.firstName = "Le prénom est requis";
   157→    if (!personalInfo.email.trim()) e.email = "L'email est requis";
   158→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(personalInfo.email))
   159→      e.email = "Email invalide";
   160→    if (!personalInfo.phone.trim()) e.phone = "Le téléphone est requis";
   161→    setErrors(e);
   162→    return Object.keys(e).length === 0;
   163→  };
   164→
   165→  const validateStep2 = (): boolean => {
   166→    const e: Record<string, string> = {};
   167→    if (!vehicleInfo.fuelType) e.fuelType = "Requis";
   168→    if (!vehicleInfo.fiscalPower) e.fiscalPower = "Requis";
   169→    if (!vehicleInfo.seats) e.seats = "Requis";
   170→    if (!vehicleInfo.year) e.year = "Requis";
   171→    if (!vehicleInfo.newValue.trim()) e.newValue = "Requis";
   172→    if (!vehicleInfo.currentValue.trim()) e.currentValue = "Requis";
   173→    if (!vehicleInfo.usage) e.usage = "Requis";
   174→    setErrors(e);
   175→    return Object.keys(e).length === 0;
   176→  };
   177→
   178→  const validateStep3 = (): boolean => {
   179→    const e: Record<string, string> = {};
   180→    if (!coverageNeeds.guaranteeCategories?.length)
   181→      e.categories = "Sélectionnez au moins une catégorie";
   182→    setErrors(e);
   183→    return Object.keys(e).length === 0;
   184→  };
   185→
   186→  // ─── Submit ─────────────────────────────────────────────────────
   187→  const handleSubmit = async () => {
   188→    if (!validateStep1() || !validateStep2() || !validateStep3()) {
   189→      setComparisonStep(1);
   190→      setDirection(0);
   191→      return;
   192→    }
   193→    setIsComparing(true);
   194→    try {
   195→      const res = await fetch("/api/compare", {
   196→        method: "POST",
   197→        headers: { "Content-Type": "application/json" },
   198→        body: JSON.stringify({
   199→          personalInfo, vehicleInfo, coverageNeeds,
   200→          userId: user.id,
   201→        }),
   202→      });
   203→      if (!res.ok) throw new Error("Erreur lors de la comparaison");
   204→      const data = await res.json();
   205→      setComparisonResults(data.results ?? []);
   206→      setView("results");
   207→    } catch (err) {
   208→      toast({
   209→        title: "Erreur",
   210→        description: err instanceof Error ? err.message : "Veuillez réessayer.",
   211→        variant: "destructive",
   212→      });
   213→    } finally {
   214→      setIsComparing(false);
   215→    }
   216→  };
   217→
   218→  const toggleCategory = (id: string) => {
   219→    const cats = coverageNeeds.guaranteeCategories || [];
   220→    setCoverageNeeds({
   221→      guaranteeCategories: cats.includes(id)
   222→        ? cats.filter((c) => c !== id)
   223→        : [...cats, id],
   224→    });
   225→  };
   226→
   227→  const toggleAllCategories = () => {
   228→    const allIds = dbCategories.map((c) => c.code);
   229→    const current = coverageNeeds.guaranteeCategories || [];
   230→    if (current.length === allIds.length && allIds.length > 0) {
   231→      setCoverageNeeds({ guaranteeCategories: [] });
   232→    } else {
   233→      setCoverageNeeds({ guaranteeCategories: allIds });
   234→    }
   235→  };
   236→
   237→  const progressValue = (comparisonStep / 3) * 100;
   238→
   239→  const FieldError = ({ field }: { field: string }) =>
   240→    errors[field] ? (
   241→      <p className="mt-1 text-xs text-destructive">{errors[field]}</p>
   242→    ) : null;
   243→
   244→  return (
   245→    <div className="min-h-[70vh] flex flex-col items-center justify-start px-4 sm:px-6 py-6 sm:py-8">
   246→      <div className="w-full max-w-2xl mx-auto">
   247→        {/* Progress bar */}
   248→        <div className="mb-8">
   249→          <Progress value={progressValue} className="h-1.5" />
   250→          <div className="flex justify-between mt-3">
   251→            {STEPS.map((step) => (
   252→              <button
   253→                key={step.id}
   254→                onClick={() => {
   255→                  if (step.id < comparisonStep) {
   256→                    setDirection(-1);
   257→                    setComparisonStep(step.id);
   258→                    setErrors({});
   259→                  }
   260→                }}
   261→                className={`flex items-center gap-2 text-sm font-medium transition-colors ${
   262→                  step.id === comparisonStep
   263→                    ? "text-primary"
   264→                    : step.id < comparisonStep
   265→                    ? "text-muted-foreground hover:text-foreground cursor-pointer"
   266→                    : "text-muted-foreground/50"
   267→                }`}
   268→              >
   269→                <span
   270→                  className={`w-7 h-7 rounded-full flex items-center justify-center text-xs font-bold transition-colors ${
   271→                    step.id < comparisonStep
   272→                      ? "bg-primary text-primary-foreground"
   273→                      : step.id === comparisonStep
   274→                      ? "bg-primary/10 text-primary border-2 border-primary"
   275→                      : "bg-muted text-muted-foreground"
   276→                  }`}
   277→                >
   278→                  {step.id < comparisonStep ? (
   279→                    <Check className="w-3.5 h-3.5" />
   280→                  ) : (
   281→                    step.id
   282→                  )}
   283→                </span>
   284→                <span className="hidden sm:inline">{step.label}</span>
   285→              </button>
   286→            ))}
   287→          </div>
   288→        </div>
   289→
   290→        {/* Steps content */}
   291→        <AnimatePresence mode="wait" custom={direction}>
   292→          <motion.div
   293→            key={comparisonStep}
   294→            custom={direction}
   295→            variants={slideVariants}
   296→            initial="enter"
   297→            animate="center"
   298→            exit="exit"
   299→            transition={{ duration: 0.3, ease: "easeInOut" }}
   300→          >
   301→            {comparisonStep === 1 && (
   302→              <Step1
   303→                personalInfo={personalInfo}
   304→                setPersonalInfo={setPersonalInfo}
   305→                errors={errors}
   306→                FieldError={FieldError}
   307→              />
   308→            )}
   309→            {comparisonStep === 2 && (
   310→              <Step2
   311→                vehicleInfo={vehicleInfo}
   312→                setVehicleInfo={setVehicleInfo}
   313→                errors={errors}
   314→                FieldError={FieldError}
   315→              />
   316→            )}
   317→            {comparisonStep === 3 && (
   318→              <Step3
   319→                categories={dbCategories}
   320→                loading={categoriesLoading}
   321→                selected={coverageNeeds.guaranteeCategories || []}
   322→                onToggle={toggleCategory}
   323→                onToggleAll={toggleAllCategories}
   324→                error={errors.categories}
   325→              />
   326→            )}
   327→          </motion.div>
   328→        </AnimatePresence>
   329→
   330→        {/* Navigation buttons */}
   331→        <div className="flex items-center justify-between mt-8 pt-6 border-t border-border">
   332→          <Button
   333→            variant="outline"
   334→            onClick={goBack}
   335→            className="rounded-full px-4 sm:px-6"
   336→          >
   337→            <ChevronLeft className="w-4 h-4 mr-1" />
   338→            <span className="hidden sm:inline">Retour</span>
   339→          </Button>
   340→
   341→          {comparisonStep < 3 ? (
   342→            <Button
   343→              onClick={() => {
   344→                if (comparisonStep === 1 && !validateStep1()) return;
   345→                if (comparisonStep === 2 && !validateStep2()) return;
   346→                goNext();
   347→              }}
   348→              className="bg-primary text-primary-foreground hover:bg-primary/90 rounded-full px-4 sm:px-6"
   349→            >
   350→              Étape suivante
   351→            </Button>
   352→          ) : (
   353→            <Button
   354→              onClick={handleSubmit}
   355→              disabled={isComparing}
   356→              className="bg-primary text-primary-foreground hover:bg-primary/90 rounded-full px-4 sm:px-6"
   357→            >
   358→              {isComparing ? (
   359→                <>
   360→                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   361→                  <span className="hidden sm:inline">Comparaison...</span>
   362→                </>
   363→              ) : (
   364→                "Comparer mes offres"
   365→              )}
   366→            </Button>
   367→          )}
   368→        </div>
   369→      </div>
   370→    </div>
   371→  );
   372→}

   617→      {/* Usage */}
   618→      <div className="space-y-2">

   621→          value={vehicleInfo.usage}

   623→        >
   624→          <SelectTrigger className={`w-full ${errors.usage ? "border-destructive" : ""}`}>

   627→          <SelectContent>
   628→            {USAGE_OPTIONS.map((o) => (

   633→          </SelectContent>
   634→        </Select>
   635→        <FieldError field="usage" />
   636→      </div>
   637→    </div>
   638→  );
