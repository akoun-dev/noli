    97→
    98→function formatLastLogin(dateStr?: string | null): string {
    99→  if (!dateStr) return "—";
   100→  const d = new Date(dateStr);
   101→  const now = new Date();
   102→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   103→  const yesterday = new Date(today);
   104→  yesterday.setDate(yesterday.getDate() - 1);
   105→  const target = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   106→  if (target.getTime() === today.getTime()) return "Aujourd'hui";
   107→  if (target.getTime() === yesterday.getTime()) return "Hier";
   108→  return d.toLocaleDateString("fr-FR");
   109→}
