   327→/* ──────── Comparison Modal ──────── */
   328→
   329→function ComparisonModal({
   330→  offers,
   331→  open,
   332→  onClose,
   333→}: {
   334→  offers: InsurerOffer[];
   335→  open: boolean;
   336→  onClose: () => void;
   337→}) {
   338→  // Collect all unique features across all offers (before early return for hooks rules)
   339→  const allFeatures = useMemo(() => {
   340→    const featureSet = new Set<string>();
   341→    offers.forEach((o) => o.features.forEach((f) => featureSet.add(f)));
   342→    return Array.from(featureSet);
   343→  }, [offers]);
   344→
   345→  if (!open || offers.length < 2) return null;
   346→
   347→  const formatMonthlyPrice = (price: number) => {
   348→    const val = Math.round(price);
   349→    if (val % 1 !== 0) return new Intl.NumberFormat("fr-FR", { minimumFractionDigits: 0, maximumFractionDigits: 3 }).format(val) + " FCFA";
   350→    return formatFCFA(val);
   351→  };
   352→
   353→  return (
   354→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   355→      <DialogContent className="sm:max-w-[900px] max-h-[90vh] overflow-y-auto p-0">
   356→        {/* Header */}
   357→        <div className="flex items-center justify-between px-6 py-4 border-b border-border/60 bg-muted/30 sticky top-0 z-10">
   358→          <DialogTitle className="text-lg font-bold text-foreground">
   359→            Comparer les garanties
   360→          </DialogTitle>
   361→          <DialogDescription className="sr-only">
   362→            Comparaison détaillée de {offers.length} offres d&apos;assurance
   363→          </DialogDescription>
   364→        </div>
   365→
   366→        {/* Comparison Table */}
   367→        <div className="overflow-x-auto">
   368→          <table className="w-full min-w-[500px]">
   369→            {/* Column Headers: Insurers */}
   370→            <thead>
   371→              <tr>
   372→                <th className="text-left p-4 w-48 bg-muted/30 sticky left-0 z-10 border-b border-border/60">
   373→                  <span className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   374→                    Critères
   375→                  </span>
   376→                </th>
   377→                {offers.map((offer) => (
   378→                  <th
   379→                    key={offer.id}
   380→                    className="text-center p-4 bg-muted/30 border-b border-border/60"
   381→                  >
   382→                    <div className="flex flex-col items-center gap-2">
   383→                      <div className="flex h-12 w-12 items-center justify-center rounded-full bg-primary/10 border-2 border-primary/20">
   384→                        <Shield className="size-6 text-primary" />
   385→                      </div>
   386→                      <span className="text-sm font-bold text-foreground leading-tight">
   387→                        {offer.insurerName}
   388→                      </span>
   389→                      <span className="text-xs text-muted-foreground capitalize">
   390→                        {offer.coverageType.replace("_", " ")}
   391→                      </span>
   392→                    </div>
   393→                  </th>
   394→                ))}
   395→              </tr>
   396→            </thead>
   397→
   398→            <tbody>
   399→              {/* Prix mensuel */}
   400→              <tr className="bg-card">
   401→                <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-card border-b border-border/30">
   402→                  Prix mensuel
   403→                </td>
   404→                {offers.map((offer) => (
   405→                  <td
   406→                    key={offer.id}
   407→                    className="p-4 text-center text-sm font-bold text-foreground border-b border-border/30"
   408→                  >
   409→                    {formatMonthlyPrice(offer.monthlyPrice)}
   410→                  </td>
   411→                ))}
   412→              </tr>
   413→
   414→              {/* Franchise */}
   415→              <tr className="bg-muted/20">
   416→                <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-muted/20 border-b border-border/30">

   419→                {offers.map((offer) => (
   420→                  <td
   421→                    key={offer.id}
   422→                    className="p-4 text-center text-sm text-foreground border-b border-border/30"
   423→                  >
   424→                    {formatFCFA(offer.deductible)}
   425→                  </td>
   426→                ))}
   427→              </tr>
   428→
   429→              {/* Note / Rating */}
   430→              <tr className="bg-card">
   431→                <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-card border-b border-border/30">
   432→                  Note assureur
   433→                </td>
   434→                {offers.map((offer) => (
   435→                  <td
   436→                    key={offer.id}
   437→                    className="p-4 text-center border-b border-border/30"

   439→                    <div className="flex items-center justify-center">

   478→                              </div>
   479→                            </div>
   480→                          )}
   481→                        </td>

   484→                  </tr>
   485→                );
   486→              })}
   487→
   488→              {/* Conditions */}
   489→              {offers.some((o) => o.conditions) && (
   490→                <tr className="bg-muted/20">
   491→                  <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-muted/20">

   501→                  ))}
   502→                </tr>
   503→              )}
   504→            </tbody>
   505→          </table>
   506→        </div>
   507→      </DialogContent>
   508→    </Dialog>
   509→  );
   510→}

   956→        next.delete(name);

   958→        next.add(name);

   983→        break;

   985→        results.sort((a, b) => b.insurerRating - a.insurerRating);
   986→        break;
   987→      case "name_asc":
   988→        results.sort((a, b) => a.name.localeCompare(b.name, "fr"));
