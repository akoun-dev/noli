     1→"use client";
     2→
     3→import { useEffect, useState, useCallback } from "react";
     4→import {
     5→  Car,
     6→  Plus,
     7→  CheckCircle2,
     8→  XCircle,
     9→  Pencil,
    10→  Trash2,
    11→  Loader2,
    12→} from "lucide-react";
    13→import { Button } from "@/components/ui/button";
    14→import { Badge } from "@/components/ui/badge";
    15→import { Skeleton } from "@/components/ui/skeleton";
    16→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
    17→import { Input } from "@/components/ui/input";
    18→import { Label } from "@/components/ui/label";
    19→import { Textarea } from "@/components/ui/textarea";
    20→import { Checkbox } from "@/components/ui/checkbox";
    21→import {
    22→  Dialog,
    23→  DialogContent,
    24→  DialogHeader,
    25→  DialogTitle,
    26→  DialogFooter,
    27→  DialogDescription,
    28→} from "@/components/ui/dialog";
    29→import {
    30→  AlertDialog,
    31→  AlertDialogAction,
    32→  AlertDialogCancel,
    33→  AlertDialogContent,
    34→  AlertDialogDescription,
    35→  AlertDialogFooter,
    36→  AlertDialogHeader,
    37→  AlertDialogTitle,
    38→} from "@/components/ui/alert-dialog";
    39→import {
    40→  Select,
    41→  SelectContent,
    42→  SelectItem,
    43→  SelectTrigger,
    44→  SelectValue,
    45→} from "@/components/ui/select";
    46→import {
    47→  Tooltip,
    48→  TooltipContent,
    49→  TooltipProvider,
    50→  TooltipTrigger,
    51→} from "@/components/ui/tooltip";
    52→import { useAppStore } from "@/store/app-store";
    53→import { useToast } from "@/hooks/use-toast";
    54→
    55→/* ── Types ── */
    56→interface Category {
    57→  id: string;
    58→  name: string;
    59→  icon: string | null;
    60→}
    61→
    62→interface Offer {
    63→  id: string;
    64→  name: string;
    65→  description: string | null;
    66→  priceMin: number | null;
    67→  priceMax: number | null;
    68→  coverageAmount: number | null;
    69→  deductible: number;
    70→  contractType: string | null;
    71→  features: string[];
    72→  isActive: boolean;
    73→  category: { id: string; name: string; icon: string | null } | null;
    74→  createdAt: string;
    75→}
    76→
    77→interface CoverageMini {
    78→  id: string;
    79→  code: string;
    80→  name: string;
    81→  type: string;
    82→  calculationType: string;
    83→  isMandatory: boolean;
    84→  isActive: boolean;
    85→  category: { id: string; name: string; code: string } | null;
    86→}
    87→
    88→interface OfferFormData {
    89→  name: string;
    90→  categoryId: string;
    91→  description: string;
    92→  priceMin: string;
    93→  priceMax: string;
    94→  coverageAmount: string;
    95→  deductible: string;
    96→  contractType: string;
    97→  selectedGuarantees: string[]; // coverage names
    98→  isActive: boolean;
    99→}
   100→
   101→const emptyForm: OfferFormData = {
   102→  name: "",
   103→  categoryId: "",
   104→  description: "",
   105→  priceMin: "",
   106→  priceMax: "",
   107→  coverageAmount: "",
   108→  deductible: "0",
   109→  contractType: "",
   110→  selectedGuarantees: [],
   111→  isActive: true,
   112→};
   113→
   114→const contractTypeLabels: Record<string, string> = {
   115→  basic: "Tiers simple",
   116→  third_party_plus: "Tiers étendu",
   117→  all_risks: "Tous risques",
   118→};
   119→
   120→const formatPrice = (amount: number | null) =>
   121→  amount != null
   122→    ? new Intl.NumberFormat("fr-FR").format(amount) + " FCFA"
   123→    : "—";
   124→
   125→export function InsurerOffersTab() {
   126→  const { user } = useAppStore();
   127→  const { toast } = useToast();
   128→
   129→  const [insurerId, setInsurerId] = useState<string | null>(null);
   130→  const [offers, setOffers] = useState<Offer[]>([]);
   131→  const [categories, setCategories] = useState<Category[]>([]);
   132→  const [loading, setLoading] = useState(true);
   133→  const [error, setError] = useState<string | null>(null);
   134→
   135→  // Dialog state
   136→  const [dialogOpen, setDialogOpen] = useState(false);
   137→  const [editingOffer, setEditingOffer] = useState<Offer | null>(null);
   138→  const [form, setForm] = useState<OfferFormData>(emptyForm);
   139→  const [submitting, setSubmitting] = useState(false);
   140→
   141→  // Delete dialog
   142→  const [deleteTarget, setDeleteTarget] = useState<Offer | null>(null);
   143→  const [deleting, setDeleting] = useState(false);
   144→
   145→  // Coverages for guarantee selection
   146→  const [formCoverages, setFormCoverages] = useState<CoverageMini[]>([]);
   147→  const [formCoveragesLoading, setFormCoveragesLoading] = useState(false);
   148→
   149→  /* ── Fetch helpers ── */
   150→  const fetchInsurer = useCallback(async (userId: string) => {
   151→    try {
   152→      const res = await fetch(`/api/insurer/account?userId=${userId}`);
   153→      if (!res.ok) throw new Error("Compte assureur non trouvé");
   154→      const data = await res.json();
   155→      setInsurerId(data.id);
   156→      return data.id;
   157→    } catch {
   158→      setError("Impossible de charger votre compte assureur");
   159→      return null;
   160→    }
   161→  }, []);
   162→
   163→  const fetchOffers = useCallback(async (iid: string) => {
   164→    try {
   165→      const res = await fetch(`/api/insurer/offers?insurerId=${iid}`);
   166→      if (!res.ok) throw new Error();
   167→      const data = await res.json();
   168→      setOffers(data.offers || []);
   169→    } catch {
   170→      setError("Erreur lors du chargement des offres");
   171→    }
   172→  }, []);
   173→
   174→  const fetchCategories = useCallback(async () => {
   175→    try {
   176→      const res = await fetch("/api/admin/insurance-categories");
   177→      if (!res.ok) return;
   178→      const data = await res.json();
   179→      setCategories(data);
   180→    } catch {
   181→      /* ignore */
   182→    }
   183→  }, []);
   184→
   185→  const refreshData = useCallback(async () => {
   186→    if (!insurerId) return;
   187→    await fetchOffers(insurerId);
   188→  }, [insurerId, fetchOffers]);
   189→
   190→  /* ── Init ── */
   191→  useEffect(() => {
   192→    if (!user.id) {
   193→      setLoading(false);
   194→      return;
   195→    }
   196→    (async () => {
   197→      const iid = await fetchInsurer(user.id!);
   198→      if (iid) await fetchOffers(iid);
   199→      await fetchCategories();
   200→      setLoading(false);
   201→    })();
   202→  }, [user.id]);
   203→
   204→  /* ── Fetch coverages when insurerId is known (for guarantee selection) ── */
   205→  useEffect(() => {
   206→    if (!insurerId) {
   207→      setFormCoverages([]);
   208→      return;
   209→    }
   210→    setFormCoveragesLoading(true);
   211→    fetch(`/api/insurer/coverages?insurerId=${insurerId}`)
   212→      .then((r) => {
   213→        if (r.ok) return r.json();
   214→        return { coverages: [] };
   215→      })
   216→      .then((data) => setFormCoverages(data.coverages || []))
   217→      .catch(() => setFormCoverages([]))
   218→      .finally(() => setFormCoveragesLoading(false));
   219→  }, [insurerId]);
   220→
   221→  /* ── Form helpers ── */
   222→  const openCreate = () => {
   223→    setEditingOffer(null);
   224→    setForm(emptyForm);
   225→    setDialogOpen(true);
   226→  };
   227→
   228→  const openEdit = (offer: Offer) => {
   229→    setEditingOffer(offer);
   230→    setForm({
   231→      name: offer.name,
   232→      categoryId: offer.category?.id || "",
   233→      description: offer.description || "",
   234→      priceMin: offer.priceMin != null ? String(offer.priceMin) : "",
   235→      priceMax: offer.priceMax != null ? String(offer.priceMax) : "",
   236→      coverageAmount:
   237→        offer.coverageAmount != null ? String(offer.coverageAmount) : "",
   238→      deductible: String(offer.deductible),
   239→      contractType: offer.contractType || "",
   240→      selectedGuarantees: offer.features || [],
   241→      isActive: offer.isActive,
   242→    });
   243→    setDialogOpen(true);
   244→  };
   245→
   246→  const handleSubmit = async () => {
   247→    if (!form.name.trim() || !insurerId) return;
   248→    setSubmitting(true);
   249→
   250→    const payload = {
   251→      insurerId,
   252→      categoryId: form.categoryId || null,
   253→      name: form.name.trim(),
   254→      description: form.description.trim() || null,
   255→      priceMin: form.priceMin ? Number(form.priceMin) : null,
   256→      priceMax: form.priceMax ? Number(form.priceMax) : null,
   257→      coverageAmount: form.coverageAmount ? Number(form.coverageAmount) : null,
   258→      deductible: form.deductible ? Number(form.deductible) : 0,
   259→      contractType: form.contractType || null,
   260→      features: form.selectedGuarantees,
   261→      isActive: form.isActive,
   262→    };
   263→
   264→    try {
   265→      const url = editingOffer
   266→        ? `/api/insurer/offers/${editingOffer.id}`
   267→        : "/api/insurer/offers";
   268→      const method = editingOffer ? "PUT" : "POST";
   269→
   270→      const res = await fetch(url, {
   271→        method,
   272→        headers: { "Content-Type": "application/json" },
   273→        body: JSON.stringify(payload),
   274→      });
   275→
   276→      if (!res.ok) {
   277→        const err = await res.json();
   278→        throw new Error(err.error || "Erreur");
   279→      }
   280→
   281→      toast({
   282→        title: editingOffer
   283→          ? "Offre mise à jour avec succès"
   284→          : "Offre créée avec succès",
   285→        description: editingOffer
   286→          ? `"${form.name}" a été modifiée.`
   287→          : `"${form.name}" a été ajoutée à vos offres.`,
   288→      });
   289→
   290→      setDialogOpen(false);
   291→      await refreshData();
   292→    } catch (e) {
   293→      toast({
   294→        title: "Erreur",
   295→        description: e instanceof Error ? e.message : "Une erreur est survenue",
   296→        variant: "destructive",
   297→      });
   298→    } finally {
   299→      setSubmitting(false);
   300→    }
   301→  };
   302→
   303→  const handleDelete = async () => {
   304→    if (!deleteTarget) return;
   305→    setDeleting(true);
   306→    try {
   307→      const res = await fetch(`/api/insurer/offers/${deleteTarget.id}`, {
   308→        method: "DELETE",
   309→      });
   310→      if (!res.ok) throw new Error();
   311→      toast({
   312→        title: "Offre supprimée",
   313→        description: `"${deleteTarget.name}" a été désactivée.`,
   314→      });
   315→      setDeleteTarget(null);
   316→      await refreshData();
   317→    } catch {
   318→      toast({
   319→        title: "Erreur",
   320→        description: "Impossible de supprimer l'offre",
   321→        variant: "destructive",
   322→      });
   323→    } finally {
   324→      setDeleting(false);
   325→    }
   326→  };
   327→
   328→  const toggleGuarantee = (coverageName: string, checked: boolean | "indeterminate") => {
   329→    if (checked === "indeterminate") return;
   330→    setForm((f) => ({
   331→      ...f,
   332→      selectedGuarantees: checked
   333→        ? [...f.selectedGuarantees, coverageName]
   334→        : f.selectedGuarantees.filter((n) => n !== coverageName),
   335→    }));
   336→  };
   337→
   338→  /* ── Group coverages by category for display ── */
   339→  const groupedCoverages = formCoverages.reduce<
   340→    Record<string, CoverageMini[]>
   341→  >((acc, c) => {
   342→    const key = c.category?.name || "Autre";
   343→    (acc[key] ??= []).push(c);
   344→    return acc;
   345→  }, {});
   346→
   347→  /* ── Render ── */
   348→  return (
   349→    <div className="space-y-6">
   350→      {/* Header */}
   351→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   352→        <div>
   353→          <h2 className="text-2xl font-bold">Mes Offres</h2>
   354→          <p className="text-muted-foreground mt-1">
   355→            Gérez vos offres d&apos;assurance disponibles sur NOLI.
   356→          </p>
   357→        </div>
   358→        <Button
   359→          className="bg-[#B9E54D] text-black hover:bg-[#a5d044]"
   360→          onClick={openCreate}
   361→        >
   362→          <Plus className="mr-2 h-4 w-4" />
   363→          Créer une offre
   364→        </Button>
   365→      </div>
   366→
   367→      {/* Loading */}
   368→      {loading && (
   369→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   370→          {Array.from({ length: 6 }).map((_, i) => (
   371→            <Card key={i}>
   372→              <CardHeader className="pb-3">
   373→                <Skeleton className="h-5 w-3/4" />
   374→                <Skeleton className="h-4 w-1/2 mt-2" />
   375→              </CardHeader>
   376→              <CardContent className="space-y-3">
   377→                <Skeleton className="h-4 w-full" />
   378→                <Skeleton className="h-4 w-2/3" />
   379→                <div className="flex gap-2 pt-2">
   380→                  <Skeleton className="h-6 w-20 rounded-full" />
   381→                  <Skeleton className="h-6 w-24 rounded-full" />
   382→                </div>
   383→              </CardContent>
   384→            </Card>
   385→          ))}
   386→        </div>
   387→      )}
   388→
   389→      {/* Error */}
   390→      {error && !loading && (
   391→        <div className="flex flex-col items-center justify-center py-16 text-center">
   392→          <div className="flex h-16 w-16 items-center justify-center rounded-full bg-red-100 dark:bg-red-900/30 mb-4">
   393→            <XCircle className="h-8 w-8 text-red-500" />
   394→          </div>
   395→          <h3 className="text-lg font-semibold">Erreur de chargement</h3>
   396→          <p className="text-sm text-muted-foreground mt-1">{error}</p>
   397→        </div>
   398→      )}
   399→
   400→      {/* Empty */}
   401→      {!loading && !error && offers.length === 0 && (
   402→        <div className="flex flex-col items-center justify-center py-16 text-center">
   403→          <div className="flex h-16 w-16 items-center justify-center rounded-full bg-muted mb-4">
   404→            <Car className="h-8 w-8 text-muted-foreground" />
   405→          </div>
   406→          <h3 className="text-lg font-semibold">Aucune offre trouvée</h3>
   407→          <p className="text-sm text-muted-foreground mt-1 max-w-md">
   408→            Vous n&apos;avez pas encore d&apos;offres d&apos;assurance.
   409→            Cliquez sur &quot;Créer une offre&quot; pour commencer.
   410→          </p>
   411→        </div>
   412→      )}
   413→
   414→      {/* Offers grid */}
   415→      {!loading && !error && offers.length > 0 && (
   416→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   417→          {offers.map((offer) => (
   418→            <Card
   419→              key={offer.id}
   420→              className="hover:shadow-md transition-shadow"
   421→            >
   422→              <CardHeader className="pb-3">
   423→                <div className="flex items-start justify-between">
   424→                  <div className="space-y-1 min-w-0 flex-1">
   425→                    <CardTitle className="text-base truncate">
   426→                      {offer.name}
   427→                    </CardTitle>
   428→                    <p className="text-xs text-muted-foreground">
   429→                      {offer.category?.name || "Non catégorisé"}
   430→                    </p>
   431→                  </div>
   432→                  <div className="flex items-center gap-1 shrink-0 ml-2">
   433→                    {offer.isActive ? (
   434→                      <Badge className="border-0 bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400 text-xs">
   435→                        <CheckCircle2 className="mr-1 h-3 w-3" />
   436→                        Active
   437→                      </Badge>
   438→                    ) : (
   439→                      <Badge variant="secondary" className="text-xs">
   440→                        Inactive
   441→                      </Badge>
   442→                    )}
   443→                  </div>
   444→                </div>
   445→              </CardHeader>
   446→              <CardContent className="space-y-3">
   447→                {/* Price range */}
   448→                <div className="flex items-baseline gap-1">
   449→                  <span className="text-xl font-bold">
   450→                    {formatPrice(offer.priceMin)}
   451→                  </span>
   452→                  {offer.priceMax &&
   453→                    offer.priceMax !== offer.priceMin && (
   454→                      <span className="text-sm text-muted-foreground">
   455→                        — {formatPrice(offer.priceMax)}
   456→                      </span>
   457→                    )}
   458→                </div>
   459→
   460→                {/* Badges */}
   461→                <div className="flex gap-2 flex-wrap">
   462→                  {offer.contractType && (
   463→                    <Badge variant="outline" className="text-xs">
   464→                      {contractTypeLabels[offer.contractType] ||
   465→                        offer.contractType}
   466→                    </Badge>
   467→                  )}
   468→                  {offer.coverageAmount && (
   469→                    <Badge variant="secondary" className="text-xs">
   470→                      Capital : {formatPrice(offer.coverageAmount)}
   471→                    </Badge>
   472→                  )}
   473→                </div>
   474→
   475→                {/* Description */}
   476→                {offer.description && (
   477→                  <p className="text-sm text-muted-foreground line-clamp-2">
   478→                    {offer.description}
   479→                  </p>
   480→                )}
   481→
   482→                {/* Features / Guarantees */}
   483→                {offer.features.length > 0 && (
   484→                  <div className="space-y-1.5">
   485→                    <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">
   486→                      Garanties
   487→                    </p>
   488→                    <TooltipProvider>
   489→                      <ul className="space-y-1">
   490→                        {offer.features.slice(0, 4).map((f, i) => (
   491→                          <li
   492→                            key={i}
   493→                            className="text-sm text-muted-foreground flex items-center gap-2"
   494→                          >
   495→                            <span className="h-1.5 w-1.5 rounded-full bg-[#B9E54D] shrink-0" />
   496→                            <Tooltip>
   497→                              <TooltipTrigger asChild>
   498→                                <span className="truncate">{f}</span>
   499→                              </TooltipTrigger>
   500→                              <TooltipContent>
   501→                                <p>{f}</p>
   502→                              </TooltipContent>
   503→                            </Tooltip>
   504→                          </li>
   505→                        ))}
   506→                        {offer.features.length > 4 && (
   507→                          <li className="text-xs text-muted-foreground pl-3.5">
   508→                            +{offer.features.length - 4} autre(s)...
   509→                          </li>
   510→                        )}
   511→                      </ul>
   512→                    </TooltipProvider>
   513→                  </div>
   514→                )}
   515→
   516→                {/* Actions */}
   517→                <div className="flex items-center gap-2 pt-2 border-t">
   518→                  <Button
   519→                    variant="ghost"
   520→                    size="sm"
   521→                    className="h-8 text-xs"
   522→                    onClick={() => openEdit(offer)}
   523→                  >
   524→                    <Pencil className="mr-1.5 h-3.5 w-3.5" />
   525→                    Modifier
   526→                  </Button>
   527→                  <Button
   528→                    variant="ghost"
   529→                    size="sm"
   530→                    className="h-8 text-xs text-destructive hover:text-destructive"
   531→                    onClick={() => setDeleteTarget(offer)}
   532→                  >
   533→                    <Trash2 className="mr-1.5 h-3.5 w-3.5" />
   534→                    Supprimer
   535→                  </Button>
   536→                </div>
   537→              </CardContent>
   538→            </Card>
   539→          ))}
   540→        </div>
   541→      )}
   542→
   543→      {/* ── Create/Edit Dialog ── */}
   544→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   545→        <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   546→          <DialogHeader>
   547→            <DialogTitle>
   548→              {editingOffer ? "Modifier l'offre" : "Nouvelle offre"}
   549→            </DialogTitle>
   550→            <DialogDescription>
   551→              {editingOffer
   552→                ? "Modifiez les informations de votre offre."
   553→                : "Remplissez les informations pour créer une nouvelle offre."}
   554→            </DialogDescription>
   555→          </DialogHeader>
   556→
   557→          <div className="space-y-4 py-2">
   558→            {/* Name */}
   559→            <div className="space-y-2">
   560→              <Label htmlFor="offer-name">Nom de l&apos;offre *</Label>
   561→              <Input
   562→                id="offer-name"
   563→                placeholder="Ex: Assurance Auto Premium"
   564→                value={form.name}
   565→                onChange={(e) =>
   566→                  setForm((f) => ({ ...f, name: e.target.value }))
   567→                }
   568→              />
   569→            </div>
   570→
   571→            {/* Category */}
   572→            <div className="space-y-2">
   573→              <Label>Catégorie de produit</Label>
   574→              <Select
   575→                value={form.categoryId}
   576→                onValueChange={(v) =>
   577→                  setForm((f) => ({ ...f, categoryId: v }))
   578→                }
   579→              >
   580→                <SelectTrigger>
   581→                  <SelectValue placeholder="Sélectionner une catégorie" />
   582→                </SelectTrigger>
   583→                <SelectContent>
   584→                  {categories.map((c) => (
   585→                    <SelectItem key={c.id} value={c.id}>
   586→                      {c.name}
   587→                    </SelectItem>
   588→                  ))}
   589→                </SelectContent>
   590→              </Select>
   591→            </div>
   592→
   593→            {/* Contract Type */}
   594→            <div className="space-y-2">
   595→              <Label>Type de contrat</Label>
   596→              <Select
   597→                value={form.contractType}
   598→                onValueChange={(v) =>
   599→                  setForm((f) => ({ ...f, contractType: v }))
   600→                }
   601→              >
   602→                <SelectTrigger>
   603→                  <SelectValue placeholder="Sélectionner un type" />
   604→                </SelectTrigger>
   605→                <SelectContent>
