   254→/* ── Component ── */
   255→export function InsurerGuaranteesTab() {
   256→  const { user } = useAppStore();
   257→  const { toast } = useToast();
   258→
   259→  const [insurerId, setInsurerId] = useState<string | null>(null);
   260→  const [coverages, setCoverages] = useState<Coverage[]>([]);
   261→  const [categories, setCategories] = useState<CoverageCategory[]>([]);
   262→  const [loading, setLoading] = useState(true);
   263→  const [error, setError] = useState<string | null>(null);
   264→
   265→  // Wizard state
   266→  const [dialogOpen, setDialogOpen] = useState(false);
   267→  const [editingItem, setEditingItem] = useState<Coverage | null>(null);
   268→  const [step, setStep] = useState(1);
   269→  const [step1, setStep1] = useState<Step1Data>(emptyStep1);
   270→  const [calcType, setCalcType] = useState<CalculationType | "">("");
   271→  const [metadata, setMetadata] = useState<Record<string, unknown>>({});
   272→  const [saving, setSaving] = useState(false);
   273→
   274→  // Matrix array states
   275→  const [tariffs, setTariffs] = useState<MatrixTariff[]>([]);
   276→  const [formulas, setFormulas] = useState<MatrixFormula[]>([]);
   277→  const [categoryTariffs, setCategoryTariffs] = useState<CategoryTariff[]>([]);
   278→
   279→  // Delete dialog
   280→  const [deleteTarget, setDeleteTarget] = useState<Coverage | null>(null);
   281→  const [deleting, setDeleting] = useState(false);
   282→
   283→  /* ── Fetch helpers ── */
   284→  const fetchInsurer = useCallback(async (userId: string) => {
   285→    try {
   286→      const res = await fetch(`/api/insurer/account?userId=${userId}`);
   287→      if (!res.ok) throw new Error();
   288→      const data = await res.json();
   289→      setInsurerId(data.id);
   290→      return data.id;
   291→    } catch {
   292→      setError("Impossible de charger votre compte assureur");
   293→      return null;
   294→    }
   295→  }, []);
   296→
   297→  const fetchCoverages = useCallback(async (iid: string) => {
   298→    try {
   299→      const res = await fetch(`/api/insurer/coverages?insurerId=${iid}`);
   300→      if (!res.ok) throw new Error();
   301→      const data = await res.json();
   302→      setCoverages(data.coverages || []);
   303→    } catch {
   304→      setError("Erreur lors du chargement des garanties");
   305→    }
   306→  }, []);
   307→
   308→  const fetchCategories = useCallback(async () => {
   309→    try {
   310→      const res = await fetch("/api/admin/coverage-categories");
   311→      if (!res.ok) return;
   312→      const data = await res.json();
   313→      setCategories(data);
   314→    } catch {
   315→      /* ignore */
   316→    }
   317→  }, []);
   318→
   319→  const refreshData = useCallback(async () => {
   320→    if (!insurerId) return;
   321→    await fetchCoverages(insurerId);
   322→  }, [insurerId, fetchCoverages]);
   323→
   324→  /* ── Init ── */
   325→  useEffect(() => {
   326→    if (!user.id) {
   327→      setLoading(false);
   328→      return;
   329→    }
   330→    (async () => {
   331→      const iid = await fetchInsurer(user.id!);
   332→      if (iid) await fetchCoverages(iid);
   333→      await fetchCategories();
   334→      setLoading(false);
   335→    })();
   336→  }, [user.id]);
   337→
   338→  /* ── Wizard ── */
   339→  const resetWizard = () => {
   340→    setStep1(emptyStep1);
   341→    setCalcType("");
   342→    setMetadata({});
   343→    setTariffs([]);
   344→    setFormulas([]);
   345→    setCategoryTariffs([]);
   346→    setStep(1);
   347→  };
   348→
   349→  const openCreate = () => {
   350→    setEditingItem(null);
   351→    resetWizard();
   352→    setDialogOpen(true);
   353→  };
   354→
   355→  const openEdit = (item: Coverage) => {
   356→    setEditingItem(item);
   357→    setStep1({
   358→      name: item.name,
   359→      categoryId: item.category?.id || "",
   360→      description: item.description || "",
   361→      isMandatory: item.isMandatory,
   362→    });
   363→    setCalcType(item.calculationType as CalculationType);
   364→
   365→    let parsed: Record<string, unknown> = {};
   366→    try {
   367→      parsed =
   368→        typeof item.metadata === "string"
   369→          ? JSON.parse(item.metadata)
   370→          : (item as unknown as Record<string, unknown>) || {};
   371→    } catch {
   372→      parsed = {};
   373→    }
   374→
   375→    setMetadata(parsed);
   376→
   377→    // Restore matrix arrays
   378→    setTariffs((parsed.tariffs as MatrixTariff[])?.map((t) => ({ ...t, id: t.id || crypto.randomUUID() })) || []);
   379→    setFormulas((parsed.formulas as MatrixFormula[])?.map((f) => ({ ...f, id: f.id || crypto.randomUUID() })) || []);
   380→    setCategoryTariffs(
   381→      (parsed.categoryTariffs as CategoryTariff[])?.map((c) => ({ ...c, id: c.id || crypto.randomUUID() })) || []
   382→    );
   383→
   384→    setStep(1);
   385→    setDialogOpen(true);
   386→  };
   387→
   388→  /* ── Metadata builder ── */
   389→  const buildMetadata = (): Record<string, unknown> => {
   390→    if (calcType === "FREE") {
   391→      return { method: "FREE" };
   392→    }
   393→
   394→    if (calcType === "FIXED_AMOUNT") {
   395→      const m: Record<string, unknown> = {
   396→        method: "FIXED_AMOUNT",
   397→        fixedAmount: Number(metadata.fixedAmount) || 0,
   398→      };
   399→      if (metadata.packPriceReduced !== undefined && metadata.packPriceReduced !== "")
   400→        m.packPriceReduced = Number(metadata.packPriceReduced);
   401→      if (metadata.capital !== undefined && metadata.capital !== "")
   402→        m.capital = Number(metadata.capital);
   403→      if (metadata.minAmount !== undefined && metadata.minAmount !== "")
   404→        m.minAmount = Number(metadata.minAmount);
   405→      if (metadata.maxAmount !== undefined && metadata.maxAmount !== "")
   406→        m.maxAmount = Number(metadata.maxAmount);
   407→      if (metadata.franchiseEnabled) {
   408→        m.franchise = {
   409→          type: metadata.franchiseType || "PERCENT",
   410→          value: Number(metadata.franchiseValue) || 0,
   411→          minAmount: Number(metadata.franchiseMin) || 0,
   412→          maxAmount: Number(metadata.franchiseMax) || 0,
   413→        };
   414→      }
   415→      if (metadata.requiresGuarantee) m.requiresGuarantee = metadata.requiresGuarantee;
   416→      return m;
   417→    }
   418→
   419→    if (calcType === "VARIABLE_BASED") {
   420→      const m: Record<string, unknown> = {
   421→        method: "VARIABLE_BASED",
   422→        variableSource: metadata.variableSource || "NEW_VALUE",
   423→        ratePercent: Number(metadata.ratePercent) || 0,
   424→      };
   425→      if (metadata.conditionedByNewValue) {
   426→        m.conditionedByNewValue = true;
   427→        m.newValueThreshold = Number(metadata.newValueThreshold) || 0;
   428→        m.rateBelowThresholdPercent = Number(metadata.rateBelowThresholdPercent) || 0;
   429→        m.rateAboveThresholdPercent = Number(metadata.rateAboveThresholdPercent) || 0;
   430→      }
   431→      if (metadata.minAmount !== undefined && metadata.minAmount !== "")
   432→        m.minAmount = Number(metadata.minAmount);
   433→      if (metadata.maxAmount !== undefined && metadata.maxAmount !== "")
   434→        m.maxAmount = Number(metadata.maxAmount);
   435→      if (metadata.franchiseEnabled) {
   436→        m.franchise = {
   437→          type: metadata.franchiseType || "PERCENT",
   438→          value: Number(metadata.franchiseValue) || 0,
   439→          minAmount: Number(metadata.franchiseMin) || 0,
   440→          maxAmount: Number(metadata.franchiseMax) || 0,
   441→        };
   442→      }
   443→      if (metadata.requiresGuarantee) m.requiresGuarantee = metadata.requiresGuarantee;
   444→      return m;
   445→    }
   446→
   447→    if (calcType === "MATRIX_BASED") {
   448→      const m: Record<string, unknown> = {
   449→        method: "MATRIX_BASED",
   450→        dimension: metadata.dimension || "FISCAL_POWER",
   451→      };
   452→      if (tariffs.length > 0) m.tariffs = tariffs.map(({ id: _id, ...rest }) => rest);
   453→      if (formulas.length > 0) m.formulas = formulas.map(({ id: _id, ...rest }) => rest);
   454→      if (categoryTariffs.length > 0)
   455→        m.categoryTariffs = categoryTariffs.map(({ id: _id, ...rest }) => rest);
   456→      return m;
   457→    }
   458→
   459→    return {};
   460→  };
   461→
   462→  /* ── Save ── */
   463→  const handleSave = async () => {
   464→    if (!step1.name.trim() || !insurerId || !calcType) {
   465→      toast({
   466→        title: "Erreur",
   467→        description: "Veuillez remplir tous les champs requis",
   468→        variant: "destructive",
   469→      });
   470→      return;
   471→    }
   472→    setSaving(true);
   473→
   474→    const payload = {
   475→      insurerId,
   476→      categoryId: step1.categoryId || null,
   477→      name: step1.name.trim(),
   478→      description: step1.description.trim() || null,
   479→      calculationType: calcType,
   480→      isMandatory: step1.isMandatory,
   481→      metadata: buildMetadata(),
   482→    };
   483→
   484→    try {
   485→      const url = editingItem
   486→        ? `/api/insurer/coverages/${editingItem.id}`
   487→        : "/api/insurer/coverages";
   488→      const method = editingItem ? "PUT" : "POST";
   489→
   490→      const res = await fetch(url, {
   491→        method,
   492→        headers: { "Content-Type": "application/json" },
   493→        body: JSON.stringify(payload),
   494→      });
   495→
   496→      if (!res.ok) {
   497→        const err = await res.json();
   498→        throw new Error(err.error || "Erreur");
   499→      }
   500→
   501→      toast({
   502→        title: editingItem ? "Garantie mise à jour" : "Garantie créée",
   503→        description: editingItem
   504→          ? `"${step1.name}" a été modifiée.`
   505→          : `"${step1.name}" a été ajoutée à vos garanties.`,
   506→      });
   507→
   508→      setDialogOpen(false);
   509→      await refreshData();
   510→    } catch (e) {
   511→      toast({
   512→        title: "Erreur",
   513→        description: e instanceof Error ? e.message : "Une erreur est survenue",
   514→        variant: "destructive",
   515→      });
   516→    } finally {
   517→      setSaving(false);
   518→    }
   519→  };
   520→
   521→  const handleDelete = async () => {
   522→    if (!deleteTarget) return;
   523→    setDeleting(true);
   524→    try {
   525→      const res = await fetch(`/api/insurer/coverages/${deleteTarget.id}`, {
   526→        method: "DELETE",
   527→      });
   528→      if (!res.ok) throw new Error();
   529→      toast({
   530→        title: "Garantie supprimée",
   531→        description: `"${deleteTarget.name}" a été désactivée.`,
   532→      });
   533→      setDeleteTarget(null);
   534→      await refreshData();
   535→    } catch {
   536→      toast({
   537→        title: "Erreur",
   538→        description: "Impossible de supprimer la garantie",
   539→        variant: "destructive",
   540→      });
   541→    } finally {
   542→      setDeleting(false);
   543→    }
   544→  };
   545→
   546→  /* ── Matrix helpers ── */
   547→  const updateTariff = (id: string, field: keyof MatrixTariff, value: string | number) => {
   548→    setTariffs((prev) =>
   549→      prev.map((t) => (t.id === id ? { ...t, [field]: value } : t))
   550→    );
   551→  };
   552→  const updateFormula = (id: string, field: keyof MatrixFormula, value: string | number) => {
   553→    setFormulas((prev) =>
   554→      prev.map((f) => (f.id === id ? { ...f, [field]: value } : f))
   555→    );
   556→  };
   557→  const updateCategoryTariff = (id: string, field: keyof CategoryTariff, value: string | number) => {
   558→    setCategoryTariffs((prev) =>
   559→      prev.map((c) => (c.id === id ? { ...c, [field]: value } : c))
   560→    );
   561→  };
   562→
   563→  /* ── Franchise sub-form (shared by FIXED_AMOUNT & VARIABLE_BASED) ── */
   564→  const renderFranchiseSection = () => (
   565→    <div className="rounded-lg border bg-muted/30 p-4 space-y-4">
   566→      <div className="flex items-center gap-2 font-medium text-sm">
   567→        <Info className="h-4 w-4 text-muted-foreground" />
   568→        Configuration de la franchise
   569→      </div>
   570→
   571→      <div className="w-full">
   572→        <Label className="text-xs">Type de franchise</Label>
   573→        <RadioGroup
   574→          className="flex gap-6 mt-1"
   575→          value={(metadata.franchiseType as string) || "PERCENT"}
   576→          onValueChange={(v) => setMetadata({ ...metadata, franchiseType: v })}
   577→        >
   578→          <div className="flex items-center gap-2">
   579→            <RadioGroupItem value="PERCENT" id="frac-pct" />
   580→            <Label htmlFor="frac-pct" className="text-sm font-normal">
   581→              Pourcentage (%)
   582→            </Label>
   583→          </div>
   584→          <div className="flex items-center gap-2">
   585→            <RadioGroupItem value="AMOUNT" id="frac-amt" />
   586→            <Label htmlFor="frac-amt" className="text-sm font-normal">
   587→              Montant fixe (FCFA)
   588→            </Label>
   589→          </div>
   590→        </RadioGroup>
