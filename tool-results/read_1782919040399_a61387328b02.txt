    35→    35→// ── Types ────────────────────────────────────────────────────────
    36→    36→interface RateCondition { condition: string; rate: number; }
    37→    37→
    38→    38→interface Guarantee {
    39→    39→  id: string; name: string; slug: string; description: string | null;
    40→    40→  icon: string | null; category: string; calcMethod: string;
    41→    41→  fixedPrice: number | null; rate: number | null;
    42→    42→  rateConditions: RateCondition[] | null;

    44→    44→  franchise: Record<string, unknown> | null;

    48→    48→
    49→    49→const fmtPrice = (n: number) => new Intl.NumberFormat("fr-FR").format(n) + " FCFA";
    50→    50→const fmtDate = (d: string) => (d ? new Date(d).toLocaleDateString("fr-FR") : "—");
    51→    51→
    52→    52→const calcColors: Record<string, string> = {
    53→    53→  FREE: "bg-emerald-100 text-emerald-800",
    54→    54→  FIXED_AMOUNT: "bg-blue-100 text-blue-800",
    55→    55→  VARIABLE_BASED: "bg-orange-100 text-orange-800",
    56→    56→  MATRIX_BASED: "bg-purple-100 text-purple-800",
    57→    57→};
    58→    58→const calcLabels: Record<string, string> = {
    59→    59→  FREE: "Gratuit", FIXED_AMOUNT: "Montant fixe", VARIABLE_BASED: "Taux variable", MATRIX_BASED: "Matrice",
    60→    60→};
    61→    61→
    62→    62→function slugify(text: string): string {
    63→    63→  return text.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)+/g, "");
    64→    64→}
    65→    65→
    66→    66→// ── Empty form ───────────────────────────────────────────────────
    67→    67→const emptyGuarantee = {
    68→    68→  name: "", slug: "", description: "", icon: "", category: "garantie",
    69→    69→  calcMethod: "FIXED_AMOUNT" as string, fixedPrice: null as number | null,
    70→    70→  rate: null as number | null, rateConditionsStr: "[]", capitalStr: "",
    71→    71→  franchiseStr: "", matrixConfigStr: "", sortOrder: 0, isActive: true,
    72→    72→};
    73→    73→
    74→    74→// ── Helpers to display parsed values ─────────────────────────────
    75→    75→function displayRateOrPrice(g: Guarantee): string {
    76→    76→  if (g.calcMethod === "FREE") return "Gratuit";
    77→    77→  if (g.calcMethod === "FIXED_AMOUNT") return g.fixedPrice ? fmtPrice(g.fixedPrice) : "—";
    78→    78→  if (g.rateConditions && g.rateConditions.length > 0) {
    79→    79→    return g.rateConditions.map((rc) => `${rc.condition}: ${rc.rate}%`).join(" / ");
    80→    80→  }
    81→    81→  if (g.rate != null) return `${g.rate} %`;
    82→    82→  return "—";
    83→    83→}
    84→    84→
    85→    85→function displayCapital(g: Guarantee): string {
    86→    86→  if (!g.capital) return "—";
    87→    87→  return Object.entries(g.capital).map(([k, v]) => `${k}: ${v}`).join("\n");
    88→    88→}
    89→    89→
    90→    90→function displayFranchise(g: Guarantee): string {
    91→    91→  if (!g.franchise) return "—";
    92→    92→  if (g.franchise.description) return g.franchise.description as string;
    93→    93→  const parts: string[] = [];
    94→    94→  if (g.franchise.percent) parts.push(`${g.franchise.percent}%`);
    95→    95→  if (g.franchise.min) parts.push(`min ${fmtPrice(g.franchise.min as number)}`);
    96→    96→  return parts.join(", ") || "—";
    97→    97→}
