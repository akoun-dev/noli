   104→   104→// ─── Component ──────────────────────────────────────────────────────
   105→   105→export function ComparisonForm() {
   106→   106→  const {
   107→   107→    comparisonStep, setComparisonStep,
   108→   108→    personalInfo, setPersonalInfo,
   109→   109→    vehicleInfo, setVehicleInfo,
   110→   110→    coverageNeeds, setCoverageNeeds,
   111→   111→    setView, setIsComparing, setComparisonResults,
   112→   112→    isComparing, user,
   113→   113→  } = useAppStore();
   114→   114→  const { toast } = useToast();
   115→   115→  const [direction, setDirection] = useState(0);
   116→   116→  const [errors, setErrors] = useState<Record<string, string>>({});
   117→   117→
   118→   118→  // Dynamic coverage categories from DB
   119→   119→  const [dbCategories, setDbCategories] = useState<DBCoverageCategory[]>([]);
   120→   120→  const [categoriesLoading, setCategoriesLoading] = useState(false);
   121→   121→
   122→   122→  // Fetch coverage categories when component mounts (or step reaches 3)
   123→   123→  useEffect(() => {
   124→   124→    if (dbCategories.length > 0) return;
   125→   125→    setCategoriesLoading(true);
   126→   126→    fetch("/api/coverage-categories")
   127→   127→      .then((res) => res.json())
   128→   128→      .then((data) => {
   129→   129→        if (Array.isArray(data)) setDbCategories(data);
   130→   130→      })
   131→   131→      .catch(() => {
   132→   132→        // Fallback: empty — user will see error on submit
   133→   133→      })
   134→   134→      .finally(() => setCategoriesLoading(false));
   135→   135→  }, [dbCategories.length]);
   136→   136→
   137→   137→  const goNext = useCallback(() => {
   138→   138→    if (comparisonStep < 3) {
   139→   139→      setDirection(1);
   140→   140→      setComparisonStep(comparisonStep + 1);
   141→   141→      setErrors({});
   142→   142→    }
   143→   143→  }, [comparisonStep, setComparisonStep]);
   144→   144→
   145→   145→  const goBack = useCallback(() => {
   146→   146→    if (comparisonStep === 1) { setView("landing"); return; }
   147→   147→    setDirection(-1);
   148→   148→    setComparisonStep(comparisonStep - 1);
   149→   149→    setErrors({});
   150→   150→  }, [comparisonStep, setComparisonStep, setView]);
   151→   151→
   152→   152→  // ─── Validation ─────────────────────────────────────────────────
   153→   153→  const validateStep1 = (): boolean => {
   154→   154→    const e: Record<string, string> = {};
   155→   155→    if (!personalInfo.lastName.trim()) e.lastName = "Le nom est requis";
   156→   156→    if (!personalInfo.firstName.trim()) e.firstName = "Le prénom est requis";
   157→   157→    if (!personalInfo.email.trim()) e.email = "L'email est requis";
   158→   158→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(personalInfo.email))
   159→   159→      e.email = "Email invalide";
   160→   160→    if (!personalInfo.phone.trim()) e.phone = "Le téléphone est requis";
   161→   161→    setErrors(e);
   162→   162→    return Object.keys(e).length === 0;
   163→   163→  };
   164→   164→
   165→   165→  const validateStep2 = (): boolean => {
   166→   166→    const e: Record<string, string> = {};
   167→   167→    if (!vehicleInfo.fuelType) e.fuelType = "Requis";
   168→   168→    if (!vehicleInfo.fiscalPower) e.fiscalPower = "Requis";
   169→   169→    if (!vehicleInfo.seats) e.seats = "Requis";
   170→   170→    if (!vehicleInfo.year) e.year = "Requis";
   171→   171→    if (!vehicleInfo.newValue.trim()) e.newValue = "Requis";
   172→   172→    if (!vehicleInfo.currentValue.trim()) e.currentValue = "Requis";
   173→   173→    if (!vehicleInfo.usage) e.usage = "Requis";
   174→   174→    setErrors(e);
   175→   175→    return Object.keys(e).length === 0;
   176→   176→  };
   177→   177→
   178→   178→  const validateStep3 = (): boolean => {
   179→   179→    const e: Record<string, string> = {};
   180→   180→    if (!coverageNeeds.guaranteeCategories?.length)
   181→   181→      e.categories = "Sélectionnez au moins une catégorie";
   182→   182→    setErrors(e);
   183→   183→    return Object.keys(e).length === 0;
   184→   184→  };
   185→   185→
   186→   186→  // ─── Submit ─────────────────────────────────────────────────────
   187→   187→  const handleSubmit = async () => {
   188→   188→    if (!validateStep1() || !validateStep2() || !validateStep3()) {
   189→   189→      setComparisonStep(1);
   190→   190→      setDirection(0);
   191→   191→      return;
   192→   192→    }
   193→   193→    setIsComparing(true);
   194→   194→    try {
   195→   195→      const res = await fetch("/api/compare", {
   196→   196→        method: "POST",
   197→   197→        headers: { "Content-Type": "application/json" },
   198→   198→        body: JSON.stringify({
   199→   199→          personalInfo, vehicleInfo, coverageNeeds,
   200→   200→          userId: user.id,
   201→   201→        }),
   202→   202→      });
   203→   203→      if (!res.ok) throw new Error("Erreur lors de la comparaison");
   204→   204→      const data = await res.json();
   205→   205→      setComparisonResults(data.results ?? []);
   206→   206→      setView("results");
   207→   207→    } catch (err) {
   208→   208→      toast({
   209→   209→        title: "Erreur",
   210→   210→        description: err instanceof Error ? err.message : "Veuillez réessayer.",
   211→   211→        variant: "destructive",
   212→   212→      });
   213→   213→    } finally {
   214→   214→      setIsComparing(false);
   215→   215→    }
   216→   216→  };
   217→   217→
   218→   218→  const toggleCategory = (id: string) => {
   219→   219→    const cats = coverageNeeds.guaranteeCategories || [];
   220→   220→    setCoverageNeeds({
   221→   221→      guaranteeCategories: cats.includes(id)
   222→   222→        ? cats.filter((c) => c !== id)
   223→   223→        : [...cats, id],
   224→   224→    });
   225→   225→  };
   226→   226→
   227→   227→  const toggleAllCategories = () => {
   228→   228→    const allIds = dbCategories.map((c) => c.code);
   229→   229→    const current = coverageNeeds.guaranteeCategories || [];
   230→   230→    if (current.length === allIds.length && allIds.length > 0) {
   231→   231→      setCoverageNeeds({ guaranteeCategories: [] });
   232→   232→    } else {
   233→   233→      setCoverageNeeds({ guaranteeCategories: allIds });
   234→   234→    }
   235→   235→  };
   236→   236→
   237→   237→  const progressValue = (comparisonStep / 3) * 100;
   238→   238→
   239→   239→  const FieldError = ({ field }: { field: string }) =>
   240→   240→    errors[field] ? (
   241→   241→      <p className="mt-1 text-xs text-destructive">{errors[field]}</p>
   242→   242→    ) : null;
   243→   243→
   244→   244→  return (
   245→   245→    <div className="min-h-[70vh] flex flex-col items-center justify-start px-4 sm:px-6 py-6 sm:py-8">
   246→   246→      <div className="w-full max-w-2xl mx-auto">
   247→   247→        {/* Progress bar */}
   248→   248→        <div className="mb-8">
   249→   249→          <Progress value={progressValue} className="h-1.5" />
   250→   250→          <div className="flex justify-between mt-3">
   251→   251→            {STEPS.map((step) => (
   252→   252→              <button
   253→   253→                key={step.id}
   254→   254→                onClick={() => {
   255→   255→                  if (step.id < comparisonStep) {
   256→   256→                    setDirection(-1);
   257→   257→                    setComparisonStep(step.id);
   258→   258→                    setErrors({});
   259→   259→                  }
   260→   260→                }}
   261→   261→                className={`flex items-center gap-2 text-sm font-medium transition-colors ${
   262→   262→                  step.id === comparisonStep
   263→   263→                    ? "text-primary"
   264→   264→                    : step.id < comparisonStep
   265→   265→                    ? "text-muted-foreground hover:text-foreground cursor-pointer"
   266→   266→                    : "text-muted-foreground/50"
   267→   267→                }`}
   268→   268→              >
   269→   269→                <span
   270→   270→                  className={`w-7 h-7 rounded-full flex items-center justify-center text-xs font-bold transition-colors ${
   271→   271→                    step.id < comparisonStep
   272→   272→                      ? "bg-primary text-primary-foreground"
   273→   273→                      : step.id === comparisonStep
   274→   274→                      ? "bg-primary/10 text-primary border-2 border-primary"
   275→   275→                      : "bg-muted text-muted-foreground"
   276→   276→                  }`}
   277→   277→                >
   278→   278→                  {step.id < comparisonStep ? (
   279→   279→                    <Check className="w-3.5 h-3.5" />
   280→   280→                  ) : (
   281→   281→                    step.id
   282→   282→                  )}
   283→   283→                </span>
   284→   284→                <span className="hidden sm:inline">{step.label}</span>
   285→   285→              </button>
   286→   286→            ))}
   287→   287→          </div>
   288→   288→        </div>
   289→   289→
   290→   290→        {/* Steps content */}
   291→   291→        <AnimatePresence mode="wait" custom={direction}>
   292→   292→          <motion.div
   293→   293→            key={comparisonStep}
   294→   294→            custom={direction}
   295→   295→            variants={slideVariants}
   296→   296→            initial="enter"
   297→   297→            animate="center"
   298→   298→            exit="exit"
   299→   299→            transition={{ duration: 0.3, ease: "easeInOut" }}
   300→   300→          >
   301→   301→            {comparisonStep === 1 && (
   302→   302→              <Step1
   303→   303→                personalInfo={personalInfo}
   304→   304→                setPersonalInfo={setPersonalInfo}
   305→   305→                errors={errors}
   306→   306→                FieldError={FieldError}
   307→   307→              />
   308→   308→            )}
   309→   309→            {comparisonStep === 2 && (
   310→   310→              <Step2
   311→   311→                vehicleInfo={vehicleInfo}
   312→   312→                setVehicleInfo={setVehicleInfo}
   313→   313→                errors={errors}
   314→   314→                FieldError={FieldError}
   315→   315→              />
   316→   316→            )}
   317→   317→            {comparisonStep === 3 && (
   318→   318→              <Step3
   319→   319→                categories={dbCategories}
   320→   320→                loading={categoriesLoading}
   321→   321→                selected={coverageNeeds.guaranteeCategories || []}
   322→   322→                onToggle={toggleCategory}
   323→   323→                onToggleAll={toggleAllCategories}
   324→   324→                error={errors.categories}
   325→   325→              />
   326→   326→            )}
   327→   327→          </motion.div>
   328→   328→        </AnimatePresence>
   329→   329→
   330→   330→        {/* Navigation buttons */}
   331→   331→        <div className="flex items-center justify-between mt-8 pt-6 border-t border-border">
   332→   332→          <Button
   333→   333→            variant="outline"
   334→   334→            onClick={goBack}
   335→   335→            className="rounded-full px-4 sm:px-6"
   336→   336→          >
   337→   337→            <ChevronLeft className="w-4 h-4 mr-1" />
   338→   338→            <span className="hidden sm:inline">Retour</span>
   339→   339→          </Button>
   340→   340→
   341→   341→          {comparisonStep < 3 ? (
   342→   342→            <Button
   343→   343→              onClick={() => {
   344→   344→                if (comparisonStep === 1 && !validateStep1()) return;
   345→   345→                if (comparisonStep === 2 && !validateStep2()) return;
   346→   346→                goNext();
   347→   347→              }}
   348→   348→              className="bg-primary text-primary-foreground hover:bg-primary/90 rounded-full px-4 sm:px-6"
   349→   349→            >
   350→   350→              Étape suivante
   351→   351→            </Button>
   352→   352→          ) : (
   353→   353→            <Button
   354→   354→              onClick={handleSubmit}
   355→   355→              disabled={isComparing}
   356→   356→              className="bg-primary text-primary-foreground hover:bg-primary/90 rounded-full px-4 sm:px-6"
   357→   357→            >
   358→   358→              {isComparing ? (
   359→   359→                <>
   360→   360→                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   361→   361→                  <span className="hidden sm:inline">Comparaison...</span>
   362→   362→                </>
   363→   363→              ) : (
   364→   364→                "Comparer mes offres"
   365→   365→              )}
   366→   366→            </Button>
   367→   367→          )}
   368→   368→        </div>
   369→   369→      </div>
   370→   370→    </div>
   371→   371→  );
   372→   372→}

   478→   478→  setVehicleInfo,
