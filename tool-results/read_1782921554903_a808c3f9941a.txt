    34→// ── Types ────────────────────────────────────────────────────────
    35→interface Insurer {
    36→  id: string; name: string; logo: string | null; description: string | null;
    37→  phone: string | null; email: string | null; website: string | null;
    38→  rating: number; isVerified: boolean; isActive: boolean;
    39→  createdAt: string; _count: { offers: number; guaranteeLinks: number };
    40→}
    41→
    42→interface InsurerFull extends Omit<Insurer, "_count"> {
    43→  guaranteeLinks: { id: string; insurerId: string; guaranteeId: string; isEnabled: boolean; customRate: number | null; customPrice: number | null; guarantee: GuaranteeMini }[];
    44→  offers: { id: string; name: string; coverageType: string; basePrice: number; isActive: boolean }[];
    45→}
    46→
    47→interface GuaranteeMini {
    48→  id: string; name: string; calcMethod: string; fixedPrice: number | null; rate: number | null;
    49→}
    50→
    51→// ── Helpers ──────────────────────────────────────────────────────
    52→const fmtDate = (d: string) => (d ? new Date(d).toLocaleDateString("fr-FR") : "—");
    53→const fmtPrice = (n: number) => new Intl.NumberFormat("fr-FR").format(n) + " FCFA";
    54→
    55→const emptyInsurer = {
    56→  name: "", logo: "", description: "", phone: "", email: "", website: "",
    57→  rating: 0, isVerified: false, isActive: true,
    58→};

    60→const coverageLabels: Record<string, string> = { tiers: "Tiers", tiers_plus: "Tiers+", tous_risques: "Tous Risques" };
    61→
    62→// ── Calc method badge ────────────────────────────────────────────
    63→function CalcBadge({ method }: { method: string }) {
    64→  const colors: Record<string, string> = {
    65→    FREE: "bg-emerald-100 text-emerald-800",
    66→    FIXED_AMOUNT: "bg-blue-100 text-blue-800",
    67→    VARIABLE_BASED: "bg-orange-100 text-orange-800",
    68→    MATRIX_BASED: "bg-purple-100 text-purple-800",
    69→  };
    70→  const labels: Record<string, string> = {
    71→    FREE: "Gratuit", FIXED_AMOUNT: "Montant fixe", VARIABLE_BASED: "Taux variable", MATRIX_BASED: "Matrice",
    72→  };
    73→  return <span className={`inline-block rounded-full px-2 py-0.5 text-[10px] font-semibold ${colors[method] || "bg-gray-100"}`}>{labels[method] || method}</span>;
    74→}

    76→// ── Main component ───────────────────────────────────────────────
    77→export function AssureursTab() {
    78→  const { toast } = useToast();
    79→  const [insurers, setInsurers] = useState<Insurer[]>([]);
    80→  const [loading, setLoading] = useState(true);
    81→  const [search, setSearch] = useState("");
    82→  const [formOpen, setFormOpen] = useState(false);
    83→  const [editing, setEditing] = useState<Insurer | null>(null);
    84→  const [form, setForm] = useState(emptyInsurer);
    85→  const [saving, setSaving] = useState(false);
    86→  const [deleteId, setDeleteId] = useState<string | null>(null);
    87→  const [deleting, setDeleting] = useState(false);
    88→
    89→  // Guarantee management
    90→  const [guaranteeDialogOpen, setGuaranteeDialogOpen] = useState(false);
    91→  const [selectedInsurer, setSelectedInsurer] = useState<InsurerFull | null>(null);
    92→  const [allGuarantees, setAllGuarantees] = useState<GuaranteeMini[]>([]);
    93→  const [guaranteeLinks, setGuaranteeLinks] = useState<{ guaranteeId: string; isEnabled: boolean; customRate: string; customPrice: string }[]>([]);
    94→  const [savingGuarantees, setSavingGuarantees] = useState(false);
    95→
    96→  // Detail view
    97→  const [detailInsurer, setDetailInsurer] = useState<InsurerFull | null>(null);
    98→  const [detailLoading, setDetailLoading] = useState(false);
    99→
   100→  const fetchInsurers = useCallback(async () => {
   101→    try {
   102→      const res = await fetch("/api/admin/insurers");
   103→      if (res.ok) setInsurers(await res.json());
   104→    } finally { setLoading(false); }
   105→  }, []);
   106→
   107→  useEffect(() => { fetchInsurers(); }, [fetchInsurers]);
   108→
   109→  const filtered = insurers.filter((i) => !search || i.name.toLowerCase().includes(search.toLowerCase()));
   110→
   111→  const openCreate = () => { setEditing(null); setForm(emptyInsurer); setFormOpen(true); };
   112→  const openEdit = (i: Insurer) => { setEditing(i); setForm({ name: i.name, logo: i.logo || "", description: i.description || "", phone: i.phone || "", email: i.email || "", website: i.website || "", rating: i.rating, isVerified: i.isVerified, isActive: i.isActive }); setFormOpen(true); };
   113→
   114→  const handleSave = async () => {
   115→    if (!form.name.trim()) { toast({ title: "Le nom est obligatoire", variant: "destructive" }); return; }
   116→    setSaving(true);
   117→    try {
   118→      const url = editing ? `/api/admin/insurers/${editing.id}` : "/api/admin/insurers";
   119→      const method = editing ? "PUT" : "POST";
   120→      const res = await fetch(url, { method, headers: { "Content-Type": "application/json" }, body: JSON.stringify(form) });
   121→      if (!res.ok) { const e = await res.json(); throw new Error(e.error || "Erreur"); }
   122→      toast({ title: editing ? "Assureur modifié" : "Assureur créé" });
   123→      setFormOpen(false);
   124→      fetchInsurers();
   125→    } catch (err) { toast({ title: (err as Error).message, variant: "destructive" }); }
   126→    finally { setSaving(false); }
   127→  };
   128→
   129→  const handleDelete = async () => {
   130→    if (!deleteId) return;
   131→    setDeleting(true);
   132→    try {
   133→      const res = await fetch(`/api/admin/insurers/${deleteId}`, { method: "DELETE" });
   134→      if (!res.ok) throw new Error();
   135→      toast({ title: "Assureur supprimé" });
   136→      setDeleteId(null);
   137→      fetchInsurers();
   138→    } catch { toast({ title: "Erreur lors de la suppression", variant: "destructive" }); }
   139→    finally { setDeleting(false); }
   140→  };
   141→
   142→  const handleToggleActive = async (i: Insurer) => {
   143→    try {
   144→      const res = await fetch(`/api/admin/insurers/${i.id}`, { method: "PUT", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ ...i, isActive: !i.isActive }) });
   145→      if (res.ok) fetchInsurers();
   146→    } catch { toast({ title: "Erreur", variant: "destructive" }); }
   147→  };
   148→
   149→  // ── Detail view ────────────────────────────────────────────────
   150→  const openDetail = async (i: Insurer) => {
   151→    setDetailLoading(true);
   152→    setDetailInsurer(null);
   153→    try {
   154→      const res = await fetch(`/api/admin/insurers/${i.id}`);
   155→      if (res.ok) {
   156→        setDetailInsurer(await res.json());
   157→      } else throw new Error();
   158→    } catch { toast({ title: "Erreur de chargement", variant: "destructive" }); }
   159→    finally { setDetailLoading(false); }
   160→  };
   161→
   162→  // ── Guarantee management ───────────────────────────────────────
   163→  const openGuaranteeDialog = async (insurer: Insurer) => {
   164→    try {
   165→      const [insurerRes, guaranteesRes] = await Promise.all([
   166→        fetch(`/api/admin/insurers/${insurer.id}`),
   167→        fetch("/api/admin/guarantees"),
   168→      ]);
   169→      if (!insurerRes.ok || !guaranteesRes.ok) throw new Error();
   170→      const insurerFull: InsurerFull = await insurerRes.json();
   171→      const guarantees: GuaranteeMini[] = await guaranteesRes.json();
   172→      setSelectedInsurer(insurerFull);
   173→      setAllGuarantees(guarantees);
   174→
   175→      const linkMap = new Map(insurerFull.guaranteeLinks.map((l) => [l.guaranteeId, l]));
   176→      const links = guarantees.map((g) => {
   177→        const existing = linkMap.get(g.id);
   178→        return {
   179→          guaranteeId: g.id,
   180→          isEnabled: existing?.isEnabled ?? false,
   181→          customRate: existing?.customRate?.toString() ?? "",
   182→          customPrice: existing?.customPrice?.toString() ?? "",
   183→        };
   184→      });
   185→      setGuaranteeLinks(links);
   186→      setGuaranteeDialogOpen(true);
   187→    } catch { toast({ title: "Erreur de chargement", variant: "destructive" }); }
   188→  };
   189→
   190→  const toggleGuarantee = (idx: number) => {
   191→    setGuaranteeLinks((prev) => prev.map((l, i) => i === idx ? { ...l, isEnabled: !l.isEnabled } : l));
   192→  };
   193→
   194→  const updateLink = (idx: number, field: "customRate" | "customPrice", value: string) => {
   195→    setGuaranteeLinks((prev) => prev.map((l, i) => i === idx ? { ...l, [field]: value } : l));
   196→  };
   197→
   198→  const saveGuarantees = async () => {
   199→    if (!selectedInsurer) return;
   200→    setSavingGuarantees(true);
   201→    try {
   202→      const payload = guaranteeLinks.map((l) => ({
   203→        guaranteeId: l.guaranteeId,
   204→        isEnabled: l.isEnabled,
   205→        customRate: l.customRate ? parseFloat(l.customRate) : null,
   206→        customPrice: l.customPrice ? parseFloat(l.customPrice) : null,
   207→      }));
   208→      const res = await fetch(`/api/admin/insurers/${selectedInsurer.id}/guarantees`, {
   209→        method: "PUT", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ guarantees: payload }),
   210→      });
   211→      if (!res.ok) throw new Error();
   212→      toast({ title: "Garanties mises à jour" });
   213→      setGuaranteeDialogOpen(false);
   214→      fetchInsurers();
   215→    } catch { toast({ title: "Erreur de sauvegarde", variant: "destructive" }); }
   216→    finally { setSavingGuarantees(false); }
   217→  };
   218→
   219→  // ── Render ─────────────────────────────────────────────────────
   220→  if (loading) return <div className="space-y-4">{Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12 rounded-lg" />)}</div>;
   221→
   222→  return (
   223→    <div>
   224→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 mb-6">
   225→        <h1 className="text-2xl font-bold">Assureurs</h1>
   226→        <div className="flex items-center gap-2">
   227→          <div className="relative flex-1 sm:w-64">
   228→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   229→            <Input placeholder="Rechercher..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-9" />
   230→          </div>
   231→          <Button onClick={openCreate} className="bg-[#B9E54D] text-black hover:bg-[#a5d044]"><Plus className="h-4 w-4 mr-1" /> Ajouter</Button>
   232→        </div>
   233→      </div>
   234→
   235→      {/* Desktop table */}
   236→      <div className="hidden md:block rounded-xl border bg-card overflow-hidden">
   237→        <Table>
   238→          <TableHeader>
   239→            <TableRow className="bg-muted/50">
   240→              <TableHead>Nom</TableHead>
   241→              <TableHead>Téléphone</TableHead>
   242→              <TableHead>Email</TableHead>
   243→              <TableHead className="text-center">Note</TableHead>
   244→              <TableHead className="text-center">Vérifié</TableHead>
   245→              <TableHead className="text-center">Offres</TableHead>
   246→              <TableHead className="text-center">Garanties</TableHead>
   247→              <TableHead className="text-center">Statut</TableHead>
   248→              <TableHead className="w-10"></TableHead>
   249→            </TableRow>
   250→          </TableHeader>
   251→          <TableBody>
   252→            {filtered.map((i) => (
   253→              <TableRow key={i.id}>
   254→                <TableCell className="font-medium">{i.name}</TableCell>
   255→                <TableCell className="text-sm text-muted-foreground">{i.phone || "—"}</TableCell>
   256→                <TableCell className="text-sm text-muted-foreground">{i.email || "—"}</TableCell>
   257→                <TableCell className="text-center"><div className="flex items-center justify-center gap-0.5"><Star className="h-3 w-3 fill-amber-400 text-amber-400" /><span className="text-sm">{i.rating}</span></div></TableCell>
   258→                <TableCell className="text-center">{i.isVerified ? <Badge className="bg-emerald-100 text-emerald-800">Oui</Badge> : <Badge variant="secondary">Non</Badge>}</TableCell>
   259→                <TableCell className="text-center">{i._count.offers}</TableCell>
   260→                <TableCell className="text-center">{i._count.guaranteeLinks}</TableCell>
   261→                <TableCell className="text-center"><Switch checked={i.isActive} onCheckedChange={() => handleToggleActive(i)} /></TableCell>
   262→                <TableCell>
   263→                  <DropdownMenu>
   264→                    <DropdownMenuTrigger asChild><Button variant="ghost" size="icon" className="h-8 w-8"><MoreHorizontal className="h-4 w-4" /></Button></DropdownMenuTrigger>
   265→                    <DropdownMenuContent align="end">
   266→                      <DropdownMenuItem onClick={() => openDetail(i)}><Eye className="h-4 w-4 mr-2" /> Voir</DropdownMenuItem>
   267→                      <DropdownMenuItem onClick={() => openGuaranteeDialog(i)}><ShieldCheck className="h-4 w-4 mr-2" /> Gérer les garanties</DropdownMenuItem>
   268→                      <DropdownMenuItem onClick={() => openEdit(i)}><Pencil className="h-4 w-4 mr-2" /> Modifier</DropdownMenuItem>
   269→                      <DropdownMenuItem className="text-destructive" onClick={() => setDeleteId(i.id)}><Trash2 className="h-4 w-4 mr-2" /> Supprimer</DropdownMenuItem>
   270→                    </DropdownMenuContent>
   271→                  </DropdownMenu>
   272→                </TableCell>
   273→              </TableRow>
   274→            ))}
   275→          </TableBody>
   276→        </Table>
   277→        {filtered.length === 0 && <p className="text-center py-8 text-muted-foreground text-sm">Aucun assureur trouvé.</p>}
   278→      </div>
   279→
   280→      {/* Mobile cards */}
   281→      <div className="md:hidden space-y-3">
   282→        {filtered.map((i) => (
   283→          <Card key={i.id} className="p-4">
   284→            <div className="flex items-start justify-between mb-2">
   285→              <div>
   286→                <p className="font-semibold">{i.name}</p>
   287→                <p className="text-xs text-muted-foreground">{i.phone || "—"}</p>
   288→              </div>
   289→              <Switch checked={i.isActive} onCheckedChange={() => handleToggleActive(i)} />
   290→            </div>
   291→            <div className="flex flex-wrap gap-2 text-xs mb-3">
   292→              <Badge variant="outline">{i._count.offers} offres</Badge>
   293→              <Badge variant="outline">{i._count.guaranteeLinks} garanties</Badge>
   294→              {i.isVerified && <Badge className="bg-emerald-100 text-emerald-800">Vérifié</Badge>}
   295→            </div>
   296→            <div className="flex gap-2">
   297→              <Button variant="outline" size="sm" className="flex-1" onClick={() => openDetail(i)}><Eye className="h-3.5 w-3.5 mr-1" /> Voir</Button>
   298→              <Button variant="outline" size="sm" onClick={() => openEdit(i)}><Pencil className="h-3.5 w-3.5" /></Button>
   299→              <Button variant="outline" size="sm" className="text-destructive" onClick={() => setDeleteId(i.id)}><Trash2 className="h-3.5 w-3.5" /></Button>
   300→            </div>
   301→          </Card>
   302→        ))}
   303→      </div>
   304→
   305→      {/* ── Detail Dialog ──────────────────────────────────────────── */}
   306→      <Dialog open={!!detailInsurer || detailLoading} onOpenChange={() => setDetailInsurer(null)}>
   307→        <DialogContent className="max-w-2xl max-h-[90vh]">
   308→          {detailLoading ? (
   309→            <div className="flex items-center justify-center py-12"><Loader2 className="h-6 w-6 animate-spin" /></div>
   310→          ) : detailInsurer ? (
   311→            <>
   312→              <DialogHeader>
   313→                <div className="flex items-center gap-3">
   314→                  <DialogTitle className="text-xl">{detailInsurer.name}</DialogTitle>
   315→                  {detailInsurer.isVerified && (
   316→                    <Badge className="bg-emerald-100 text-emerald-800 flex items-center gap-1">
   317→                      <ShieldCheck className="h-3 w-3" /> Vérifié
   318→                    </Badge>
   319→                  )}
   320→                  <div className="flex items-center gap-0.5 ml-1">
   321→                    <Star className="h-4 w-4 fill-amber-400 text-amber-400" />
   322→                    <span className="font-semibold text-sm">{detailInsurer.rating}</span>
   323→                  </div>
   324→                </div>
   325→                <DialogDescription>{detailInsurer.description || "Aucune description"}</DialogDescription>
   326→              </DialogHeader>
   327→
   328→              <ScrollArea className="max-h-[65vh] pr-2">
   329→                <div className="space-y-6 pb-4">
   330→                  {/* Stats row */}
   331→                  <div className="grid grid-cols-3 gap-3">
   332→                    <Card><CardContent className="p-4 text-center"><p className="text-2xl font-bold">{detailInsurer.offers.length}</p><p className="text-xs text-muted-foreground">Offres</p></CardContent></Card>
   333→                    <Card><CardContent className="p-4 text-center"><p className="text-2xl font-bold">{detailInsurer.guaranteeLinks.length}</p><p className="text-xs text-muted-foreground">Garanties liées</p></CardContent></Card>
   334→                    <Card><CardContent className="p-4 text-center"><p className="text-2xl font-bold">{detailInsurer.offers.reduce((acc, o) => acc + 1, 0)}</p><p className="text-xs text-muted-foreground">Devis reçus</p></CardContent></Card>
   335→                  </div>
   336→
   337→                  {/* Contact info */}
   338→                  <div className="space-y-2">
   339→                    <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide">Contact</h3>
   340→                    <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 text-sm">
   341→                      <div className="flex items-center gap-2">
   342→                        <Phone className="h-4 w-4 text-muted-foreground" />
   343→                        <span>{detailInsurer.phone || "—"}</span>
   344→                      </div>
   345→                      <div className="flex items-center gap-2">
   346→                        <Mail className="h-4 w-4 text-muted-foreground" />
   347→                        <span className="truncate">{detailInsurer.email || "—"}</span>
   348→                      </div>
   349→                      <div className="flex items-center gap-2">
   350→                        <Globe className="h-4 w-4 text-muted-foreground" />
   351→                        <span className="truncate">{detailInsurer.website || "—"}</span>
   352→                      </div>
   353→                    </div>
   354→                  </div>
   355→
   356→                  <Separator />
   357→
   358→                  {/* Offers */}
   359→                  <div className="space-y-2">
   360→                    <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide">Offres ({detailInsurer.offers.length})</h3>
   361→                    {detailInsurer.offers.length > 0 ? (
   362→                      <div className="rounded-xl border overflow-hidden">
   363→                        <Table>
   364→                          <TableHeader>
   365→                            <TableRow className="bg-muted/50">
   366→                              <TableHead>Nom</TableHead>
   367→                              <TableHead>Type</TableHead>
   368→                              <TableHead>Prix</TableHead>
   369→                              <TableHead className="text-center">Statut</TableHead>
   370→                            </TableRow>
   371→                          </TableHeader>
   372→                          <TableBody>
   373→                            {detailInsurer.offers.map((o) => (
   374→                              <TableRow key={o.id}>
   375→                                <TableCell className="font-medium text-sm">{o.name}</TableCell>
   376→                                <TableCell><Badge variant="outline" className="text-xs">{coverageLabels[o.coverageType] || o.coverageType}</Badge></TableCell>
   377→                                <TableCell className="text-sm font-mono">{fmtPrice(o.basePrice)}</TableCell>
   378→                                <TableCell className="text-center">
   379→                                  <Badge className={o.isActive ? "bg-emerald-100 text-emerald-800" : "bg-slate-100 text-slate-800"}>
   380→                                    {o.isActive ? "Active" : "Inactive"}
   381→                                  </Badge>
   382→                                </TableCell>
   383→                              </TableRow>
   384→                            ))}
   385→                          </TableBody>
   386→                        </Table>
   387→                      </div>
   388→                    ) : (
   389→                      <p className="text-sm text-muted-foreground">Aucune offre.</p>
   390→                    )}
   391→                  </div>
   392→
   393→                  <Separator />
   394→
   395→                  {/* Guarantee links */}
   396→                  <div className="space-y-2">
   397→                    <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide">Garanties ({detailInsurer.guaranteeLinks.length})</h3>
   398→                    {detailInsurer.guaranteeLinks.length > 0 ? (
   399→                      <div className="rounded-xl border overflow-hidden">
   400→                        <Table>
   401→                          <TableHeader>
   402→                            <TableRow className="bg-muted/50">
   403→                              <TableHead>Garantie</TableHead>
   404→                              <TableHead>Méthode</TableHead>
   405→                              <TableHead className="text-center">Activée</TableHead>
   406→                              <TableHead className="text-right">Surcharge</TableHead>
   407→                            </TableRow>
   408→                          </TableHeader>
   409→                          <TableBody>
   410→                            {detailInsurer.guaranteeLinks.map((gl) => (
   411→                              <TableRow key={gl.id}>
   412→                                <TableCell className="font-medium text-sm">{gl.guarantee.name}</TableCell>
   413→                                <TableCell><CalcBadge method={gl.guarantee.calcMethod} /></TableCell>
   414→                                <TableCell className="text-center">
   415→                                  <Badge className={gl.isEnabled ? "bg-emerald-100 text-emerald-800" : "bg-slate-100 text-slate-800"}>
   416→                                    {gl.isEnabled ? "Oui" : "Non"}
   417→                                  </Badge>
