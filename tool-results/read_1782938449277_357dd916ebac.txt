   121→/* ── Component ─────────────────────────────────────────────── */
   122→export function CoveragesTab() {
   123→  const { toast } = useToast();
   124→  const [items, setItems] = useState<Coverage[]>([]);
   125→  const [insurers, setInsurers] = useState<InsurerOption[]>([]);
   126→  const [categories, setCategories] = useState<CatOption[]>([]);
   127→  const [search, setSearch] = useState("");
   128→  const [filterInsurer, setFilterInsurer] = useState("");
   129→  const [filterCategory, setFilterCategory] = useState("");
   130→  const [filterCalcType, setFilterCalcType] = useState("");
   131→  const [loading, setLoading] = useState(true);
   132→  const [dialogOpen, setDialogOpen] = useState(false);
   133→  const [editing, setEditing] = useState<Coverage | null>(null);
   134→  const [step, setStep] = useState(1);
   135→  const [step1, setStep1] = useState<Step1Data>(emptyStep1);
   136→  const [calcType, setCalcType] = useState<CalculationType | "">("");
   137→  const [metadata, setMetadata] = useState<Record<string, unknown>>({});
   138→  const [saving, setSaving] = useState(false);
   139→  const [deleteId, setDeleteId] = useState<string | null>(null);
   140→
   141→  // Tariff rules panel
   142→  const [selectedId, setSelectedId] = useState<string | null>(null);
   143→  const [tariffRules, setTariffRules] = useState<TariffRule[]>([]);
   144→  const [trLoading, setTrLoading] = useState(false);
   145→  const [trDialogOpen, setTrDialogOpen] = useState(false);
   146→  const [trEditing, setTrEditing] = useState<TariffRule | null>(null);
   147→  const [trForm, setTrForm] = useState({
   148→    fuelType: "", minFiscalPower: "", maxFiscalPower: "", baseRate: "", fixedAmount: "", minAmount: "", maxAmount: "", formulaName: "",
   149→  });
   150→  const [trSaving, setTrSaving] = useState(false);
   151→
   152→  const fetchItems = useCallback(async () => {
   153→    try {
   154→      const params = new URLSearchParams();
   155→      if (search) params.set("search", search);
   156→      if (filterInsurer) params.set("insurerId", filterInsurer);
   157→      if (filterCategory) params.set("categoryId", filterCategory);
   158→      if (filterCalcType) params.set("calculationType", filterCalcType);
   159→      const res = await fetch(`/api/admin/coverages?${params}`);
   160→      if (!res.ok) throw new Error();
   161→      setItems(await res.json());
   162→    } catch {
   163→      toast({ title: "Erreur", description: "Impossible de charger les garanties", variant: "destructive" });
   164→    } finally {
   165→      setLoading(false);
   166→    }
   167→  }, [search, filterInsurer, filterCategory, filterCalcType, toast]);
   168→
   169→  const fetchLookups = useCallback(async () => {
   170→    try {
   171→      const [insRes, catRes] = await Promise.all([
   172→        fetch("/api/admin/insurers"),
   173→        fetch("/api/admin/coverage-categories"),
   174→      ]);
   175→      if (insRes.ok) setInsurers(await insRes.json());
   176→      if (catRes.ok) setCategories(await catRes.json());
   177→    } catch { /* silent */ }
   178→  }, []);
   179→
   180→  useEffect(() => { fetchLookups(); }, [fetchLookups]);
   181→  useEffect(() => { setLoading(true); fetchItems(); }, [fetchItems]);
   182→
   183→  const fetchTariffRules = useCallback(async (coverageId: string) => {
   184→    setTrLoading(true);
   185→    try {
   186→      const res = await fetch(`/api/admin/coverages/${coverageId}/tariff-rules`);
   187→      if (!res.ok) throw new Error();
   188→      setTariffRules(await res.json());
   189→    } catch {
   190→      toast({ title: "Erreur", description: "Impossible de charger les règles", variant: "destructive" });
   191→    } finally {
   192→      setTrLoading(false);
   193→    }
   194→  }, [toast]);
   195→
   196→  const handleRowClick = (id: string) => {
   197→    if (selectedId === id) {
   198→      setSelectedId(null);
   199→      return;
   200→    }
   201→    setSelectedId(id);
   202→    fetchTariffRules(id);
   203→  };
   204→
   205→  /* ── Wizard ─────────────────────────────────────────────── */
   206→  const openCreate = () => {
   207→    setEditing(null);
   208→    setStep1(emptyStep1);
   209→    setCalcType("");
   210→    setMetadata({});
   211→    setStep(1);
   212→    setDialogOpen(true);
   213→  };
   214→
   215→  const openEdit = (item: Coverage) => {
   216→    setEditing(item);
   217→    setStep1({
   218→      code: item.code,
   219→      name: item.name,
   220→      type: item.type,
   221→      description: item.description ?? "",
   222→      insurerId: item.insurerId,
   223→      categoryId: item.categoryId ?? "",
   224→      isMandatory: item.isMandatory,
   225→      displayOrder: item.displayOrder.toString(),
   226→      isActive: item.isActive,
   227→    });
   228→    setCalcType(item.calculationType as CalculationType);
   229→    setMetadata(item.metadata);
   230→    setStep(1);
   231→    setDialogOpen(true);
   232→  };
   233→
   234→  const buildMetadataFromStep3 = (): Record<string, unknown> => {
   235→    if (calcType === "FREE") return {};
   236→    if (calcType === "FIXED_AMOUNT") {
   237→      return { fixedAmount: (metadata.fixedAmount as number) || 0 };
   238→    }
   239→    if (calcType === "VARIABLE_BASED") {
   240→      const m: Record<string, unknown> = { variable: (metadata.variable as string) || "VN" };
   241→      if (metadata.hasConditional) {
   242→        m.hasConditional = true;
   243→        m.threshold = Number(metadata.threshold) || 0;
   244→        m.rateBelow = Number(metadata.rateBelow) || 0;
   245→        m.rateAbove = Number(metadata.rateAbove) || 0;
   246→      } else {
   247→        m.rate = Number(metadata.rate) || 0;
   248→      }
   249→      if (metadata.franchiseEnabled) {
   250→        m.franchiseEnabled = true;
   251→        m.franchisePercent = Number(metadata.franchisePercent) || 0;
   252→        m.franchiseMin = Number(metadata.franchiseMin) || 0;
   253→      }
   254→      return m;
   255→    }
   256→    if (calcType === "MATRIX_BASED") {
   257→      return { matrixType: (metadata.matrixType as string) || "FISCAL_POWER" };
   258→    }
   259→    return {};
   260→  };
   261→
   262→  const handleSave = async () => {
   263→    if (!step1.code.trim() || !step1.name.trim() || !step1.insurerId || !calcType) {
   264→      toast({ title: "Erreur", description: "Veuillez remplir tous les champs requis", variant: "destructive" });
   265→      return;
   266→    }
   267→    setSaving(true);
   268→    try {
   269→      const url = editing ? `/api/admin/coverages/${editing.id}` : "/api/admin/coverages";
   270→      const method = editing ? "PUT" : "POST";
   271→      const body = {
   272→        ...step1,
   273→        displayOrder: parseInt(step1.displayOrder, 10) || 0,
   274→        calculationType: calcType,
   275→        metadata: buildMetadataFromStep3(),
   276→      };
   277→      const res = await fetch(url, { method, headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
   278→      if (!res.ok) {
   279→        const data = await res.json();
   280→        throw new Error(data.error || "Erreur");
   281→      }
   282→      toast({ title: editing ? "Garantie modifiée" : "Garantie créée" });
   283→      setDialogOpen(false);
   284→      fetchItems();
   285→    } catch (e) {
   286→      toast({ title: "Erreur", description: e instanceof Error ? e.message : "Erreur", variant: "destructive" });
   287→    } finally {
   288→      setSaving(false);
   289→    }
   290→  };
   291→
   292→  const handleDelete = async () => {
   293→    if (!deleteId) return;
   294→    try {
   295→      const res = await fetch(`/api/admin/coverages/${deleteId}`, { method: "DELETE" });
   296→      if (!res.ok) throw new Error();
   297→      toast({ title: "Garantie supprimée" });
   298→      setDeleteId(null);
   299→      if (selectedId === deleteId) setSelectedId(null);
   300→      fetchItems();
   301→    } catch {
   302→      toast({ title: "Erreur", variant: "destructive" });
   303→    }
   304→  };
   305→
   306→  /* ── Tariff Rule CRUD ────────────────────────────────────── */
   307→  const openTrCreate = () => {
   308→    setTrEditing(null);
   309→    setTrForm({ fuelType: "", minFiscalPower: "", maxFiscalPower: "", baseRate: "", fixedAmount: "", minAmount: "", maxAmount: "", formulaName: "" });
   310→    setTrDialogOpen(true);
   311→  };
   312→
   313→  const openTrEdit = (rule: TariffRule) => {
   314→    setTrEditing(rule);
   315→    setTrForm({
   316→      fuelType: rule.fuelType ?? "", minFiscalPower: rule.minFiscalPower?.toString() ?? "", maxFiscalPower: rule.maxFiscalPower?.toString() ?? "",
   317→      baseRate: rule.baseRate?.toString() ?? "", fixedAmount: rule.fixedAmount?.toString() ?? "", minAmount: rule.minAmount?.toString() ?? "",
   318→      maxAmount: rule.maxAmount?.toString() ?? "", formulaName: rule.formulaName ?? "",
   319→    });
   320→    setTrDialogOpen(true);
   321→  };
   322→
   323→  const saveTr = async () => {
   324→    if (!selectedId) return;
   325→    setTrSaving(true);
   326→    try {
   327→      const body = {
   328→        fuelType: trForm.fuelType || null,
   329→        minFiscalPower: trForm.minFiscalPower ? parseInt(trForm.minFiscalPower, 10) : null,
   330→        maxFiscalPower: trForm.maxFiscalPower ? parseInt(trForm.maxFiscalPower, 10) : null,
   331→        baseRate: trForm.baseRate ? parseFloat(trForm.baseRate) : null,
   332→        fixedAmount: trForm.fixedAmount ? parseFloat(trForm.fixedAmount) : null,
   333→        minAmount: trForm.minAmount ? parseFloat(trForm.minAmount) : null,
   334→        maxAmount: trForm.maxAmount ? parseFloat(trForm.maxAmount) : null,
   335→        formulaName: trForm.formulaName || null,
   336→      };
   337→      const res = await fetch(`/api/admin/coverages/${selectedId}/tariff-rules`, {
   338→        method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body),
   339→      });
   340→      if (!res.ok) throw new Error();
   341→      toast({ title: "Règle ajoutée" });
   342→      setTrDialogOpen(false);
   343→      fetchTariffRules(selectedId);
   344→    } catch {
   345→      toast({ title: "Erreur", variant: "destructive" });
   346→    } finally {
   347→      setTrSaving(false);
   348→    }
   349→  };
   350→
   351→  /* ── Render Step 3 ──────────────────────────────────────── */
   352→  const renderStep3 = () => {
   353→    if (calcType === "FREE") {
   354→      return (
   355→        <div className="rounded-lg bg-emerald-50 border border-emerald-200 p-6 text-center">
   356→          <p className="text-emerald-700 font-medium text-lg">🟢 Gratuit</p>
   357→          <p className="text-emerald-600 text-sm mt-1">Aucune configuration supplémentaire requise.</p>
   358→        </div>
   359→      );
   360→    }
   361→    if (calcType === "FIXED_AMOUNT") {
   362→      return (
   363→        <div className="space-y-4">
   364→          <Label>Montant fixe (FCFA) *</Label>
   365→          <Input type="number" value={(metadata.fixedAmount as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, fixedAmount: parseFloat(e.target.value) || 0 })} placeholder="Ex: 50000" />
   366→        </div>
   367→      );
   368→    }
   369→    if (calcType === "VARIABLE_BASED") {
   370→      return (
   371→        <div className="space-y-4">
   372→          <div>
   373→            <Label>Variable</Label>
   374→            <Select value={(metadata.variable as string) || "VN"} onValueChange={(v) => setMetadata({ ...metadata, variable: v })}>
   375→              <SelectTrigger><SelectValue /></SelectTrigger>
   376→              <SelectContent>
   377→                <SelectItem value="VN">Valeur Neuve (VN)</SelectItem>
   378→                <SelectItem value="VN_REPLACEMENT_VALUE">VN / Valeur de remplacement</SelectItem>
   379→              </SelectContent>
   380→            </Select>
   381→          </div>
   382→          <div className="flex items-center gap-2">
   383→            <Checkbox id="conditional" checked={!!metadata.hasConditional} onCheckedChange={(v) => setMetadata({ ...metadata, hasConditional: !!v })} />
   384→            <Label htmlFor="conditional">Taux conditionné par seuil</Label>
   385→          </div>
   386→          {metadata.hasConditional ? (
   387→            <div className="grid grid-cols-3 gap-3 rounded-lg border p-4 bg-muted/50">
   388→              <div><Label className="text-xs">Seuil (FCFA)</Label><Input type="number" value={(metadata.threshold as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, threshold: parseFloat(e.target.value) || 0 })} /></div>
   389→              <div><Label className="text-xs">Taux sous (%)</Label><Input type="number" step="0.01" value={(metadata.rateBelow as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rateBelow: parseFloat(e.target.value) || 0 })} /></div>
   390→              <div><Label className="text-xs">Taux au-dessus (%)</Label><Input type="number" step="0.01" value={(metadata.rateAbove as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rateAbove: parseFloat(e.target.value) || 0 })} /></div>
   391→            </div>
   392→          ) : (
   393→            <div><Label>Taux (%)</Label><Input type="number" step="0.01" value={(metadata.rate as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rate: parseFloat(e.target.value) || 0 })} /></div>
   394→          )}
   395→          <Separator />
   396→          <div className="flex items-center gap-2">
   397→            <Checkbox id="franchise" checked={!!metadata.franchiseEnabled} onCheckedChange={(v) => setMetadata({ ...metadata, franchiseEnabled: !!v })} />
   398→            <Label htmlFor="franchise">Appliquer une franchise</Label>
   399→          </div>
   400→          {metadata.franchiseEnabled && (
   401→            <div className="grid grid-cols-2 gap-3 rounded-lg border p-4 bg-muted/50">
   402→              <div><Label className="text-xs">Franchise (%)</Label><Input type="number" step="0.01" value={(metadata.franchisePercent as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, franchisePercent: parseFloat(e.target.value) || 0 })} /></div>
   403→              <div><Label className="text-xs">Montant min (FCFA)</Label><Input type="number" value={(metadata.franchiseMin as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, franchiseMin: parseFloat(e.target.value) || 0 })} /></div>
   404→            </div>
   405→          )}
   406→        </div>
   407→      );
   408→    }
   409→    if (calcType === "MATRIX_BASED") {
   410→      return (
   411→        <div className="space-y-4">
   412→          <div>
   413→            <Label>Type de matrice</Label>
   414→            <Select value={(metadata.matrixType as string) || "FISCAL_POWER"} onValueChange={(v) => setMetadata({ ...metadata, matrixType: v })}>
   415→              <SelectTrigger><SelectValue /></SelectTrigger>
   416→              <SelectContent>
   417→                <SelectItem value="FISCAL_POWER">Puissance fiscale</SelectItem>
   418→                <SelectItem value="FORMULA">Formule (IC/IPT)</SelectItem>
   419→                <SelectItem value="TIERCE_COMPLETE">Tierce complète</SelectItem>
   420→                <SelectItem value="TIERCE_COLLISION">Tierce collision</SelectItem>
   421→                <SelectItem value="FUEL_TYPE">Type carburant</SelectItem>
   422→                <SelectItem value="SEATS">Nombre de places</SelectItem>
   423→              </SelectContent>
   424→            </Select>
   425→          </div>
   426→          <p className="text-sm text-muted-foreground">
   427→            Les règles tarifaires associées définissent les valeurs de la matrice. Utilisez la section ci-dessous pour les gérer.
   428→          </p>
   429→        </div>
   430→      );
   431→    }
   432→    return null;
   433→  };
   434→
   435→  const selectedCoverage = items.find((c) => c.id === selectedId);
   436→
   437→  return (
   438→    <div className="space-y-4">
   439→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   440→        <h1 className="text-2xl font-bold">Garanties</h1>
   441→        <div className="flex items-center gap-2 flex-wrap">
   442→          <div className="relative">
   443→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   444→            <Input placeholder="Rechercher..." value={search} onChange={(e) => { setSearch(e.target.value); setLoading(true); }} className="pl-9 w-44" />
   445→          </div>
   446→          <Select value={filterInsurer} onValueChange={(v) => { setFilterInsurer(v === "__all__" ? "" : v); setLoading(true); }}>
   447→            <SelectTrigger className="w-36"><SelectValue placeholder="Assureur" /></SelectTrigger>
   448→            <SelectContent>
   449→              <SelectItem value="__all__">Tous</SelectItem>
   450→              {insurers.map((i) => <SelectItem key={i.id} value={i.id}>{i.name}</SelectItem>)}
   451→            </SelectContent>
   452→          </Select>
   453→          <Select value={filterCategory} onValueChange={(v) => { setFilterCategory(v === "__all__" ? "" : v); setLoading(true); }}>
   454→            <SelectTrigger className="w-36"><SelectValue placeholder="Catégorie" /></SelectTrigger>
   455→            <SelectContent>
   456→              <SelectItem value="__all__">Toutes</SelectItem>
   457→              {categories.map((c) => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}
   458→            </SelectContent>
   459→          </Select>
   460→          <Select value={filterCalcType} onValueChange={(v) => { setFilterCalcType(v === "__all__" ? "" : v); setLoading(true); }}>
   461→            <SelectTrigger className="w-36"><SelectValue placeholder="Méthode" /></SelectTrigger>
   462→            <SelectContent>
