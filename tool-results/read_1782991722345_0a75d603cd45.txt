   108→/* ── Component ─────────────────────────────────────────────── */
   109→export function CoveragesTab() {
   110→  const { toast } = useToast();
   111→  const [items, setItems] = useState<Coverage[]>([]);
   112→  const [insurers, setInsurers] = useState<InsurerOption[]>([]);
   113→  const [categories, setCategories] = useState<CatOption[]>([]);
   114→  const [insuranceCategories, setInsuranceCategories] = useState<InsCatOption[]>([]);
   115→  const [search, setSearch] = useState("");
   116→  const [filterInsurer, setFilterInsurer] = useState("");
   117→  const [filterCategory, setFilterCategory] = useState("");
   118→  const [filterCalcType, setFilterCalcType] = useState("");
   119→  const [loading, setLoading] = useState(true);
   120→  const [dialogOpen, setDialogOpen] = useState(false);
   121→  const [editing, setEditing] = useState<Coverage | null>(null);
   122→  const [step, setStep] = useState(1);
   123→  const [step1, setStep1] = useState<Step1Data>(emptyStep1);
   124→  const [calcType, setCalcType] = useState<CalculationType | "">("");
   125→  const [metadata, setMetadata] = useState<Record<string, unknown>>({});
   126→  const [saving, setSaving] = useState(false);
   127→  const [deleteId, setDeleteId] = useState<string | null>(null);
   128→
   129→  // Tariff rules panel
   130→  const [selectedId, setSelectedId] = useState<string | null>(null);
   131→  const [tariffRules, setTariffRules] = useState<TariffRule[]>([]);
   132→  const [trLoading, setTrLoading] = useState(false);
   133→  const [trDialogOpen, setTrDialogOpen] = useState(false);
   134→  const [trEditing, setTrEditing] = useState<TariffRule | null>(null);
   135→  const [trForm, setTrForm] = useState({
   136→    fuelType: "", minFiscalPower: "", maxFiscalPower: "", baseRate: "", fixedAmount: "", minAmount: "", maxAmount: "", formulaName: "",
   137→  });
   138→  const [trSaving, setTrSaving] = useState(false);
   139→
   140→  const fetchItems = useCallback(async () => {
   141→    try {
   142→      const params = new URLSearchParams();
   143→      if (search) params.set("search", search);
   144→      if (filterInsurer) params.set("insurerId", filterInsurer);
   145→      if (filterCategory) params.set("categoryId", filterCategory);
   146→      if (filterCalcType) params.set("calculationType", filterCalcType);
   147→      const res = await fetch(`/api/admin/coverages?${params}`);
   148→      if (!res.ok) throw new Error();
   149→      setItems(await res.json());
   150→    } catch {
   151→      toast({ title: "Erreur", description: "Impossible de charger les garanties", variant: "destructive" });
   152→    } finally {
   153→      setLoading(false);
   154→    }
   155→  }, [search, filterInsurer, filterCategory, filterCalcType, toast]);
   156→
   157→  const fetchLookups = useCallback(async () => {
   158→    try {
   159→      const [insRes, catRes, insCatRes] = await Promise.all([
   160→        fetch("/api/admin/insurers"),
   161→        fetch("/api/admin/coverage-categories"),
   162→        fetch("/api/admin/insurance-categories"),
   163→      ]);
   164→      if (insRes.ok) setInsurers(await insRes.json());
   165→      if (catRes.ok) setCategories(await catRes.json());
   166→      if (insCatRes.ok) setInsuranceCategories(await insCatRes.json());
   167→    } catch { /* silent */ }
   168→  }, []);
   169→
   170→  useEffect(() => { fetchLookups(); }, [fetchLookups]);
   171→  useEffect(() => { setLoading(true); fetchItems(); }, [fetchItems]);
   172→
   173→  const fetchTariffRules = useCallback(async (coverageId: string) => {
   174→    setTrLoading(true);
   175→    try {
   176→      const res = await fetch(`/api/admin/coverages/${coverageId}/tariff-rules`);
   177→      if (!res.ok) throw new Error();
   178→      setTariffRules(await res.json());
   179→    } catch {
   180→      toast({ title: "Erreur", description: "Impossible de charger les règles", variant: "destructive" });
   181→    } finally {
   182→      setTrLoading(false);
   183→    }
   184→  }, [toast]);
   185→
   186→  const handleRowClick = (id: string) => {
   187→    if (selectedId === id) {
   188→      setSelectedId(null);
   189→      return;
   190→    }
   191→    setSelectedId(id);
   192→    fetchTariffRules(id);
   193→  };
   194→
   195→  /* ── Wizard ─────────────────────────────────────────────── */
   196→  const openCreate = () => {
   197→    setEditing(null);
   198→    setStep1(emptyStep1);
   199→    setCalcType("");
   200→    setMetadata({});
   201→    setStep(1);
   202→    setDialogOpen(true);
   203→  };
   204→
   205→  const openEdit = (item: Coverage) => {
   206→    setEditing(item);
   207→    setStep1({
   208→      name: item.name,
   209→      insuranceCategoryId: "", // will be resolved after insuranceCategories load
   210→      description: item.description ?? "",
   211→      insurerId: item.insurerId,
   212→      categoryId: item.categoryId ?? "",
   213→      isMandatory: item.isMandatory,
   214→      displayOrder: item.displayOrder.toString(),
   215→      isActive: item.isActive,
   216→    });
   217→    setCalcType(item.calculationType as CalculationType);
   218→    setMetadata(item.metadata);
   219→    setStep(1);
   220→    setDialogOpen(true);
   221→  };
   222→
   223→  const buildMetadataFromStep3 = (): Record<string, unknown> => {
   224→    if (calcType === "FREE") return {};
   225→    if (calcType === "FIXED_AMOUNT") {
   226→      return { fixedAmount: (metadata.fixedAmount as number) || 0 };
   227→    }
   228→    if (calcType === "VARIABLE_BASED") {
   229→      const m: Record<string, unknown> = { variable: (metadata.variable as string) || "VN" };
   230→      if (metadata.hasConditional) {
   231→        m.hasConditional = true;
   232→        m.threshold = Number(metadata.threshold) || 0;
   233→        m.rateBelow = Number(metadata.rateBelow) || 0;
   234→        m.rateAbove = Number(metadata.rateAbove) || 0;
   235→      } else {
   236→        m.rate = Number(metadata.rate) || 0;
   237→      }
   238→      if (metadata.franchiseEnabled) {
   239→        m.franchiseEnabled = true;
   240→        m.franchisePercent = Number(metadata.franchisePercent) || 0;
   241→        m.franchiseMin = Number(metadata.franchiseMin) || 0;
   242→      }
   243→      return m;
   244→    }
   245→    if (calcType === "MATRIX_BASED") {
   246→      return { matrixType: (metadata.matrixType as string) || "FISCAL_POWER" };
   247→    }
   248→    return {};
   249→  };
   250→
   251→  const handleSave = async () => {
   252→    if (!step1.name.trim() || !step1.insurerId || !calcType) {
   253→      toast({ title: "Erreur", description: "Veuillez remplir tous les champs requis", variant: "destructive" });
   254→      return;
   255→    }
   256→    setSaving(true);
   257→    try {
   258→      const url = editing ? `/api/admin/coverages/${editing.id}` : "/api/admin/coverages";
   259→      const method = editing ? "PUT" : "POST";
   260→      // Resolve insurance category name for the `type` field
   261→      const insCat = insuranceCategories.find((c) => c.id === step1.insuranceCategoryId);
   262→      const body = {
   263→        name: step1.name,
   264→        type: insCat?.name || "",
   265→        description: step1.description,
   266→        insurerId: step1.insurerId,
   267→        categoryId: step1.categoryId || null,
   268→        isMandatory: step1.isMandatory,
   269→        displayOrder: parseInt(step1.displayOrder, 10) || 0,
   270→        isActive: step1.isActive,
   271→        calculationType: calcType,
   272→        metadata: buildMetadataFromStep3(),
   273→      };
   274→      const res = await fetch(url, { method, headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
   275→      if (!res.ok) {
   276→        const data = await res.json();
   277→        throw new Error(data.error || "Erreur");
   278→      }
   279→      toast({ title: editing ? "Garantie modifiée" : "Garantie créée" });
   280→      setDialogOpen(false);
   281→      fetchItems();
   282→    } catch (e) {
   283→      toast({ title: "Erreur", description: e instanceof Error ? e.message : "Erreur", variant: "destructive" });
   284→    } finally {
   285→      setSaving(false);
   286→    }
   287→  };
   288→
   289→  const handleDelete = async () => {
   290→    if (!deleteId) return;
   291→    try {
   292→      const res = await fetch(`/api/admin/coverages/${deleteId}`, { method: "DELETE" });
   293→      if (!res.ok) throw new Error();
   294→      toast({ title: "Garantie supprimée" });
   295→      setDeleteId(null);
   296→      if (selectedId === deleteId) setSelectedId(null);
   297→      fetchItems();
   298→    } catch {
   299→      toast({ title: "Erreur", variant: "destructive" });
   300→    }
   301→  };
   302→
   303→  /* ── Tariff Rule CRUD ────────────────────────────────────── */
   304→  const openTrCreate = () => {
   305→    setTrEditing(null);
   306→    setTrForm({ fuelType: "", minFiscalPower: "", maxFiscalPower: "", baseRate: "", fixedAmount: "", minAmount: "", maxAmount: "", formulaName: "" });
   307→    setTrDialogOpen(true);
   308→  };
   309→
   310→  const openTrEdit = (rule: TariffRule) => {
   311→    setTrEditing(rule);
   312→    setTrForm({
   313→      fuelType: rule.fuelType ?? "", minFiscalPower: rule.minFiscalPower?.toString() ?? "", maxFiscalPower: rule.maxFiscalPower?.toString() ?? "",
   314→      baseRate: rule.baseRate?.toString() ?? "", fixedAmount: rule.fixedAmount?.toString() ?? "", minAmount: rule.minAmount?.toString() ?? "",
   315→      maxAmount: rule.maxAmount?.toString() ?? "", formulaName: rule.formulaName ?? "",
   316→    });
   317→    setTrDialogOpen(true);
   318→  };
   319→
   320→  const saveTr = async () => {
   321→    if (!selectedId) return;
   322→    setTrSaving(true);
   323→    try {
   324→      const body = {
   325→        fuelType: trForm.fuelType || null,
   326→        minFiscalPower: trForm.minFiscalPower ? parseInt(trForm.minFiscalPower, 10) : null,
   327→        maxFiscalPower: trForm.maxFiscalPower ? parseInt(trForm.maxFiscalPower, 10) : null,
   328→        baseRate: trForm.baseRate ? parseFloat(trForm.baseRate) : null,
   329→        fixedAmount: trForm.fixedAmount ? parseFloat(trForm.fixedAmount) : null,
   330→        minAmount: trForm.minAmount ? parseFloat(trForm.minAmount) : null,
   331→        maxAmount: trForm.maxAmount ? parseFloat(trForm.maxAmount) : null,
   332→        formulaName: trForm.formulaName || null,
   333→      };
   334→      const res = await fetch(`/api/admin/coverages/${selectedId}/tariff-rules`, {
   335→        method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body),
   336→      });
   337→      if (!res.ok) throw new Error();
   338→      toast({ title: "Règle ajoutée" });
   339→      setTrDialogOpen(false);
   340→      fetchTariffRules(selectedId);
   341→    } catch {
   342→      toast({ title: "Erreur", variant: "destructive" });
   343→    } finally {
   344→      setTrSaving(false);
   345→    }
   346→  };
   347→
   348→  /* ── Render Step 3 ──────────────────────────────────────── */
   349→  const renderStep3 = () => {
   350→    if (calcType === "FREE") {
   351→      return (
   352→        <div className="rounded-lg bg-emerald-50 border border-emerald-200 p-6 text-center">
   353→          <p className="text-emerald-700 font-medium text-lg">🟢 Gratuit</p>
   354→          <p className="text-emerald-600 text-sm mt-1">Aucune configuration supplémentaire requise.</p>
   355→        </div>
   356→      );
   357→    }
   358→    if (calcType === "FIXED_AMOUNT") {
   359→      return (
   360→        <div className="space-y-4">
   361→          <Label>Montant fixe (FCFA) *</Label>
   362→          <Input className="w-full" type="number" value={(metadata.fixedAmount as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, fixedAmount: parseFloat(e.target.value) || 0 })} placeholder="Ex: 50000" />
   363→        </div>
   364→      );
   365→    }
   366→    if (calcType === "VARIABLE_BASED") {
   367→      return (
   368→        <div className="space-y-4">
   369→          <div>
   370→            <Label>Variable</Label>
   371→            <Select value={(metadata.variable as string) || "VN"} onValueChange={(v) => setMetadata({ ...metadata, variable: v })}>
   372→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   373→              <SelectContent>
   374→                <SelectItem value="VN">Valeur Neuve (VN)</SelectItem>
   375→                <SelectItem value="VN_REPLACEMENT_VALUE">VN / Valeur de remplacement</SelectItem>
   376→              </SelectContent>
   377→            </Select>
   378→          </div>
   379→          <div className="flex items-center gap-2">
   380→            <Checkbox id="conditional" checked={!!metadata.hasConditional} onCheckedChange={(v) => setMetadata({ ...metadata, hasConditional: !!v })} />
   381→            <Label htmlFor="conditional">Taux conditionné par seuil</Label>
   382→          </div>
   383→          {metadata.hasConditional ? (
   384→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 rounded-lg border p-4 bg-muted/50">
   385→              <div><Label className="text-xs">Seuil (FCFA)</Label><Input type="number" value={(metadata.threshold as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, threshold: parseFloat(e.target.value) || 0 })} /></div>
   386→              <div><Label className="text-xs">Taux sous (%)</Label><Input type="number" step="0.01" value={(metadata.rateBelow as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rateBelow: parseFloat(e.target.value) || 0 })} /></div>
   387→              <div><Label className="text-xs">Taux au-dessus (%)</Label><Input type="number" step="0.01" value={(metadata.rateAbove as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rateAbove: parseFloat(e.target.value) || 0 })} /></div>
   388→            </div>
   389→          ) : (
   390→            <div className="w-full"><Label>Taux (%)</Label><Input className="w-full" type="number" step="0.01" value={(metadata.rate as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rate: parseFloat(e.target.value) || 0 })} /></div>
   391→          )}
   392→          <Separator />
   393→          <div className="flex items-center gap-2">
   394→            <Checkbox id="franchise" checked={!!metadata.franchiseEnabled} onCheckedChange={(v) => setMetadata({ ...metadata, franchiseEnabled: !!v })} />
   395→            <Label htmlFor="franchise">Appliquer une franchise</Label>
   396→          </div>
   397→          {metadata.franchiseEnabled && (
   398→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 rounded-lg border p-4 bg-muted/50">
   399→              <div><Label className="text-xs">Franchise (%)</Label><Input type="number" step="0.01" value={(metadata.franchisePercent as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, franchisePercent: parseFloat(e.target.value) || 0 })} /></div>
   400→              <div><Label className="text-xs">Montant min (FCFA)</Label><Input type="number" value={(metadata.franchiseMin as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, franchiseMin: parseFloat(e.target.value) || 0 })} /></div>
   401→            </div>
   402→          )}
   403→        </div>
   404→      );
   405→    }
   406→    if (calcType === "MATRIX_BASED") {
   407→      return (
   408→        <div className="space-y-4">
   409→          <div>
   410→            <Label>Type de matrice</Label>
   411→            <Select value={(metadata.matrixType as string) || "FISCAL_POWER"} onValueChange={(v) => setMetadata({ ...metadata, matrixType: v })}>
   412→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   413→              <SelectContent>
   414→                <SelectItem value="FISCAL_POWER">Puissance fiscale</SelectItem>
   415→                <SelectItem value="FORMULA">Formule (IC/IPT)</SelectItem>
   416→                <SelectItem value="TIERCE_COMPLETE">Tierce complète</SelectItem>
   417→                <SelectItem value="TIERCE_COLLISION">Tierce collision</SelectItem>
   418→                <SelectItem value="FUEL_TYPE">Type carburant</SelectItem>
   419→                <SelectItem value="SEATS">Nombre de places</SelectItem>
   420→              </SelectContent>
   421→            </Select>
   422→          </div>
   423→          <p className="text-sm text-muted-foreground">
   424→            Les règles tarifaires associées définissent les valeurs de la matrice. Utilisez la section ci-dessous pour les gérer.
   425→          </p>
   426→        </div>
   427→      );
   428→    }
   429→    return null;
   430→  };
   431→
   432→  const selectedCoverage = items.find((c) => c.id === selectedId);
   433→
   434→  return (
   435→    <div className="space-y-4">
   436→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   437→        <h1 className="text-2xl font-bold">Garanties</h1>
   438→        <div className="flex items-center gap-2 flex-wrap">
   439→          <div className="relative">
   440→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   441→            <Input placeholder="Rechercher..." value={search} onChange={(e) => { setSearch(e.target.value); setLoading(true); }} className="pl-9 w-44" />
   442→          </div>
   443→          <Select value={filterInsurer} onValueChange={(v) => { setFilterInsurer(v === "__all__" ? "" : v); setLoading(true); }}>
   444→            <SelectTrigger className="w-36"><SelectValue placeholder="Assureur" /></SelectTrigger>
   445→            <SelectContent>
   446→              <SelectItem value="__all__">Tous</SelectItem>
   447→              {insurers.map((i) => <SelectItem key={i.id} value={i.id}>{i.name}</SelectItem>)}
   448→            </SelectContent>
   449→          </Select>
   450→          <Select value={filterCategory} onValueChange={(v) => { setFilterCategory(v === "__all__" ? "" : v); setLoading(true); }}>
   451→            <SelectTrigger className="w-36"><SelectValue placeholder="Catégorie" /></SelectTrigger>
   452→            <SelectContent>
   453→              <SelectItem value="__all__">Toutes</SelectItem>
   454→              {categories.map((c) => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}
   455→            </SelectContent>
   456→          </Select>
   457→          <Select value={filterCalcType} onValueChange={(v) => { setFilterCalcType(v === "__all__" ? "" : v); setLoading(true); }}>
