   129→// ── Helpers ──────────────────────────────────────────────────────
   130→function displayRateOrPrice(g: Guarantee): string {
   131→  if (g.calcMethod === "FREE") return "Gratuit";
   132→  if (g.calcMethod === "FIXED_AMOUNT") return g.fixedPrice ? fmtPrice(g.fixedPrice) : "—";
   133→  if (g.rateConditions && g.rateConditions.length > 0) {
   134→    return g.rateConditions.map((rc) => `${rc.condition}: ${rc.rate}%`).join(" / ");
   135→  }
   136→  if (g.rate != null) return `${g.rate} %`;
   137→  return "—";
   138→}
   139→
   140→function displayCapital(g: Guarantee): string {
   141→  if (!g.capital) return "—";
   142→  return Object.entries(g.capital).map(([k, v]) => `${k}: ${v}`).join(", ");
   143→}
   144→
   145→function displayFranchise(g: Guarantee): string {
   146→  if (!g.franchise) return "—";
   147→  if (g.franchise.description) return g.franchise.description as string;
   148→  const parts: string[] = [];
   149→  if (g.franchise.percent) parts.push(`${g.franchise.percent}%`);
   150→  if (g.franchise.min) parts.push(`min ${fmtPrice(g.franchise.min as number)}`);
   151→  return parts.join(", ") || "—";
   152→}

   154→function parseJsonSafe(str: string): unknown {
   155→  try { return JSON.parse(str); } catch { return null; }
   156→}
