   125→   125→export function InsurerOffersTab() {
   126→   126→  const { user } = useAppStore();
   127→   127→  const { toast } = useToast();
   128→   128→
   129→   129→  const [insurerId, setInsurerId] = useState<string | null>(null);
   130→   130→  const [offers, setOffers] = useState<Offer[]>([]);
   131→   131→  const [categories, setCategories] = useState<Category[]>([]);
   132→   132→  const [loading, setLoading] = useState(true);
   133→   133→  const [error, setError] = useState<string | null>(null);
   134→   134→
   135→   135→  // Dialog state
   136→   136→  const [dialogOpen, setDialogOpen] = useState(false);
   137→   137→  const [editingOffer, setEditingOffer] = useState<Offer | null>(null);
   138→   138→  const [form, setForm] = useState<OfferFormData>(emptyForm);
   139→   139→  const [submitting, setSubmitting] = useState(false);
   140→   140→
   141→   141→  // Delete dialog
   142→   142→  const [deleteTarget, setDeleteTarget] = useState<Offer | null>(null);
   143→   143→  const [deleting, setDeleting] = useState(false);
   144→   144→
   145→   145→  // Coverages for guarantee selection
   146→   146→  const [formCoverages, setFormCoverages] = useState<CoverageMini[]>([]);
   147→   147→  const [formCoveragesLoading, setFormCoveragesLoading] = useState(false);
   148→   148→
   149→   149→  /* ── Fetch helpers ── */
   150→   150→  const fetchInsurer = useCallback(async (userId: string) => {
   151→   151→    try {
   152→   152→      const res = await fetch(`/api/insurer/account?userId=${userId}`);
   153→   153→      if (!res.ok) throw new Error("Compte assureur non trouvé");
   154→   154→      const data = await res.json();
   155→   155→      setInsurerId(data.id);
   156→   156→      return data.id;
   157→   157→    } catch {
   158→   158→      setError("Impossible de charger votre compte assureur");
   159→   159→      return null;
   160→   160→    }
   161→   161→  }, []);
   162→   162→
   163→   163→  const fetchOffers = useCallback(async (iid: string) => {
   164→   164→    try {
   165→   165→      const res = await fetch(`/api/insurer/offers?insurerId=${iid}`);
   166→   166→      if (!res.ok) throw new Error();
   167→   167→      const data = await res.json();
   168→   168→      setOffers(data.offers || []);
   169→   169→    } catch {
   170→   170→      setError("Erreur lors du chargement des offres");
   171→   171→    }
   172→   172→  }, []);
   173→   173→
   174→   174→  const fetchCategories = useCallback(async () => {
   175→   175→    try {
   176→   176→      const res = await fetch("/api/admin/insurance-categories");
   177→   177→      if (!res.ok) return;
   178→   178→      const data = await res.json();
   179→   179→      setCategories(data);
   180→   180→    } catch {
   181→   181→      /* ignore */
   182→   182→    }
   183→   183→  }, []);
   184→   184→
   185→   185→  const refreshData = useCallback(async () => {
   186→   186→    if (!insurerId) return;
   187→   187→    await fetchOffers(insurerId);
   188→   188→  }, [insurerId, fetchOffers]);
   189→   189→
   190→   190→  /* ── Init ── */
   191→   191→  useEffect(() => {
   192→   192→    if (!user.id) {
   193→   193→      setLoading(false);
   194→   194→      return;
   195→   195→    }
   196→   196→    (async () => {
   197→   197→      const iid = await fetchInsurer(user.id!);
   198→   198→      if (iid) await fetchOffers(iid);
   199→   199→      await fetchCategories();
   200→   200→      setLoading(false);
   201→   201→    })();
   202→   202→  }, [user.id]);
   203→   203→
   204→   204→  /* ── Fetch coverages when insurerId is known (for guarantee selection) ── */
   205→   205→  useEffect(() => {
   206→   206→    if (!insurerId) {
   207→   207→      setFormCoverages([]);
   208→   208→      return;
   209→   209→    }
   210→   210→    setFormCoveragesLoading(true);
   211→   211→    fetch(`/api/insurer/coverages?insurerId=${insurerId}`)
   212→   212→      .then((r) => {
   213→   213→        if (r.ok) return r.json();
   214→   214→        return { coverages: [] };
   215→   215→      })
   216→   216→      .then((data) => setFormCoverages(data.coverages || []))
   217→   217→      .catch(() => setFormCoverages([]))
   218→   218→      .finally(() => setFormCoveragesLoading(false));
   219→   219→  }, [insurerId]);
   220→   220→
   221→   221→  /* ── Form helpers ── */
   222→   222→  const openCreate = () => {
   223→   223→    setEditingOffer(null);
   224→   224→    setForm(emptyForm);
   225→   225→    setDialogOpen(true);
   226→   226→  };
   227→   227→
   228→   228→  const openEdit = (offer: Offer) => {
   229→   229→    setEditingOffer(offer);
   230→   230→    setForm({
   231→   231→      name: offer.name,
   232→   232→      categoryId: offer.category?.id || "",
   233→   233→      description: offer.description || "",
   234→   234→      priceMin: offer.priceMin != null ? String(offer.priceMin) : "",
   235→   235→      priceMax: offer.priceMax != null ? String(offer.priceMax) : "",
   236→   236→      coverageAmount:
   237→   237→        offer.coverageAmount != null ? String(offer.coverageAmount) : "",
   238→   238→      deductible: String(offer.deductible),
   239→   239→      contractType: offer.contractType || "",
   240→   240→      selectedGuarantees: offer.features || [],
   241→   241→      isActive: offer.isActive,
   242→   242→    });
   243→   243→    setDialogOpen(true);
   244→   244→  };
   245→   245→
   246→   246→  const handleSubmit = async () => {
   247→   247→    if (!form.name.trim() || !insurerId) return;
   248→   248→    setSubmitting(true);
   249→   249→
   250→   250→    const payload = {
   251→   251→      insurerId,
   252→   252→      categoryId: form.categoryId || null,
   253→   253→      name: form.name.trim(),
   254→   254→      description: form.description.trim() || null,
   255→   255→      priceMin: form.priceMin ? Number(form.priceMin) : null,
   256→   256→      priceMax: form.priceMax ? Number(form.priceMax) : null,
   257→   257→      coverageAmount: form.coverageAmount ? Number(form.coverageAmount) : null,
   258→   258→      deductible: form.deductible ? Number(form.deductible) : 0,
   259→   259→      contractType: form.contractType || null,
   260→   260→      features: form.selectedGuarantees,
   261→   261→      isActive: form.isActive,
   262→   262→    };
   263→   263→
   264→   264→    try {
   265→   265→      const url = editingOffer
   266→   266→        ? `/api/insurer/offers/${editingOffer.id}`
   267→   267→        : "/api/insurer/offers";
   268→   268→      const method = editingOffer ? "PUT" : "POST";
   269→   269→
   270→   270→      const res = await fetch(url, {
   271→   271→        method,
   272→   272→        headers: { "Content-Type": "application/json" },
   273→   273→        body: JSON.stringify(payload),
   274→   274→      });
   275→   275→
   276→   276→      if (!res.ok) {
   277→   277→        const err = await res.json();
   278→   278→        throw new Error(err.error || "Erreur");
   279→   279→      }
   280→   280→
   281→   281→      toast({
   282→   282→        title: editingOffer
   283→   283→          ? "Offre mise à jour avec succès"
   284→   284→          : "Offre créée avec succès",
   285→   285→        description: editingOffer
   286→   286→          ? `"${form.name}" a été modifiée.`
   287→   287→          : `"${form.name}" a été ajoutée à vos offres.`,
   288→   288→      });
   289→   289→
   290→   290→      setDialogOpen(false);
   291→   291→      await refreshData();
   292→   292→    } catch (e) {
   293→   293→      toast({
   294→   294→        title: "Erreur",
   295→   295→        description: e instanceof Error ? e.message : "Une erreur est survenue",
   296→   296→        variant: "destructive",
   297→   297→      });
   298→   298→    } finally {
   299→   299→      setSubmitting(false);
   300→   300→    }
   301→   301→  };
   302→   302→
   303→   303→  const handleDelete = async () => {
   304→   304→    if (!deleteTarget) return;
   305→   305→    setDeleting(true);
   306→   306→    try {
   307→   307→      const res = await fetch(`/api/insurer/offers/${deleteTarget.id}`, {
   308→   308→        method: "DELETE",
   309→   309→      });
   310→   310→      if (!res.ok) throw new Error();
   311→   311→      toast({
   312→   312→        title: "Offre supprimée",
   313→   313→        description: `"${deleteTarget.name}" a été désactivée.`,
   314→   314→      });
   315→   315→      setDeleteTarget(null);
   316→   316→      await refreshData();
   317→   317→    } catch {
   318→   318→      toast({
   319→   319→        title: "Erreur",
   320→   320→        description: "Impossible de supprimer l'offre",
   321→   321→        variant: "destructive",
   322→   322→      });
   323→   323→    } finally {
   324→   324→      setDeleting(false);
   325→   325→    }
   326→   326→  };
   327→   327→
   328→   328→  const toggleGuarantee = (coverageName: string, checked: boolean | "indeterminate") => {
   329→   329→    if (checked === "indeterminate") return;
   330→   330→    setForm((f) => ({
   331→   331→      ...f,
   332→   332→      selectedGuarantees: checked
   333→   333→        ? [...f.selectedGuarantees, coverageName]
   334→   334→        : f.selectedGuarantees.filter((n) => n !== coverageName),
   335→   335→    }));
   336→   336→  };
   337→   337→
   338→   338→  /* ── Group coverages by category for display ── */
   339→   339→  const groupedCoverages = formCoverages.reduce<
   340→   340→    Record<string, CoverageMini[]>
   341→   341→  >((acc, c) => {
   342→   342→    const key = c.category?.name || "Autre";
   343→   343→    (acc[key] ??= []).push(c);
   344→   344→    return acc;
   345→   345→  }, {});
   346→   346→
   347→   347→  /* ── Render ── */
   348→   348→  return (
   349→   349→    <div className="space-y-6">
   350→   350→      {/* Header */}
   351→   351→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   352→   352→        <div>
   353→   353→          <h2 className="text-2xl font-bold">Mes Offres</h2>
   354→   354→          <p className="text-muted-foreground mt-1">
   355→   355→            Gérez vos offres d&apos;assurance disponibles sur NOLI.
   356→   356→          </p>
   357→   357→        </div>
   358→   358→        <Button
   359→   359→          className="bg-[#B9E54D] text-black hover:bg-[#a5d044]"
   360→   360→          onClick={openCreate}
   361→   361→        >
   362→   362→          <Plus className="mr-2 h-4 w-4" />
   363→   363→          Créer une offre
   364→   364→        </Button>
   365→   365→      </div>
   366→   366→
   367→   367→      {/* Loading */}
   368→   368→      {loading && (
   369→   369→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   370→   370→          {Array.from({ length: 6 }).map((_, i) => (
   371→   371→            <Card key={i}>
   372→   372→              <CardHeader className="pb-3">
   373→   373→                <Skeleton className="h-5 w-3/4" />
   374→   374→                <Skeleton className="h-4 w-1/2 mt-2" />
   375→   375→              </CardHeader>
   376→   376→              <CardContent className="space-y-3">
   377→   377→                <Skeleton className="h-4 w-full" />
   378→   378→                <Skeleton className="h-4 w-2/3" />
   379→   379→                <div className="flex gap-2 pt-2">
   380→   380→                  <Skeleton className="h-6 w-20 rounded-full" />
   381→   381→                  <Skeleton className="h-6 w-24 rounded-full" />
   382→   382→                </div>
   383→   383→              </CardContent>
   384→   384→            </Card>
   385→   385→          ))}
   386→   386→        </div>
   387→   387→      )}
   388→   388→
   389→   389→      {/* Error */}
   390→   390→      {error && !loading && (
   391→   391→        <div className="flex flex-col items-center justify-center py-16 text-center">
   392→   392→          <div className="flex h-16 w-16 items-center justify-center rounded-full bg-red-100 dark:bg-red-900/30 mb-4">
   393→   393→            <XCircle className="h-8 w-8 text-red-500" />
   394→   394→          </div>
   395→   395→          <h3 className="text-lg font-semibold">Erreur de chargement</h3>
   396→   396→          <p className="text-sm text-muted-foreground mt-1">{error}</p>
   397→   397→        </div>
   398→   398→      )}
   399→   399→
   400→   400→      {/* Empty */}
   401→   401→      {!loading && !error && offers.length === 0 && (
   402→   402→        <div className="flex flex-col items-center justify-center py-16 text-center">
   403→   403→          <div className="flex h-16 w-16 items-center justify-center rounded-full bg-muted mb-4">
   404→   404→            <Car className="h-8 w-8 text-muted-foreground" />
   405→   405→          </div>
   406→   406→          <h3 className="text-lg font-semibold">Aucune offre trouvée</h3>
   407→   407→          <p className="text-sm text-muted-foreground mt-1 max-w-md">
   408→   408→            Vous n&apos;avez pas encore d&apos;offres d&apos;assurance.
   409→   409→            Cliquez sur &quot;Créer une offre&quot; pour commencer.
   410→   410→          </p>
   411→   411→        </div>
   412→   412→      )}
   413→   413→
   414→   414→      {/* Offers grid */}
   415→   415→      {!loading && !error && offers.length > 0 && (
   416→   416→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   417→   417→          {offers.map((offer) => (
   418→   418→            <Card
   419→   419→              key={offer.id}
   420→   420→              className="hover:shadow-md transition-shadow"
   421→   421→            >
   422→   422→              <CardHeader className="pb-3">

   424→   424→                  <div className="space-y-1 min-w-0 flex-1">
   425→   425→                    <CardTitle className="text-base truncate">
   426→   426→                      {offer.name}
   427→   427→                    </CardTitle>
   428→   428→                    <p className="text-xs text-muted-foreground">
   429→   429→                      {offer.category?.name || "Non catégorisé"}
   430→   430→                    </p>
   431→   431→                  </div>
   432→   432→                  <div className="flex items-center gap-1 shrink-0 ml-2">
   433→   433→                    {offer.isActive ? (
   434→   434→                      <Badge className="border-0 bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400 text-xs">
   435→   435→                        <CheckCircle2 className="mr-1 h-3 w-3" />
   436→   436→                        Active
   437→   437→                      </Badge>
   438→   438→                    ) : (
   439→   439→                      <Badge variant="secondary" className="text-xs">
   440→   440→                        Inactive
   441→   441→                      </Badge>
   442→   442→                    )}
   443→   443→                  </div>
   444→   444→                </div>
   445→   445→              </CardHeader>
   446→   446→              <CardContent className="space-y-3">
   447→   447→                {/* Price range */}
   448→   448→                <div className="flex items-baseline gap-1">
   449→   449→                  <span className="text-xl font-bold">
   450→   450→                    {formatPrice(offer.priceMin)}
   451→   451→                  </span>
   452→   452→                  {offer.priceMax &&
   453→   453→                    offer.priceMax !== offer.priceMin && (

   457→   457→                    )}

   470→   470→                      Capital : {formatPrice(offer.coverageAmount)}
   471→   471→                    </Badge>
   472→   472→                  )}
   473→   473→                </div>

   475→   475→                {/* Description */}

   481→   481→
   482→   482→                {/* Features / Guarantees */}
   483→   483→                {offer.features.length > 0 && (
   484→   484→                  <div className="space-y-1.5">
   485→   485→                    <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">
