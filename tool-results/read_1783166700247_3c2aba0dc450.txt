    98→

   144→
   145→export function InsurerGuaranteesTab() {
   146→  const { user } = useAppStore();
   147→  const { toast } = useToast();
   148→
   149→  const [insurerId, setInsurerId] = useState<string | null>(null);
   150→  const [coverages, setCoverages] = useState<Coverage[]>([]);
   151→  const [categories, setCategories] = useState<CoverageCategory[]>([]);
   152→  const [loading, setLoading] = useState(true);
   153→  const [error, setError] = useState<string | null>(null);
   154→
   155→  // Wizard state
   156→  const [dialogOpen, setDialogOpen] = useState(false);
   157→  const [editingItem, setEditingItem] = useState<Coverage | null>(null);
   158→  const [step, setStep] = useState(1);
   159→  const [step1, setStep1] = useState<Step1Data>(emptyStep1);
   160→  const [calcType, setCalcType] = useState<CalculationType | "">("");
   161→  const [metadata, setMetadata] = useState<Record<string, unknown>>({});
   162→  const [saving, setSaving] = useState(false);
   163→
   164→  // Delete dialog
   165→  const [deleteTarget, setDeleteTarget] = useState<Coverage | null>(null);
   166→  const [deleting, setDeleting] = useState(false);
   167→
   168→  /* ── Fetch helpers ── */
   169→  const fetchInsurer = useCallback(async (userId: string) => {
   170→    try {
   171→      const res = await fetch(`/api/insurer/account?userId=${userId}`);
   172→      if (!res.ok) throw new Error();
   173→      const data = await res.json();
   174→      setInsurerId(data.id);
   175→      return data.id;
   176→    } catch {
   177→      setError("Impossible de charger votre compte assureur");
   178→      return null;
   179→    }
   180→  }, []);
   181→
   182→  const fetchCoverages = useCallback(async (iid: string) => {
   183→    try {
   184→      const res = await fetch(`/api/insurer/coverages?insurerId=${iid}`);
   185→      if (!res.ok) throw new Error();
   186→      const data = await res.json();
   187→      setCoverages(data.coverages || []);
   188→    } catch {
   189→      setError("Erreur lors du chargement des garanties");
   190→    }
   191→  }, []);
   192→
   193→  const fetchCategories = useCallback(async () => {
   194→    try {
   195→      const res = await fetch("/api/admin/coverage-categories");
   196→      if (!res.ok) return;
   197→      const data = await res.json();
   198→      setCategories(data);
   199→    } catch {
   200→      /* ignore */
   201→    }
   202→  }, []);
   203→
   204→  const refreshData = useCallback(async () => {
   205→    if (!insurerId) return;
   206→    await fetchCoverages(insurerId);
   207→  }, [insurerId, fetchCoverages]);
   208→
   209→  /* ── Init ── */
   210→  useEffect(() => {
   211→    if (!user.id) {
   212→      setLoading(false);
   213→      return;
   214→    }
   215→    (async () => {
   216→      const iid = await fetchInsurer(user.id!);
   217→      if (iid) await fetchCoverages(iid);
   218→      await fetchCategories();
   219→      setLoading(false);
   220→    })();
   221→  }, [user.id]);
   222→
   223→  /* ── Wizard ── */
   224→  const openCreate = () => {
   225→    setEditingItem(null);
   226→    setStep1(emptyStep1);
   227→    setCalcType("");
   228→    setMetadata({});
   229→    setStep(1);
   230→    setDialogOpen(true);
   231→  };
   232→
   233→  const openEdit = (item: Coverage) => {
   234→    setEditingItem(item);
   235→    setStep1({
   236→      name: item.name,
   237→      categoryId: item.category?.id || "",
   238→      description: item.description || "",
   239→      isMandatory: item.isMandatory,
   240→    });
   241→    setCalcType(item.calculationType as CalculationType);
   242→    try {
   243→      setMetadata(
   244→        typeof item.metadata === "string"
   245→          ? JSON.parse(item.metadata)
   246→          : (item as unknown as { metadata: Record<string, unknown> }).metadata || {}
   247→      );
   248→    } catch {
   249→      setMetadata({});
   250→    }
   251→    setStep(1);
   252→    setDialogOpen(true);
   253→  };
   254→
   255→  const buildMetadataFromStep3 = (): Record<string, unknown> => {
   256→    if (calcType === "FREE") return {};
   257→    if (calcType === "FIXED_AMOUNT") {
   258→      return { fixedAmount: (metadata.fixedAmount as number) || 0 };
   259→    }
   260→    if (calcType === "VARIABLE_BASED") {
   261→      const m: Record<string, unknown> = {
   262→        variable: (metadata.variable as string) || "VN",
   263→      };
   264→      if (metadata.hasConditional) {
   265→        m.hasConditional = true;
   266→        m.threshold = Number(metadata.threshold) || 0;
   267→        m.rateBelow = Number(metadata.rateBelow) || 0;
   268→        m.rateAbove = Number(metadata.rateAbove) || 0;
   269→      } else {
   270→        m.rate = Number(metadata.rate) || 0;
   271→      }
   272→      if (metadata.franchiseEnabled) {
   273→        m.franchiseEnabled = true;
   274→        m.franchisePercent = Number(metadata.franchisePercent) || 0;
   275→        m.franchiseMin = Number(metadata.franchiseMin) || 0;
   276→      }
   277→      return m;
   278→    }
   279→    if (calcType === "MATRIX_BASED") {
   280→      return {
   281→        matrixType: (metadata.matrixType as string) || "FISCAL_POWER",
   282→      };
   283→    }
   284→    return {};
   285→  };
   286→
   287→  const handleSave = async () => {
   288→    if (!step1.name.trim() || !insurerId || !calcType) {
   289→      toast({
   290→        title: "Erreur",
   291→        description: "Veuillez remplir tous les champs requis",
   292→        variant: "destructive",
   293→      });
   294→      return;
   295→    }
   296→    setSaving(true);
   297→
   298→    const payload = {
   299→      insurerId,
   300→      categoryId: step1.categoryId || null,
   301→      name: step1.name.trim(),
   302→      description: step1.description.trim() || null,
   303→      calculationType: calcType,
   304→      isMandatory: step1.isMandatory,
   305→      metadata: buildMetadataFromStep3(),
   306→    };
   307→
   308→    try {
   309→      const url = editingItem
   310→        ? `/api/insurer/coverages/${editingItem.id}`
   311→        : "/api/insurer/coverages";
   312→      const method = editingItem ? "PUT" : "POST";
   313→
   314→      const res = await fetch(url, {
   315→        method,
   316→        headers: { "Content-Type": "application/json" },
   317→        body: JSON.stringify(payload),
   318→      });
   319→
   320→      if (!res.ok) {
   321→        const err = await res.json();
   322→        throw new Error(err.error || "Erreur");
   323→      }
   324→
   325→      toast({
   326→        title: editingItem
   327→          ? "Garantie mise à jour avec succès"
   328→          : "Garantie créée avec succès",
   329→        description: editingItem
   330→          ? `"${step1.name}" a été modifiée.`
   331→          : `"${step1.name}" a été ajoutée à vos garanties.`,
   332→      });
   333→
   334→      setDialogOpen(false);
   335→      await refreshData();
   336→    } catch (e) {
   337→      toast({
   338→        title: "Erreur",
   339→        description: e instanceof Error ? e.message : "Une erreur est survenue",
   340→        variant: "destructive",
   341→      });
   342→    } finally {
   343→      setSaving(false);
   344→    }
   345→  };
   346→
   347→  const handleDelete = async () => {
   348→    if (!deleteTarget) return;
   349→    setDeleting(true);
   350→    try {
   351→      const res = await fetch(
   352→        `/api/insurer/coverages/${deleteTarget.id}`,
   353→        { method: "DELETE" }
   354→      );
   355→      if (!res.ok) throw new Error();
   356→      toast({
   357→        title: "Garantie supprimée",
   358→        description: `"${deleteTarget.name}" a été désactivée.`,
   359→      });
   360→      setDeleteTarget(null);
   361→      await refreshData();
   362→    } catch {
   363→      toast({
   364→        title: "Erreur",
   365→        description: "Impossible de supprimer la garantie",
   366→        variant: "destructive",
   367→      });
   368→    } finally {
   369→      setDeleting(false);
   370→    }
   371→  };
   372→
   373→  /* ── Render Step 3 ── */
   374→  const renderStep3 = () => {
   375→    if (calcType === "FREE") {
   376→      return (
   377→        <div className="rounded-lg bg-emerald-50 border border-emerald-200 dark:bg-emerald-900/20 dark:border-emerald-800 p-6 text-center">
   378→          <p className="text-emerald-700 dark:text-emerald-400 font-medium text-lg">
   379→            🟢 Gratuit
   380→          </p>
   381→          <p className="text-emerald-600 dark:text-emerald-500 text-sm mt-1">
   382→            Aucune configuration supplémentaire requise.
   383→          </p>
   384→        </div>
   385→      );
   386→    }
   387→    if (calcType === "FIXED_AMOUNT") {
   388→      return (
   389→        <div className="space-y-4">
   390→          <div className="w-full">
   391→            <Label>Montant fixe (FCFA) *</Label>
   392→            <Input
   393→              className="w-full"
   394→              type="number"
   395→              value={(metadata.fixedAmount as number) ?? ""}
   396→              onChange={(e) =>
   397→                setMetadata({
   398→                  ...metadata,
   399→                  fixedAmount: parseFloat(e.target.value) || 0,
   400→                })
   401→              }
   402→              placeholder="Ex: 50000"
   403→            />
   404→          </div>
   405→        </div>
   406→      );
   407→    }
   408→    if (calcType === "VARIABLE_BASED") {
   409→      return (
   410→        <div className="space-y-4">
   411→          <div className="w-full">
   412→            <Label>Variable</Label>
   413→            <Select
   414→              value={(metadata.variable as string) || "VN"}
   415→              onValueChange={(v) =>
   416→                setMetadata({ ...metadata, variable: v })
   417→              }
   418→            >
   419→              <SelectTrigger className="w-full">
   420→                <SelectValue />
   421→              </SelectTrigger>
   422→              <SelectContent>
   423→                <SelectItem value="VN">Valeur Neuve (VN)</SelectItem>
   424→                <SelectItem value="VN_REPLACEMENT_VALUE">
   425→                  VN / Valeur de remplacement
   426→                </SelectItem>
   427→              </SelectContent>
   428→            </Select>
   429→          </div>
   430→          <div className="flex items-center gap-2">
   431→            <Checkbox
   432→              id="conditional"
   433→              checked={!!metadata.hasConditional}
   434→              onCheckedChange={(v) =>
   435→                setMetadata({ ...metadata, hasConditional: !!v })
   436→              }
   437→            />
   438→            <Label htmlFor="conditional">Taux conditionné par seuil</Label>
   439→          </div>
   440→          {metadata.hasConditional ? (
   441→            <div className="w-full grid grid-cols-1 sm:grid-cols-3 gap-3 rounded-lg border p-4 bg-muted/50">
   442→              <div>
   443→                <Label className="text-xs">Seuil (FCFA)</Label>
   444→                <Input
   445→                  className="w-full"
   446→                  type="number"
   447→                  value={(metadata.threshold as number) ?? ""}
   448→                  onChange={(e) =>
   449→                    setMetadata({
   450→                      ...metadata,
   451→                      threshold: parseFloat(e.target.value) || 0,
   452→                    })
   453→                  }
   454→                />
   455→              </div>
   456→              <div>
   457→                <Label className="text-xs">Taux sous (%)</Label>
   458→                <Input
   459→                  className="w-full"
   460→                  type="number"
   461→                  step="0.01"
   462→                  value={(metadata.rateBelow as number) ?? ""}
   463→                  onChange={(e) =>
   464→                    setMetadata({
   465→                      ...metadata,
   466→                      rateBelow: parseFloat(e.target.value) || 0,
   467→                    })
   468→                  }
   469→                />
   470→              </div>
   471→              <div>
   472→                <Label className="text-xs">Taux au-dessus (%)</Label>
   473→                <Input
   474→                  className="w-full"
   475→                  type="number"
   476→                  step="0.01"
   477→                  value={(metadata.rateAbove as number) ?? ""}
   478→                  onChange={(e) =>
   479→                    setMetadata({
   480→                      ...metadata,
   481→                      rateAbove: parseFloat(e.target.value) || 0,
   482→                    })
   483→                  }
   484→                />
   485→              </div>
   486→            </div>
   487→          ) : (
   488→            <div className="w-full">
   489→              <Label>Taux (%)</Label>
   490→              <Input
   491→                className="w-full"
   492→                type="number"
   493→                step="0.01"
   494→                value={(metadata.rate as number) ?? ""}
   495→                onChange={(e) =>
   496→                  setMetadata({
   497→                    ...metadata,
   498→                    rate: parseFloat(e.target.value) || 0,
   499→                  })
   500→                }
   501→              />
   502→            </div>
   503→          )}
   504→          <Separator />
   505→          <div className="flex items-center gap-2">
   506→            <Checkbox
   507→              id="franchise"
   508→              checked={!!metadata.franchiseEnabled}
   509→              onCheckedChange={(v) =>
   510→                setMetadata({ ...metadata, franchiseEnabled: !!v })
   511→              }
   512→            />
   513→            <Label htmlFor="franchise">Appliquer une franchise</Label>
   514→          </div>
   515→          {metadata.franchiseEnabled && (
   516→            <div className="w-full grid grid-cols-1 sm:grid-cols-2 gap-3 rounded-lg border p-4 bg-muted/50">
   517→              <div>
   518→                <Label className="text-xs">Franchise (%)</Label>
   519→                <Input
   520→                  className="w-full"
   521→                  type="number"
   522→                  step="0.01"
   523→                  value={(metadata.franchisePercent as number) ?? ""}
   524→                  onChange={(e) =>
   525→                    setMetadata({
   526→                      ...metadata,
   527→                      franchisePercent: parseFloat(e.target.value) || 0,
   528→                    })
   529→                  }
   530→                />
   531→              </div>
   532→              <div>
   533→                <Label className="text-xs">Montant min (FCFA)</Label>
   534→                <Input
   535→                  className="w-full"
   536→                  type="number"
   537→                  value={(metadata.franchiseMin as number) ?? ""}
   538→                  onChange={(e) =>
   539→                    setMetadata({
   540→                      ...metadata,
   541→                      franchiseMin: parseFloat(e.target.value) || 0,
   542→                    })
   543→                  }
   544→                />
   545→              </div>
   546→            </div>
   547→          )}
   548→        </div>
   549→      );
   550→    }
   551→    if (calcType === "MATRIX_BASED") {
   552→      return (
   553→        <div className="space-y-4">
   554→          <div className="w-full">
   555→            <Label>Type de matrice</Label>
   556→            <Select
   557→              value={(metadata.matrixType as string) || "FISCAL_POWER"}
   558→              onValueChange={(v) =>
   559→                setMetadata({ ...metadata, matrixType: v })
   560→              }
   561→            >
   562→              <SelectTrigger className="w-full">
   563→                <SelectValue />
   564→              </SelectTrigger>
   565→              <SelectContent>
   566→                <SelectItem value="FISCAL_POWER">
   567→                  Puissance fiscale
   568→                </SelectItem>
   569→                <SelectItem value="FORMULA">Formule (IC/IPT)</SelectItem>
   570→                <SelectItem value="TIERCE_COMPLETE">
   571→                  Tierce complète
   572→                </SelectItem>
   573→                <SelectItem value="TIERCE_COLLISION">
   574→                  Tierce collision
   575→                </SelectItem>
   576→                <SelectItem value="FUEL_TYPE">Type carburant</SelectItem>
   577→                <SelectItem value="SEATS">Nombre de places</SelectItem>
   578→              </SelectContent>
   579→            </Select>
   580→          </div>
   581→          <p className="text-sm text-muted-foreground">
   582→            Les règles tarifaires associées définissent les valeurs de la
   583→            matrice.
   584→          </p>
   585→        </div>
   586→      );
   587→    }
   588→    return null;
   589→  };
   590→
   591→  /* ── Render ── */
   592→  return (
   593→    <div className="space-y-6">
   594→      {/* Header */}
   595→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   596→        <div>
   597→          <h2 className="text-2xl font-bold">Mes Garanties</h2>
   598→          <p className="text-muted-foreground mt-1">
   599→            Gérez les garanties et couvertures proposées dans vos offres.
   600→          </p>
   601→        </div>
   602→        <Button
   603→          className="bg-[#B9E54D] text-black hover:bg-[#a5d044]"
   604→          onClick={openCreate}
   605→        >
   606→          <Plus className="mr-2 h-4 w-4" />
   607→          Ajouter une garantie
   608→        </Button>
   609→      </div>
   610→
   611→      {/* Guarantees table */}
   612→      <div className="rounded-xl border bg-card">
   613→        <div className="p-6 pb-4">
   614→          <h3 className="text-lg font-semibold">Liste des garanties</h3>
   615→          <p className="text-sm text-muted-foreground">
   616→            Toutes vos garanties configurées
   617→          </p>
   618→        </div>
   619→        <div className="max-h-96 overflow-y-auto">
   620→          {loading ? (
   621→            <div className="p-8 space-y-3">
   622→              {Array.from({ length: 5 }).map((_, i) => (
   623→                <div key={i} className="flex items-center gap-4">
   624→                  <div className="h-4 w-20 bg-muted rounded animate-pulse" />
   625→                  <div className="h-4 w-32 bg-muted rounded animate-pulse" />
   626→                  <div className="h-4 w-24 bg-muted rounded animate-pulse" />
   627→                  <div className="flex-1" />
   628→                </div>
   629→              ))}
   630→            </div>
   631→          ) : error ? (

   902→          <DialogFooter className="gap-2">

   924→                className="bg-[#B9E54D] text-black hover:bg-[#a5d044]"
