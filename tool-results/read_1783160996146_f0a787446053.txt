   108→/* ── Component ─────────────────────────────────────────────── */
   109→export function CoveragesTab() {
   110→  const { toast } = useToast();
   111→  const [items, setItems] = useState<Coverage[]>([]);
   112→  const [insurers, setInsurers] = useState<InsurerOption[]>([]);
   113→  const [categories, setCategories] = useState<CatOption[]>([]);
   114→  const [insuranceCategories, setInsuranceCategories] = useState<InsCatOption[]>([]);
   115→  const [search, setSearch] = useState("");
   116→  const [filterInsurer, setFilterInsurer] = useState("");
   117→  const [filterCategory, setFilterCategory] = useState("");
   118→  const [filterCalcType, setFilterCalcType] = useState("");
   119→  const [loading, setLoading] = useState(true);
   120→  const [dialogOpen, setDialogOpen] = useState(false);
   121→  const [editing, setEditing] = useState<Coverage | null>(null);
   122→  const [step, setStep] = useState(1);
   123→  const [step1, setStep1] = useState<Step1Data>(emptyStep1);
   124→  const [calcType, setCalcType] = useState<CalculationType | "">("");
   125→  const [metadata, setMetadata] = useState<Record<string, unknown>>({});
   126→  const [saving, setSaving] = useState(false);
   127→  const [deleteId, setDeleteId] = useState<string | null>(null);
   128→
   129→  // Tariff rules panel
   130→  const [selectedId, setSelectedId] = useState<string | null>(null);
   131→  const [tariffRules, setTariffRules] = useState<TariffRule[]>([]);
   132→  const [trLoading, setTrLoading] = useState(false);
   133→  const [trDialogOpen, setTrDialogOpen] = useState(false);
   134→  const [trEditing, setTrEditing] = useState<TariffRule | null>(null);
   135→  const [trDeleteId, setTrDeleteId] = useState<string | null>(null);
   136→  const [trForm, setTrForm] = useState({
   137→    fuelType: "", minFiscalPower: "", maxFiscalPower: "", baseRate: "", fixedAmount: "", minAmount: "", maxAmount: "", formulaName: "",
   138→  });
   139→  const [trSaving, setTrSaving] = useState(false);
   140→
   141→  const fetchItems = useCallback(async () => {
   142→    try {
   143→      const params = new URLSearchParams();
   144→      if (search) params.set("search", search);
   145→      if (filterInsurer) params.set("insurerId", filterInsurer);
   146→      if (filterCategory) params.set("categoryId", filterCategory);
   147→      if (filterCalcType) params.set("calculationType", filterCalcType);
   148→      const res = await fetch(`/api/admin/coverages?${params}`);
   149→      if (!res.ok) throw new Error();
   150→      setItems(await res.json());
   151→    } catch {
   152→      toast({ title: "Erreur", description: "Impossible de charger les garanties", variant: "destructive" });
   153→    } finally {
   154→      setLoading(false);
   155→    }
   156→  }, [search, filterInsurer, filterCategory, filterCalcType, toast]);
   157→
   158→  const fetchLookups = useCallback(async () => {
   159→    try {
   160→      const [insRes, catRes, insCatRes] = await Promise.all([
   161→        fetch("/api/admin/insurers"),
   162→        fetch("/api/admin/coverage-categories"),
   163→        fetch("/api/admin/insurance-categories"),
   164→      ]);
   165→      if (insRes.ok) setInsurers(await insRes.json());
   166→      if (catRes.ok) setCategories(await catRes.json());
   167→      if (insCatRes.ok) setInsuranceCategories(await insCatRes.json());
   168→    } catch { /* silent */ }
   169→  }, []);
   170→
   171→  useEffect(() => { fetchLookups(); }, [fetchLookups]);
   172→  useEffect(() => { setLoading(true); fetchItems(); }, [fetchItems]);
   173→
   174→  const fetchTariffRules = useCallback(async (coverageId: string) => {
   175→    setTrLoading(true);
   176→    try {
   177→      const res = await fetch(`/api/admin/coverages/${coverageId}/tariff-rules`);
   178→      if (!res.ok) throw new Error();
   179→      setTariffRules(await res.json());
   180→    } catch {
   181→      toast({ title: "Erreur", description: "Impossible de charger les règles", variant: "destructive" });
   182→    } finally {
   183→      setTrLoading(false);
   184→    }
   185→  }, [toast]);
   186→
   187→  const handleRowClick = (id: string) => {
   188→    if (selectedId === id) {
   189→      setSelectedId(null);
   190→      return;
   191→    }
   192→    setSelectedId(id);
   193→    fetchTariffRules(id);
   194→  };
   195→
   196→  /* ── Wizard ─────────────────────────────────────────────── */
   197→  const openCreate = () => {
   198→    setEditing(null);
   199→    setStep1(emptyStep1);
   200→    setCalcType("");
   201→    setMetadata({});
   202→    setStep(1);
   203→    setDialogOpen(true);
   204→  };
   205→
   206→  const openEdit = (item: Coverage) => {
   207→    setEditing(item);
   208→    setStep1({
   209→      name: item.name,
   210→      insuranceCategoryId: "", // will be resolved after insuranceCategories load
   211→      description: item.description ?? "",
   212→      insurerId: item.insurerId,
   213→      categoryId: item.categoryId ?? "",
   214→      isMandatory: item.isMandatory,
   215→      displayOrder: item.displayOrder.toString(),
   216→      isActive: item.isActive,
   217→    });
   218→    setCalcType(item.calculationType as CalculationType);
   219→    setMetadata(item.metadata);
   220→    setStep(1);
   221→    setDialogOpen(true);
   222→  };
   223→
   224→  const buildMetadataFromStep3 = (): Record<string, unknown> => {
   225→    if (calcType === "FREE") return {};
   226→    if (calcType === "FIXED_AMOUNT") {
   227→      return { fixedAmount: (metadata.fixedAmount as number) || 0 };
   228→    }
   229→    if (calcType === "VARIABLE_BASED") {
   230→      const m: Record<string, unknown> = { variable: (metadata.variable as string) || "VN" };
   231→      if (metadata.hasConditional) {
   232→        m.hasConditional = true;
   233→        m.threshold = Number(metadata.threshold) || 0;
   234→        m.rateBelow = Number(metadata.rateBelow) || 0;
   235→        m.rateAbove = Number(metadata.rateAbove) || 0;
   236→      } else {
   237→        m.rate = Number(metadata.rate) || 0;
   238→      }
   239→      if (metadata.franchiseEnabled) {
   240→        m.franchiseEnabled = true;
   241→        m.franchisePercent = Number(metadata.franchisePercent) || 0;
   242→        m.franchiseMin = Number(metadata.franchiseMin) || 0;
   243→      }
   244→      return m;
   245→    }
   246→    if (calcType === "MATRIX_BASED") {
   247→      return { matrixType: (metadata.matrixType as string) || "FISCAL_POWER" };
   248→    }
   249→    return {};
   250→  };
   251→
   252→  const handleSave = async () => {
   253→    if (!step1.name.trim() || !step1.insurerId || !calcType) {
   254→      toast({ title: "Erreur", description: "Veuillez remplir tous les champs requis", variant: "destructive" });
   255→      return;
   256→    }
   257→    setSaving(true);
   258→    try {
   259→      const url = editing ? `/api/admin/coverages/${editing.id}` : "/api/admin/coverages";
   260→      const method = editing ? "PUT" : "POST";
   261→      // Resolve insurance category name for the `type` field
   262→      const insCat = insuranceCategories.find((c) => c.id === step1.insuranceCategoryId);
   263→      const body = {
   264→        name: step1.name,
   265→        type: insCat?.name || "",
   266→        description: step1.description,
   267→        insurerId: step1.insurerId,
   268→        categoryId: step1.categoryId || null,
   269→        isMandatory: step1.isMandatory,
   270→        displayOrder: parseInt(step1.displayOrder, 10) || 0,
   271→        isActive: step1.isActive,
   272→        calculationType: calcType,
   273→        metadata: buildMetadataFromStep3(),
   274→      };
   275→      const res = await fetch(url, { method, headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
   276→      if (!res.ok) {
   277→        const data = await res.json();
   278→        throw new Error(data.error || "Erreur");
   279→      }
   280→      toast({ title: editing ? "Garantie modifiée" : "Garantie créée" });
   281→      setDialogOpen(false);
   282→      fetchItems();
   283→    } catch (e) {
   284→      toast({ title: "Erreur", description: e instanceof Error ? e.message : "Erreur", variant: "destructive" });
   285→    } finally {
   286→      setSaving(false);
   287→    }
   288→  };
   289→
   290→  const handleDelete = async () => {
   291→    if (!deleteId) return;
   292→    try {
   293→      const res = await fetch(`/api/admin/coverages/${deleteId}`, { method: "DELETE" });
   294→      if (!res.ok) throw new Error();
   295→      toast({ title: "Garantie supprimée" });
   296→      setDeleteId(null);
   297→      if (selectedId === deleteId) setSelectedId(null);
   298→      fetchItems();
   299→    } catch {
   300→      toast({ title: "Erreur", variant: "destructive" });
   301→    }
   302→  };
   303→
   304→  /* ── Tariff Rule CRUD ────────────────────────────────────── */
   305→  const openTrCreate = () => {
   306→    setTrEditing(null);
   307→    setTrForm({ fuelType: "", minFiscalPower: "", maxFiscalPower: "", baseRate: "", fixedAmount: "", minAmount: "", maxAmount: "", formulaName: "" });
   308→    setTrDialogOpen(true);
   309→  };
   310→
   311→  const openTrEdit = (rule: TariffRule) => {
   312→    setTrEditing(rule);
   313→    setTrForm({
   314→      fuelType: rule.fuelType ?? "", minFiscalPower: rule.minFiscalPower?.toString() ?? "", maxFiscalPower: rule.maxFiscalPower?.toString() ?? "",
   315→      baseRate: rule.baseRate?.toString() ?? "", fixedAmount: rule.fixedAmount?.toString() ?? "", minAmount: rule.minAmount?.toString() ?? "",
   316→      maxAmount: rule.maxAmount?.toString() ?? "", formulaName: rule.formulaName ?? "",
   317→    });
   318→    setTrDialogOpen(true);
   319→  };
   320→
   321→  const saveTr = async () => {
   322→    if (!selectedId) return;
   323→    setTrSaving(true);
   324→    try {
   325→      const body = {
   326→        fuelType: trForm.fuelType || null,
   327→        minFiscalPower: trForm.minFiscalPower ? parseInt(trForm.minFiscalPower, 10) : null,
   328→        maxFiscalPower: trForm.maxFiscalPower ? parseInt(trForm.maxFiscalPower, 10) : null,
   329→        baseRate: trForm.baseRate ? parseFloat(trForm.baseRate) : null,
   330→        fixedAmount: trForm.fixedAmount ? parseFloat(trForm.fixedAmount) : null,
   331→        minAmount: trForm.minAmount ? parseFloat(trForm.minAmount) : null,
   332→        maxAmount: trForm.maxAmount ? parseFloat(trForm.maxAmount) : null,
   333→        formulaName: trForm.formulaName || null,
   334→      };
   335→      const res = await fetch(`/api/admin/coverages/${selectedId}/tariff-rules`, {
   336→        method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body),
   337→      });
   338→      if (!res.ok) throw new Error();
   339→      toast({ title: "Règle ajoutée" });
   340→      setTrDialogOpen(false);
   341→      fetchTariffRules(selectedId);
   342→    } catch {
   343→      toast({ title: "Erreur", variant: "destructive" });
   344→    } finally {
   345→      setTrSaving(false);
   346→    }
   347→  };
   348→
   349→  const handleTrDelete = async () => {
   350→    if (!trDeleteId || !selectedId) return;
   351→    try {
   352→      const res = await fetch(`/api/admin/tariff-rules/${trDeleteId}`, { method: "DELETE" });
   353→      if (!res.ok) throw new Error();
   354→      toast({ title: "Règle tarifaire supprimée" });
   355→      setTrDeleteId(null);
   356→      fetchTariffRules(selectedId);
   357→    } catch {
   358→      toast({ title: "Erreur", variant: "destructive" });
   359→    }
   360→  };
   361→
   362→  /* ── Render Step 3 ──────────────────────────────────────── */
   363→  const renderStep3 = () => {
   364→    if (calcType === "FREE") {
   365→      return (
   366→        <div className="rounded-lg bg-emerald-50 border border-emerald-200 p-6 text-center">
   367→          <p className="text-emerald-700 font-medium text-lg">🟢 Gratuit</p>
   368→          <p className="text-emerald-600 text-sm mt-1">Aucune configuration supplémentaire requise.</p>
   369→        </div>
   370→      );
   371→    }
   372→    if (calcType === "FIXED_AMOUNT") {
   373→      return (
   374→        <div className="space-y-4">
   375→          <Label>Montant fixe (FCFA) *</Label>
   376→          <Input className="w-full" type="number" value={(metadata.fixedAmount as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, fixedAmount: parseFloat(e.target.value) || 0 })} placeholder="Ex: 50000" />
   377→        </div>
   378→      );
   379→    }
   380→    if (calcType === "VARIABLE_BASED") {
   381→      return (
   382→        <div className="space-y-4">
   383→          <div>
   384→            <Label>Variable</Label>
   385→            <Select value={(metadata.variable as string) || "VN"} onValueChange={(v) => setMetadata({ ...metadata, variable: v })}>
   386→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   387→              <SelectContent>
   388→                <SelectItem value="VN">Valeur Neuve (VN)</SelectItem>
   389→                <SelectItem value="VN_REPLACEMENT_VALUE">VN / Valeur de remplacement</SelectItem>
   390→              </SelectContent>
   391→            </Select>
   392→          </div>
   393→          <div className="flex items-center gap-2">
   394→            <Checkbox id="conditional" checked={!!metadata.hasConditional} onCheckedChange={(v) => setMetadata({ ...metadata, hasConditional: !!v })} />
   395→            <Label htmlFor="conditional">Taux conditionné par seuil</Label>
   396→          </div>
   397→          {metadata.hasConditional ? (
   398→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 rounded-lg border p-4 bg-muted/50">
   399→              <div><Label className="text-xs">Seuil (FCFA)</Label><Input type="number" value={(metadata.threshold as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, threshold: parseFloat(e.target.value) || 0 })} /></div>
   400→              <div><Label className="text-xs">Taux sous (%)</Label><Input type="number" step="0.01" value={(metadata.rateBelow as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rateBelow: parseFloat(e.target.value) || 0 })} /></div>
   401→              <div><Label className="text-xs">Taux au-dessus (%)</Label><Input type="number" step="0.01" value={(metadata.rateAbove as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rateAbove: parseFloat(e.target.value) || 0 })} /></div>
   402→            </div>
   403→          ) : (
   404→            <div className="w-full"><Label>Taux (%)</Label><Input className="w-full" type="number" step="0.01" value={(metadata.rate as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, rate: parseFloat(e.target.value) || 0 })} /></div>
   405→          )}
   406→          <Separator />
   407→          <div className="flex items-center gap-2">
   408→            <Checkbox id="franchise" checked={!!metadata.franchiseEnabled} onCheckedChange={(v) => setMetadata({ ...metadata, franchiseEnabled: !!v })} />
   409→            <Label htmlFor="franchise">Appliquer une franchise</Label>
   410→          </div>
   411→          {metadata.franchiseEnabled && (
   412→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 rounded-lg border p-4 bg-muted/50">
   413→              <div><Label className="text-xs">Franchise (%)</Label><Input type="number" step="0.01" value={(metadata.franchisePercent as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, franchisePercent: parseFloat(e.target.value) || 0 })} /></div>
   414→              <div><Label className="text-xs">Montant min (FCFA)</Label><Input type="number" value={(metadata.franchiseMin as number) ?? ""} onChange={(e) => setMetadata({ ...metadata, franchiseMin: parseFloat(e.target.value) || 0 })} /></div>
   415→            </div>
   416→          )}
   417→        </div>
   418→      );
   419→    }
   420→    if (calcType === "MATRIX_BASED") {
   421→      return (
   422→        <div className="space-y-4">
   423→          <div>
   424→            <Label>Type de matrice</Label>
   425→            <Select value={(metadata.matrixType as string) || "FISCAL_POWER"} onValueChange={(v) => setMetadata({ ...metadata, matrixType: v })}>
   426→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   427→              <SelectContent>
   428→                <SelectItem value="FISCAL_POWER">Puissance fiscale</SelectItem>
   429→                <SelectItem value="FORMULA">Formule (IC/IPT)</SelectItem>
   430→                <SelectItem value="TIERCE_COMPLETE">Tierce complète</SelectItem>
   431→                <SelectItem value="TIERCE_COLLISION">Tierce collision</SelectItem>
   432→                <SelectItem value="FUEL_TYPE">Type carburant</SelectItem>
   433→                <SelectItem value="SEATS">Nombre de places</SelectItem>
   434→              </SelectContent>
   435→            </Select>
   436→          </div>
   437→          <p className="text-sm text-muted-foreground">
   438→            Les règles tarifaires associées définissent les valeurs de la matrice. Utilisez la section ci-dessous pour les gérer.
   439→          </p>
   440→        </div>
   441→      );
   442→    }
   443→    return null;
   444→  };
   445→
   446→  const selectedCoverage = items.find((c) => c.id === selectedId);
   447→
   448→  return (
   449→    <div className="space-y-4">
   450→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   451→        <h1 className="text-2xl font-bold">Garanties</h1>
   452→        <div className="flex items-center gap-2 flex-wrap">
   453→          <div className="relative">
   454→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   455→            <Input placeholder="Rechercher..." value={search} onChange={(e) => { setSearch(e.target.value); setLoading(true); }} className="pl-9 w-44" />
   456→          </div>
   457→          <Select value={filterInsurer} onValueChange={(v) => { setFilterInsurer(v === "__all__" ? "" : v); setLoading(true); }}>
   458→            <SelectTrigger className="w-36"><SelectValue placeholder="Assureur" /></SelectTrigger>
   459→            <SelectContent>
   460→              <SelectItem value="__all__">Tous</SelectItem>
   461→              {insurers.map((i) => <SelectItem key={i.id} value={i.id}>{i.name}</SelectItem>)}
