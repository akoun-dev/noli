   577→      comparisonResults.map((o) => o.insurerName)
   578→    );
   579→    setCheckedInsurers(allNames);
   580→  }, [comparisonResults]);
   581→
   582→  const toggleInsurer = useCallback((name: string) => {
   583→    setCheckedInsurers((prev) => {
   584→      const next = new Set(prev);
   585→      if (next.has(name)) {
   586→        next.delete(name);
   587→      } else {
   588→        next.add(name);
   589→      }
   590→      return next;
   591→    });
   592→  }, []);
   593→
   594→  const filteredAndSorted = useMemo(() => {
   595→    let results = [...comparisonResults];
   596→
   597→    // Coverage filter
   598→    if (coverageFilter !== "all") {
   599→      results = results.filter((o) => o.coverageType === coverageFilter);
   600→    }
   601→
   602→    // Insurer filter (multi-select from checkboxes)
   603→    if (checkedInsurers.size > 0) {
   604→      results = results.filter((o) => checkedInsurers.has(o.insurerName));
   605→    }
   606→
   607→    // Budget filter (monthly price)
   608→    results = results.filter((o) => o.monthlyPrice <= budgetMax);
   609→
   610→    // Sort
   611→    switch (sortBy) {
   612→      case "price_asc":
   613→        results.sort((a, b) => a.monthlyPrice - b.monthlyPrice);
   614→        break;
   615→      case "price_desc":
   616→        results.sort((a, b) => b.monthlyPrice - a.monthlyPrice);
   617→        break;
   618→      case "rating_desc":
   619→        results.sort((a, b) => b.insurerRating - a.insurerRating);
   620→        break;
   621→      case "name_asc":
   622→        results.sort((a, b) => a.name.localeCompare(b.name, "fr"));
   623→        break;
   624→    }
   625→
   626→    return results;
   627→  }, [comparisonResults, sortBy, coverageFilter, checkedInsurers, budgetMax]);
   628→
   629→  /* ── handlers ── */
   630→  const handleRequestQuote = (offer: InsurerOffer) => {
   631→    toast({
   632→      title: "Devis enregistré",
   633→      description: `Votre demande de devis pour ${offer.name} a été enregistrée.`,
   634→    });
   635→
   636→    const quote: import("@/types").QuoteRecord = {
   637→      id: crypto.randomUUID(),
   638→      reference: `DEV-${Date.now().toString(36).toUpperCase()}`,
   639→      status: "pending",
   640→      personalInfo: { ...personalInfo },
   641→      vehicleInfo: { ...vehicleInfo },
   642→      coverageNeeds: { ...coverageNeeds },
   643→      proposedPrice: offer.annualPrice,
   644→      finalPrice: null,
   645→      insurerName: offer.insurerName,
   646→      offerName: offer.name,
   647→      createdAt: new Date().toISOString(),
   648→    };
   649→
   650→    setUserQuotes([quote, ...userQuotes]);
   651→  };
   652→
   653→  const handleViewDetails = (offer: InsurerOffer) => {
   654→    setSelectedOffer(offer);
   655→  };
   656→
   657→  const resetFilters = () => {
   658→    setCoverageFilter("all");
   659→    setCheckedInsurers(new Set(uniqueInsurers));
   660→    setBudgetMax(300000);
   661→    setSortBy("price_asc");
   662→    setSelectedInsurerFilter("all");
   663→  };
   664→
   665→  const activeFilterCount = [
   666→    coverageFilter !== "all",
   667→    checkedInsurers.size < uniqueInsurers.length,
   668→    budgetMax < 300000,
   669→  ].filter(Boolean).length;
   670→
   671→  /* ── empty state (no comparison results at all) ── */
   672→  if (comparisonResults.length === 0) {
   673→    return <EmptyResultsState onGoBack={() => setView("compare")} />;
   674→  }
   675→
   676→  /* ── main render ── */
   677→  return (
   678→    <section className="max-w-7xl mx-auto px-4 sm:px-6 py-6 sm:py-10">
   679→      {/* ── Top Bar ── */}
   680→      <div className="flex flex-col sm:flex-row sm:items-center gap-4 mb-6">
   681→        {/* Back button */}
   682→        <Button
   683→          variant="ghost"
   684→          size="sm"
   685→          className="-ml-2 text-muted-foreground hover:text-foreground shrink-0"
   686→          onClick={() => setView("compare")}
   687→        >
   688→          <ArrowLeft className="size-4 mr-1.5" />
   689→          <span className="hidden sm:inline">Retour au formulaire</span>
   690→          <span className="sm:hidden">Retour</span>
   691→        </Button>
   692→
   693→        {/* Title */}
   694→        <h1 className="text-xl sm:text-2xl font-bold text-foreground flex-1">
   695→          {filteredAndSorted.length} offre
   696→          {filteredAndSorted.length !== 1 ? "s" : ""} trouvée
   697→          {filteredAndSorted.length !== 1 ? "s" : ""}
   698→        </h1>
   699→
   700→        {/* Price mode toggle */}
   701→        <div className="flex items-center bg-muted/30 rounded-full p-0.5 shrink-0 self-start sm:self-auto">
   702→          <button
   703→            onClick={() => setPriceMode("annual")}
   704→            className={`px-4 py-1.5 rounded-full text-sm font-medium transition-all duration-200 ${
   705→              priceMode === "annual"
   706→                ? "bg-primary text-primary-foreground shadow-sm"
   707→                : "text-muted-foreground hover:text-foreground"
   708→            }`}
   709→          >
   710→            Par an
   711→          </button>
   712→          <button
   713→            onClick={() => setPriceMode("monthly")}
   714→            className={`px-4 py-1.5 rounded-full text-sm font-medium transition-all duration-200 ${
   715→              priceMode === "monthly"
   716→                ? "bg-primary text-primary-foreground shadow-sm"
   717→                : "text-muted-foreground hover:text-foreground"
   718→            }`}
   719→          >
   720→            Par mois
   721→          </button>
   722→        </div>
   723→      </div>
   724→
   725→      {/* ── Mobile Filter Toggle ── */}
   726→      <div className="lg:hidden mb-4">
   727→        <Button
   728→          variant="outline"
   729→          className="w-full justify-between rounded-xl border-border"
   730→          onClick={() => setMobileFiltersOpen(!mobileFiltersOpen)}
   731→        >
   732→          <div className="flex items-center gap-2">
   733→            <Filter className="size-4" />
   734→            <span className="font-medium">Filtres</span>
   735→            {activeFilterCount > 0 && (
   736→              <Badge className="bg-primary text-primary-foreground rounded-full px-1.5 text-xs min-w-5 h-5 flex items-center justify-center">
   737→                {activeFilterCount}
   738→              </Badge>
   739→            )}
   740→          </div>
   741→          <ChevronDown
   742→            className={`size-4 text-muted-foreground transition-transform duration-200 ${
   743→              mobileFiltersOpen ? "rotate-180" : ""
   744→            }`}
   745→          />
   746→        </Button>
   747→      </div>
   748→
   749→      {/* ── Mobile Filters (collapsible) ── */}
   750→      <AnimatePresence>
   751→        {mobileFiltersOpen && (
   752→          <motion.div
   753→            initial={{ height: 0, opacity: 0 }}
   754→            animate={{ height: "auto", opacity: 1 }}
   755→            exit={{ height: 0, opacity: 0 }}
   756→            transition={{ duration: 0.25 }}
   757→            className="overflow-hidden lg:hidden mb-6"
   758→          >
   759→            <FiltersSidebar
   760→              coverageFilter={coverageFilter}
   761→              setCoverageFilter={setCoverageFilter}
   762→              checkedInsurers={checkedInsurers}
   763→              toggleInsurer={toggleInsurer}
   764→              uniqueInsurers={uniqueInsurers}
   765→              budgetMax={budgetMax}
   766→              setBudgetMax={setBudgetMax}
   767→              onReset={resetFilters}
   768→              totalOffers={comparisonResults.length}
   769→            />
   770→          </motion.div>
   771→        )}
   772→      </AnimatePresence>
   773→
   774→      {/* ── Two-column layout ── */}
   775→      <div className="flex gap-6 items-start">
   776→        {/* Desktop Sidebar */}
   777→        <div className="hidden lg:block w-72 shrink-0">
   778→          <FiltersSidebar
   779→            coverageFilter={coverageFilter}
   780→            setCoverageFilter={setCoverageFilter}
   781→            checkedInsurers={checkedInsurers}
   782→            toggleInsurer={toggleInsurer}
   783→            uniqueInsurers={uniqueInsurers}
   784→            budgetMax={budgetMax}
   785→            setBudgetMax={setBudgetMax}
   786→            onReset={resetFilters}
   787→            totalOffers={comparisonResults.length}
   788→          />
   789→        </div>
   790→
   791→        {/* Main Content */}
   792→        <main className="flex-1 min-w-0 space-y-4">
   793→          {filteredAndSorted.length > 0 ? (
   794→            <AnimatePresence mode="popLayout">
   795→              {filteredAndSorted.map((offer) => (
   796→                <OfferCard
   797→                  key={offer.id}
   798→                  offer={offer}
   799→                  priceMode={priceMode}
   800→                  onViewDetails={handleViewDetails}
   801→                  onRequestQuote={handleRequestQuote}
   802→                />
   803→              ))}
   804→            </AnimatePresence>
   805→          ) : (
   806→            /* Filtered empty state */
   807→            <div className="text-center py-16">
   808→              <div className="rounded-full bg-muted/40 p-6 mb-4 inline-block">
   809→                <SearchX className="size-10 text-muted-foreground" />
   810→              </div>
   811→              <h3 className="text-lg font-semibold mb-2">
   812→                Aucune offre ne correspond à vos critères
   813→              </h3>
   814→              <p className="text-muted-foreground mb-6 text-sm">
   815→                Essayez d'élargir vos filtres ou de modifier le budget.
   816→              </p>
   817→              <Button
   818→                variant="outline"
   819→                className="rounded-full"
   820→                onClick={resetFilters}
   821→              >
   822→                <RotateCcw className="size-4 mr-2" />
   823→                Réinitialiser les filtres
   824→              </Button>
   825→            </div>
   826→          )}
   827→        </main>
   828→      </div>
   829→
   830→      {/* Detail dialog */}
   831→      <OfferDetailDialog
   832→        offer={selectedOffer}
   833→        open={selectedOffer !== null}
   834→        onClose={() => setSelectedOffer(null)}
   835→        onRequestQuote={handleRequestQuote}
   836→      />
   837→    </section>
   838→  );
   839→}
