   163→/** Arrondit au multiple de 500 le plus proche */
   164→function roundTo500(value: number): number {
   165→  return Math.round(value / 500) * 500;
   166→}

   168→/** Parse JSON string safely, returns empty object on failure */
   169→function safeParseJSON<T>(raw: string): T {
   170→  try {
   171→    return JSON.parse(raw) as T;
   172→  } catch {
   173→    return {} as T;
   174→  }
   175→}
   176→
   177→/** Parse a JSON string field that stores a string array */
   178→function safeParseStringArray(raw: string | undefined | null): string[] {
   179→  if (!raw) return [];
   180→  try {
   181→    const parsed = JSON.parse(raw);
   182→    return Array.isArray(parsed) ? parsed : [];
   183→  } catch {
   184→    return [];
   185→  }
   186→}
   187→
   188→/** Clamp a value between min and max (bounds are inclusive) */
   189→function clamp(value: number, min?: number | null, max?: number | null): number {
   190→  let result = value;
   191→  if (min != null && !isNaN(min)) result = Math.max(result, min);
   192→  if (max != null && !isNaN(max)) result = Math.min(result, max);
   193→  return result;
   194→}

   196→/** Détermine la classe de valeur à neuf (VN) pour TCM / TCL */
   197→function getVNRangeClass(newValue: number): string {
   198→  if (newValue <= 12_000_000) return "A";
   199→  if (newValue <= 25_000_000) return "B";
   200→  if (newValue <= 40_000_000) return "C";
   201→  if (newValue <= 90_000_000) return "D";
   202→  if (newValue <= 110_000_000) return "E";
   203→  return "F";
   204→}

   206→/** Parse a numeric string (strips spaces, handles commas) */
   207→function parseFCFA(raw: string | undefined | null): number {
   208→  if (!raw) return 0;
   209→  return parseFloat(String(raw).replace(/[\s,]/g, "")) || 0;
   210→}

   222→// ── 1. calculateGuaranteePremium ─────────────────────────────
   223→
   224→export function calculateGuaranteePremium(
   225→  coverage: CoverageInput,
   226→  vehicleData: VehiclePricingData,
   227→  options?: {
   228→    isInPackage?: boolean;
   229→    activeGuarantees?: string[];
   230→  }
   231→): PricingResult {
   232→  const meta = safeParseJSON<GuaranteeMetadata>(coverage.metadata);
   233→
   234→  switch (coverage.calculationType) {
   235→    case "FREE":
   236→      return calculateFree(meta);
   237→    case "FIXED_AMOUNT":
   238→      return calculateFixedAmount(meta as FixedAmountMetadata, options);
   239→    case "VARIABLE_BASED":
   240→      return calculateVariableBased(meta as VariableBasedMetadata, vehicleData);
   241→    case "MATRIX_BASED":
   242→      return calculateMatrixBased(
   243→        meta as MatrixBasedMetadata,
   244→        vehicleData,
   245→        coverage.tariffRules
   246→      );
   247→    default:
   248→      return {
   249→        amount: 0,
   250→        method: coverage.calculationType || "INCONNU",
   251→        breakdown: `Méthode de calcul non reconnue : ${coverage.calculationType}`,
   252→      };
   253→  }
   254→}

   266→// ── FIXED_AMOUNT ──────────────────────────────────────────────
   267→
   268→function calculateFixedAmount(
   269→  meta: FixedAmountMetadata,
   270→  options?: { isInPackage?: boolean; activeGuarantees?: string[] }
   271→): PricingResult {
   272→  // Vérifie si une garantie requise est active
   273→  if (meta.requiresGuarantee) {
   274→    const hasRequired =
   275→      options?.activeGuarantees?.includes(meta.requiresGuarantee) ?? false;
   276→    if (!hasRequired) {
   277→      return {
   278→        amount: 0,
   279→        method: "FIXED_AMOUNT",
   280→        breakdown: `Garantie non applicable — garantie requise « ${meta.requiresGuarantee} » non sélectionnée`,
   281→      };
   282→    }
   283→  }
   284→
   285→  // Détermine le montant de base
   286→  let amount: number;
   287→  if (options?.isInPackage && meta.packPriceReduced != null) {
   288→    amount = meta.packPriceReduced;
   289→  } else {
   290→    amount = meta.fixedAmount ?? 0;
   291→  }
   292→
   293→  // Applique les bornes min / max
   294→  amount = clamp(amount, meta.minAmount, meta.maxAmount);
   295→
   296→  const details: string[] = [];
   297→  if (options?.isInPackage && meta.packPriceReduced != null) {
   298→    details.push(
   299→      `Montant pack réduit : ${meta.packPriceReduced.toLocaleString("fr-FR")} FCFA`
   300→    );
   301→  } else {
   302→    details.push(
   303→      `Montant fixe : ${meta.fixedAmount?.toLocaleString("fr-FR") ?? "0"} FCFA`
   304→    );
   305→  }
   306→  if (meta.capital != null) {
   307→    details.push(`Capital : ${meta.capital.toLocaleString("fr-FR")} FCFA`);
   308→  }
   309→  if (meta.minAmount != null) {
   310→    details.push(`Minimum : ${meta.minAmount.toLocaleString("fr-FR")} FCFA`);
   311→  }
   312→  if (meta.maxAmount != null) {
   313→    details.push(`Plafond : ${meta.maxAmount.toLocaleString("fr-FR")} FCFA`);
   314→  }
   315→
   316→  return {
   317→    amount,
   318→    method: "FIXED_AMOUNT",
   319→    breakdown: details.join(" — "),
   320→  };
   321→}

   323→// ── VARIABLE_BASED ───────────────────────────────────────────
   324→
   325→function calculateVariableBased(
   326→  meta: VariableBasedMetadata,
   327→  vehicle: VehiclePricingData
   328→): PricingResult {
   329→  // Vérifie la garantie requise
   330→  if (meta.requiresGuarantee) {
   331→    // Si aucune liste de garanties actives fournie, on suppose applicable
   332→    // (le contrôle de garantie requise nécessiterait le contexte complet)
   333→  }
   334→
   335→  // Récupère la valeur de la variable de référence
   336→  let variableValue = 0;
   337→  let variableLabel = "";
   338→
   339→  switch (meta.variableSource) {
   340→    case "NEW_VALUE":
   341→      variableValue = parseFCFA(vehicle.newValue);
   342→      variableLabel = `Valeur à neuf (VN)`;
   343→      break;
   344→    case "VENAL_VALUE":
   345→      variableValue = parseFCFA(vehicle.currentValue);
   346→      variableLabel = `Valeur vénale (VA)`;
   347→      break;
   348→    case "FISCAL_POWER":
   349→      variableValue = parseFloat(vehicle.fiscalPower) || 0;
   350→      variableLabel = `Puissance fiscale (CV)`;
   351→      break;
   352→    default:
   353→      return {
   354→        amount: 0,
   355→        method: "VARIABLE_BASED",
   356→        breakdown: `Source de variable non reconnue : ${meta.variableSource}`,
   357→      };
   358→  }
   359→
   360→  // Détermine le taux applicable
   361→  let ratePercent = meta.ratePercent ?? 0;
   362→  let rateReason = `Taux : ${ratePercent}%`;
   363→
   364→  if (meta.conditionedByNewValue) {
   365→    const threshold = meta.newValueThreshold ?? 0;
   366→    const nv = parseFCFA(vehicle.newValue);
   367→
   368→    if (nv > 0 && nv <= threshold) {
   369→      ratePercent = meta.rateBelowThresholdPercent ?? meta.ratePercent ?? 0;
   370→      rateReason = `VN ≤ ${threshold.toLocaleString("fr-FR")} FCFA → taux réduit ${ratePercent}%`;
   371→    } else {
   372→      ratePercent = meta.rateAboveThresholdPercent ?? meta.ratePercent ?? 0;
   373→      rateReason = `VN > ${threshold.toLocaleString("fr-FR")} FCFA → taux majoré ${ratePercent}%`;
   374→    }
   375→  }
   376→
   377→  // Calcul : valeur × (taux / 100), arrondi au 500 le plus proche
   378→  let amount = variableValue * (ratePercent / 100);
   379→  amount = roundTo500(amount);
   380→
   381→  // Applique les bornes min / max
   382→  amount = clamp(amount, meta.minAmount, meta.maxAmount);
   383→
   384→  const details: string[] = [
   385→    `${variableLabel} : ${variableValue.toLocaleString("fr-FR")} FCFA`,
   386→    rateReason,
   387→    `Brut : ${(variableValue * (ratePercent / 100)).toLocaleString("fr-FR")} FCFA`,
   388→    `Arrondi (×500) : ${amount.toLocaleString("fr-FR")} FCFA`,
   389→  ];
   390→  if (meta.minAmount != null) {
   391→    details.push(`Minimum : ${meta.minAmount.toLocaleString("fr-FR")} FCFA`);
   392→  }
   393→  if (meta.maxAmount != null) {
   394→    details.push(`Plafond : ${meta.maxAmount.toLocaleString("fr-FR")} FCFA`);
   395→  }
   396→
   397→  return {
   398→    amount,
   399→    method: "VARIABLE_BASED",
   400→    breakdown: details.join(" — "),
   401→  };
   402→}

   404→// ── MATRIX_BASED ─────────────────────────────────────────────
   405→
   406→function calculateMatrixBased(
   407→  meta: MatrixBasedMetadata,
   408→  vehicle: VehiclePricingData,
   409→  tariffRules?: CoverageInput["tariffRules"]
   410→): PricingResult {
   411→  const fp = parseInt(vehicle.fiscalPower) || 0;
   412→  const fuelType = (vehicle.fuelType || "").toLowerCase().trim();
   413→  const seatsRaw = (vehicle.seats || "").trim();
   414→  const seatsNum = seatsRaw === "+8" ? 9 : parseInt(seatsRaw) || 5;
   415→  const nv = parseFCFA(vehicle.newValue);
   416→
   417→  let amount = 0;
   418→  let breakdown = "";
   419→  let source = "matrice";
   420→
   421→  switch (meta.dimension) {
   422→    // ── Puissance fiscale ──
   423→    case "FISCAL_POWER": {
   424→      const tariffs = meta.tariffs || [];
   425→      // Essai avec fuelType, puis fallback sans fuelType
   426→      let match = tariffs.find(
   427→        (t) =>
   428→          fp >= (t.fiscalPowerMin ?? -Infinity) &&
   429→          fp <= (t.fiscalPowerMax ?? Infinity) &&
   430→          (t.fuelType
   431→            ? t.fuelType.toLowerCase() === fuelType
   432→            : true)
   433→      );
   434→      if (!match) {
   435→        match = tariffs.find(
   436→          (t) =>
   437→            fp >= (t.fiscalPowerMin ?? -Infinity) &&
   438→            fp <= (t.fiscalPowerMax ?? Infinity)
   439→        );
   440→      }
   441→      if (match) {
   442→        amount = match.prime;
   443→        breakdown = `CV ${fp} dans [${match.fiscalPowerMin ?? "?"}–${match.fiscalPowerMax ?? "?"}]${match.fuelType ? `, carburant ${match.fuelType}` : ""} → prime ${amount.toLocaleString("fr-FR")} FCFA`;
   444→      } else {
   445→        // Fallback : chercher dans les règles de tarification DB
   446→        amount = findTariffRuleAmount(tariffRules, { fiscalPower: fp, fuelType }, meta);
   447→        breakdown = amount > 0
   448→          ? `Règle tarifaire DB → ${amount.toLocaleString("fr-FR")} FCFA`
   449→          : `Aucune tranche trouvée pour CV ${fp}`;
   450→      }
   451→      source = "PUISSANCE_FISCALE";
   452→      break;
   453→    }
   454→
   455→    // ── Type de carburant ──
   456→    case "FUEL_TYPE": {
   457→      const tariffs = meta.tariffs || [];
   458→      const match = tariffs.find((t) => {
   459→        const tFuel = (t.fuelType || "").toLowerCase().trim();
   460→        return tFuel === fuelType;
   461→      });
   462→      if (match) {
   463→        amount = match.prime;
   464→        breakdown = `Carburant « ${fuelType} » → prime ${amount.toLocaleString("fr-FR")} FCFA`;
   465→      } else {
   466→        breakdown = `Aucune tarification trouvée pour le carburant « ${fuelType} »`;
   467→      }
   468→      source = "TYPE_CARBURANT";
   469→      break;
   470→    }
   471→
   472→    // ── Catégorie véhicule ──
   473→    case "VEHICLE_CATEGORY": {
   474→      const tariffs = meta.tariffs || [];
   475→      // La catégorie est déduite de l'usage
   476→      const category = mapUsageToVehicleCategory(vehicle.usage);
   477→      const match = tariffs.find((t) => {
   478→        const tCat = (t.vehicleCategory || "").toUpperCase().trim();
   479→        return tCat === category;
   480→      });
   481→      if (match) {
   482→        amount = match.prime;
   483→        breakdown = `Catégorie véhicule « ${category} » → prime ${amount.toLocaleString("fr-FR")} FCFA`;
   484→      } else {
   485→        breakdown = `Aucune tarification trouvée pour la catégorie « ${category} »`;
   486→      }
   487→      source = "CATEGORIE_VEHICULE";
   488→      break;
   489→    }
   490→
   491→    // ── Places assises ──
   492→    case "SEATS": {
   493→      const tariffs = meta.tariffs || [];
   494→      const match = tariffs.find((t) => {
   495→        const min = t.fiscalPowerMin ?? -Infinity; // réutilise fiscalPowerMin/Max pour les sièges
   496→        const max = t.fiscalPowerMax ?? Infinity;
   497→        return seatsNum >= min && seatsNum <= max;
   498→      });
   499→      if (match) {
   500→        amount = match.prime;
   501→        breakdown = `${seatsRaw} places (≥${match.fiscalPowerMin ?? "?"} et ≤${match.fiscalPowerMax ?? "?"}) → prime ${amount.toLocaleString("fr-FR")} FCFA`;
   502→      } else {
   503→        breakdown = `Aucune tarification trouvée pour ${seatsRaw} places`;
   504→      }
   505→      source = "PLACES";
   506→      break;
   507→    }
   508→
   509→    // ── Formule (IC / IPT) ──
   510→    case "FORMULA": {
   511→      const formulas = meta.formulas || [];
   512→      const tariffs = meta.tariffs || [];
   513→
   514→      // 1) Cherche dans tariffs par formulaName correspondant aux sièges
   515→      const seatsLabel = seatsRaw === "+8" ? "+8" : seatsRaw;
   516→      const tariffMatch = tariffs.find((t) => {
   517→        // Cherche par formule correspondant aux places
   518→        return (
   519→          (t.fuelType === seatsLabel ||
   520→            t.vehicleCategory === seatsLabel) &&
   521→          t.prime
   522→        );
   523→      });
   524→
   525→      if (tariffMatch) {
   526→        amount = tariffMatch.prime;
   527→        breakdown = `Formule sièges « ${seatsLabel} » → prime ${amount.toLocaleString("fr-FR")} FCFA`;
   528→      } else {
   529→        // 2) Cherche dans formulas par nom correspondant aux sièges
   530→        const formulaMatch = formulas.find((f) =>
   531→          f.name.toLowerCase().includes(seatsLabel)
   532→        );
   533→        if (formulaMatch) {
   534→          // Calcule : baseRate % de la VN, plafonné au ceiling
   535→          const gross = nv * (formulaMatch.baseRate / 100);
   536→          amount = formulaMatch.ceiling
   537→            ? Math.min(gross, formulaMatch.ceiling)
   538→            : gross;
   539→          amount = roundTo500(amount);
   540→          breakdown = `Formule « ${formulaMatch.name} » : ${formulaMatch.baseRate}% × VN ${nv.toLocaleString("fr-FR")} = ${gross.toLocaleString("fr-FR")} FCFA${formulaMatch.ceiling ? ` (plafond ${formulaMatch.ceiling.toLocaleString("fr-FR")})` : ""} → ${amount.toLocaleString("fr-FR")} FCFA`;
   541→        } else {
   542→          // 3) Fallback : première formule disponible
   543→          if (formulas.length > 0) {
   544→            const fallback = formulas[0];
   545→            const gross = nv * (fallback.baseRate / 100);
   546→            amount = fallback.ceiling
   547→              ? Math.min(gross, fallback.ceiling)
   548→              : gross;
   549→            amount = roundTo500(amount);
   550→            breakdown = `Formule par défaut « ${fallback.name} » : ${fallback.baseRate}% × VN → ${amount.toLocaleString("fr-FR")} FCFA`;
   551→          } else {
   552→            breakdown = `Aucune formule trouvée pour ${seatsRaw} places`;
   553→          }
   554→        }
   555→      }
   556→      source = "FORMULE";
   557→      break;
   558→    }
   559→
   560→    // ── Tierce Complète (TCM) ──
   561→    case "TIERCE_COMPLETE": {
   562→      const result = calculateTierceAmount(meta, vehicle, fp, nv, "TCM");
   563→      amount = result.amount;
   564→      breakdown = result.breakdown;
   565→      source = "TIERCE_COMPLETE";
   566→      break;
   567→    }
   568→
   569→    // ── Tierce Collision (TCL) ──
   570→    case "TIERCE_COLLISION": {
   571→      const result = calculateTierceAmount(meta, vehicle, fp, nv, "TCL");
   572→      amount = result.amount;
   573→      breakdown = result.breakdown;
   574→      source = "TIERCE_COLLISION";
   575→      break;
   576→    }
   577→
   578→    default:
   579→      breakdown = `Dimension de matrice non reconnue : ${meta.dimension}`;
   580→  }
   581→
   582→  // Applique les bornes min / max depuis la metadata
   583→  amount = clamp(amount, meta.minAmount, meta.maxAmount);
   584→
   585→  return {
   586→    amount: Math.max(0, amount),
   587→    method: `MATRIX_BASED (${source})`,
   588→    breakdown,
   589→  };
   590→}

   592→/** Calcul pour TCM / TCL à partir des categoryTariffs */
   593→function calculateTierceAmount(
   594→  meta: MatrixBasedMetadata,
   595→  vehicle: VehiclePricingData,
   596→  fp: number,
   597→  nv: number,
   598→  code: string
   599→): { amount: number; breakdown: string } {
   600→  const catTariffs = meta.categoryTariffs || [];
   601→  const vnClass = getVNRangeClass(nv);
   602→  const vnBounds = VN_RANGE_BOUNDS[vnClass];
   603→  const usageCategory = mapUsageToVehicleCategory(vehicle.usage);
   604→
   605→  // Cherche la tarification correspondant à la classe VN + catégorie d'usage
   606→  const match = catTariffs.find((t) => {
   607→    const tCat = (t.category || "").toUpperCase().trim();
   608→    return tCat === vnClass || tCat === code || tCat === usageCategory;
   609→  });
   610→
   611→  if (match) {
   612→    const amount = roundTo500(match.prime);
   613→    return {
   614→      amount,
   615→      breakdown: `${code} — Classe VN ${vnClass} (${nvBounds.min.toLocaleString("fr-FR")} – ${vnBounds.max === Infinity ? "∞" : vnBounds.max.toLocaleString("fr-FR")} FCFA), catégorie « ${match.category} » → prime ${match.prime.toLocaleString("fr-FR")} FCFA (arrondi ${amount.toLocaleString("fr-FR")} FCFA)`,
   616→    };
   617→  }
   618→
   619→  // Fallback : cherche par valeur dans les bornes valueMin / valueMax
   620→  const valueMatch = catTariffs.find((t) => {
   621→    return nv >= (t.valueMin ?? -Infinity) && nv <= (t.valueMax ?? Infinity);
   622→  });
   623→
   624→  if (valueMatch) {
   625→    const amount = roundTo500(valueMatch.prime);
   626→    return {
   627→      amount,
   628→      breakdown: `${code} — VN ${nv.toLocaleString("fr-FR")} FCFA dans [${valueMatch.valueMin?.toLocaleString("fr-FR") ?? "?"} – ${valueMatch.valueMax?.toLocaleString("fr-FR") ?? "?"}] → prime ${amount.toLocaleString("fr-FR")} FCFA`,
   629→    };
   630→  }
   631→
   632→  return {
   633→    amount: 0,
   634→    breakdown: `${code} — Aucune tarification trouvée pour VN ${nv.toLocaleString("fr-FR")} FCFA (classe ${vnClass})`,
   635→  };
   636→}

   638→/** Cherche un montant dans les règles de tarification DB (fallback) */
   639→function findTariffRuleAmount(
   640→  rules: CoverageInput["tariffRules"] | undefined,
   641→  params: { fiscalPower: number; fuelType: string },
   642→  meta: MatrixBasedMetadata
   643→): number {
   644→  if (!rules || rules.length === 0) return 0;
   645→
   646→  const match = rules.find((r) => {
   647→    const fpOk =
   648→      params.fiscalPower >= (r.minFiscalPower ?? -Infinity) &&
   649→      params.fiscalPower <= (r.maxFiscalPower ?? Infinity);
   650→    const fuelOk =
   651→      !r.fuelType ||
   652→      r.fuelType.toLowerCase() === params.fuelType.toLowerCase() ||
   653→      r.fuelType === "";
   654→    return fpOk && fuelOk;
   655→  });
   656→
   657→  if (match?.fixedAmount != null) return match.fixedAmount;
   658→  if (match?.baseRate != null) {
   659→    // baseRate sur une valeur de référence (par défaut 10 000 000)
   660→    const ref = 10_000_000;
   661→    return roundTo500(ref * (match.baseRate / 100));
   662→  }
   663→
   664→  return 0;
   665→}

   668→function mapUsageToVehicleCategory(usage: string): string {
   669→  const mapping: Record<string, string> = {
   670→    personnel: "VP",
   671→    professionnel: "VP",
   672→    taxi_vtc: "VT",
   673→    autre: "VP",
   674→  };
   675→  return mapping[usage] || "VP";

   678→// ── 2. calculateNetPremium ───────────────────────────────────
   679→
   680→export function calculateNetPremium(
   681→  grossPremium: number,
   682→  options?: NetPremiumOptions
   683→): number {
   684→  const taxDiscountPercent = options?.taxDiscountPercent ?? 5;
   685→  const fees = options?.fees ?? 2_500;
   686→
   687→  const taxDiscount = grossPremium * (taxDiscountPercent / 100);
   688→  const net = grossPremium - taxDiscount + fees;
   689→
   690→  // La prime nette ne peut pas être négative
   691→  return Math.max(0, Math.round(net));
   692→}

   694→// ── 3. scoreOffer ────────────────────────────────────────────
   695→
   696→export function scoreOffer(
   697→  offer: OfferInput,
   698→  vehicleData: VehiclePricingData,
   699→  matchedGuarantees: string[]
   700→): ScoreResult {
   701→  let score = 0;
   702→  const reasons: string[] = [];
   703→
   704→  // +25 pts par garantie correspondante
   705→  const guaranteePoints = matchedGuarantees.length * 25;
   706→  if (guaranteePoints > 0) {
   707→    score += guaranteePoints;
   708→    reasons.push(
   709→      `+${guaranteePoints} pts — ${matchedGuarantees.length} garantie(s) correspondante(s)`
   710→    );
   711→  }
   712→
   713→  // +15 à +30 pts si le type de contrat est défini
   714→  const contractType = offer.contractType;
   715→  if (contractType) {
   716→    const contractPoints: Record<string, number> = {
   717→      basic: 15,
   718→      third_party_plus: 20,
   719→      all_risks: 30,
   720→    };
   721→    const pts = contractPoints[contractType] ?? 15;
   722→    score += pts;
   723→    const labels: Record<string, string> = {
   724→      basic: "Tiers",
   725→      third_party_plus: "Tiers+",
   726→      all_risks: "Tous Risques",
   727→    };
   728→    reasons.push(
   729→      `+${pts} pts — type de contrat « ${labels[contractType] || contractType} »`
   730→    );
   731→  }
   732→
   733→  // +5 à +20 pts si le prix de l'offre est dans une plage raisonnable
   734→  if (offer.priceMin != null && offer.priceMax != null) {
   735→    const nv = parseFCFA(vehicleData.newValue);
   736→    // Plage de prix attendue : ~0.5% à 3% de la VN
   737→    const expectedMin = nv * 0.005;
   738→    const expectedMax = nv * 0.03;
   739→    const priceRange = [offer.priceMin, offer.priceMax];
   740→    const midPrice = (priceRange[0] + priceRange[1]) / 2;
   741→
   742→    if (midPrice >= expectedMin && midPrice <= expectedMax) {
   743→      score += 20;
   744→      reasons.push(
   745→        `+20 pts — plage de prix cohérente avec la valeur du véhicule`
   746→      );
   747→    } else if (midPrice >= expectedMin * 0.5 && midPrice <= expectedMax * 1.5) {
   748→      score += 10;
   749→      reasons.push(`+10 pts — plage de prix approximativement cohérente`);
   750→    } else {
   751→      score += 5;
   752→      reasons.push(`+5 pts — plage de prix définie`);
   753→    }
   754→  } else if (offer.priceMin != null) {
   755→    score += 5;
   756→    reasons.push(`+5 pts — prix minimum défini`);
   757→  }
   758→
   759→  // +15 pts si la puissance fiscale est dans la plage de l'offre
   760→  const fp = parseInt(vehicleData.fiscalPower) || 0;
   761→  if (offer.fiscalPowerMin != null || offer.fiscalPowerMax != null) {
   762→    const fpMin = offer.fiscalPowerMin ?? -Infinity;
   763→    const fpMax = offer.fiscalPowerMax ?? Infinity;
   764→    if (fp >= fpMin && fp <= fpMax) {
   765→      score += 15;
   766→      reasons.push(
   767→        `+15 pts — puissance fiscale ${fp} CV dans la plage [${offer.fiscalPowerMin ?? "?"}–${offer.fiscalPowerMax ?? "?"}]`
   768→      );
   769→    }
   770→  }
   771→
   772→  // +10 pts si le type de carburant est dans la liste de l'offre
   773→  const offerFuelTypes = safeParseStringArray(offer.fuelTypes);
   774→  if (offerFuelTypes.length > 0) {
   775→    const vehicleFuel = (vehicleData.fuelType || "").toLowerCase().trim();
   776→    if (offerFuelTypes.some((ft) => ft.toLowerCase() === vehicleFuel)) {
   777→      score += 10;
   778→      reasons.push(
   779→        `+10 pts — carburant « ${vehicleData.fuelType} » accepté par l'offre`
   780→      );
   781→    }
   782→  } else {
   783→    // Si l'offre n'a pas de restriction carburant, c'est un avantage
   784→    score += 5;
   785→    reasons.push(`+5 pts — tous carburants acceptés`);
   786→  }
   787→
   788→  // +10 pts si la valeur à neuf est dans la plage de l'offre
   789→  const nv = parseFCFA(vehicleData.newValue);
   790→  if (offer.newValueMin != null || offer.newValueMax != null) {
   791→    const nvMin = offer.newValueMin ?? -Infinity;
   792→    const nvMax = offer.newValueMax ?? Infinity;
   793→    if (nv >= nvMin && nv <= nvMax) {
   794→      score += 10;
   795→      reasons.push(
   796→        `+10 pts — valeur à neuf dans la plage de l'offre`
   797→      );
   798→    }
   799→  }
   800→
   801→  // +10 pts si l'usage du véhicule est accepté par l'offre
   802→  const offerUsages = safeParseStringArray(offer.vehicleUsage);
   803→  if (offerUsages.length > 0) {
   804→    const vehicleUsage = (vehicleData.usage || "").toLowerCase().trim();
   805→    if (offerUsages.some((u) => u.toLowerCase() === vehicleUsage)) {
   806→      score += 10;
   807→      reasons.push(
   808→        `+10 pts — usage « ${vehicleData.usage} » accepté par l'offre`
   809→      );
   810→    }
   811→  } else {
   812→    score += 5;
   813→    reasons.push(`+5 pts — tous usages acceptés`);
   814→  }
   815→
   816→  return { score, reasons };
   817→}

   819→// ── 4. isVehicleEligible ─────────────────────────────────────
   820→
   821→export function isVehicleEligible(
   822→  offer: OfferInput,
   823→  vehicleData: VehiclePricingData
   824→): boolean {
   825→  const fp = parseInt(vehicleData.fiscalPower) || 0;
   826→  const nv = parseFCFA(vehicleData.newValue);
   827→  const va = parseFCFA(vehicleData.currentValue);
   828→  const fuel = (vehicleData.fuelType || "").toLowerCase().trim();
   829→  const usage = (vehicleData.usage || "").toLowerCase().trim();
   830→
   831→  // Puissance fiscale
   832→  if (offer.fiscalPowerMin != null && fp < offer.fiscalPowerMin) return false;
   833→  if (offer.fiscalPowerMax != null && fp > offer.fiscalPowerMax) return false;
   834→
   835→  // Type de carburant (liste vide = tous acceptés)
   836→  const fuelTypes = safeParseStringArray(offer.fuelTypes);
   837→  if (fuelTypes.length > 0) {
   838→    const fuelMatch = fuelTypes.some((ft) => ft.toLowerCase() === fuel);
   839→    if (!fuelMatch) return false;
   840→  }
   841→
   842→  // Valeur à neuf
   843→  if (offer.newValueMin != null && nv < offer.newValueMin) return false;
   844→  if (offer.newValueMax != null && nv > offer.newValueMax) return false;
   845→
   846→  // Valeur vénale
   847→  if (offer.venalValueMin != null && va < offer.venalValueMin) return false;
   848→  if (offer.venalValueMax != null && va > offer.venalValueMax) return false;
   849→
   850→  // Usage du véhicule (liste vide = tous acceptés)
   851→  const usages = safeParseStringArray(offer.vehicleUsage);
   852→  if (usages.length > 0) {
   853→    const usageMatch = usages.some((u) => u.toLowerCase() === usage);
   854→    if (!usageMatch) return false;
   855→  }
   856→
   857→  return true;
   858→}
