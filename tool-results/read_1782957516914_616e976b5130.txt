   309→
   310→function ComparisonModal({
   311→  offers,
   312→  open,
   313→  onClose,
   314→}: {
   315→  offers: InsurerOffer[];
   316→  open: boolean;
   317→  onClose: () => void;
   318→}) {
   319→  // Collect all unique features across all offers (before early return for hooks rules)
   320→  const allFeatures = useMemo(() => {
   321→    const featureSet = new Set<string>();
   322→    offers.forEach((o) => o.features.forEach((f) => featureSet.add(f)));
   323→    return Array.from(featureSet);
   324→  }, [offers]);
   325→
   326→  if (!open || offers.length < 2) return null;
   327→
   328→  const formatMonthlyPrice = (price: number) => {
   329→    const val = Math.round(price);
   330→    if (val % 1 !== 0) return new Intl.NumberFormat("fr-FR", { minimumFractionDigits: 0, maximumFractionDigits: 3 }).format(val) + " FCFA";
   331→    return formatFCFA(val);
   332→  };
   333→
   334→  return (
   335→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   336→      <DialogContent className="sm:max-w-[900px] max-h-[90vh] overflow-y-auto p-0">
   337→        {/* Header */}
   338→        <div className="flex items-center justify-between px-6 py-4 border-b border-border/60 bg-muted/30 sticky top-0 z-10">
   339→          <DialogTitle className="text-lg font-bold text-foreground">
   340→            Comparer les garanties
   341→          </DialogTitle>
   342→          <DialogDescription className="sr-only">
   343→            Comparaison détaillée de {offers.length} offres d&apos;assurance
   344→          </DialogDescription>
   345→        </div>
   346→
   347→        {/* Comparison Table */}
   348→        <div className="overflow-x-auto">
   349→          <table className="w-full min-w-[500px]">
   350→            {/* Column Headers: Insurers */}
   351→            <thead>
   352→              <tr>
   353→                <th className="text-left p-4 w-48 bg-muted/30 sticky left-0 z-10 border-b border-border/60">
   354→                  <span className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   355→                    Critères
   356→                  </span>
   357→                </th>
   358→                {offers.map((offer) => (
   359→                  <th
   360→                    key={offer.id}
   361→                    className="text-center p-4 bg-muted/30 border-b border-border/60"
   362→                  >
   363→                    <div className="flex flex-col items-center gap-2">
   364→                      <div className="flex h-12 w-12 items-center justify-center rounded-full bg-primary/10 border-2 border-primary/20">
   365→                        <Shield className="size-6 text-primary" />
   366→                      </div>
   367→                      <span className="text-sm font-bold text-foreground leading-tight">
   368→                        {offer.insurerName}
   369→                      </span>
   370→                      <span className="text-xs text-muted-foreground capitalize">
   371→                        {offer.coverageType.replace("_", " ")}
   372→                      </span>
   373→                    </div>
   374→                  </th>
   375→                ))}
   376→              </tr>
   377→            </thead>
   378→
   379→            <tbody>
   380→              {/* Prix mensuel */}
   381→              <tr className="bg-card">
   382→                <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-card border-b border-border/30">
   383→                  Prix mensuel
   384→                </td>
   385→                {offers.map((offer) => (
   386→                  <td
   387→                    key={offer.id}
   388→                    className="p-4 text-center text-sm font-bold text-foreground border-b border-border/30"
   389→                  >
   390→                    {formatMonthlyPrice(offer.monthlyPrice)}
   391→                  </td>
   392→                ))}
   393→              </tr>
   394→
   395→              {/* Franchise */}
   396→              <tr className="bg-muted/20">
   397→                <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-muted/20 border-b border-border/30">
   398→                  Franchise
   399→                </td>
   400→                {offers.map((offer) => (
   401→                  <td
   402→                    key={offer.id}
   403→                    className="p-4 text-center text-sm text-foreground border-b border-border/30"
   404→                  >
   405→                    {formatFCFA(offer.deductible)}
   406→                  </td>
   407→                ))}
   408→              </tr>
   409→
   410→              {/* Note / Rating */}
   411→              <tr className="bg-card">
   412→                <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-card border-b border-border/30">
   413→                  Note assureur
   414→                </td>
   415→                {offers.map((offer) => (
   416→                  <td
   417→                    key={offer.id}
   418→                    className="p-4 text-center border-b border-border/30"
   419→                  >
   420→                    <div className="flex items-center justify-center">
   421→                      <StarRating rating={offer.insurerRating} />
   422→                    </div>
   423→                  </td>
   424→                ))}
   425→              </tr>
   426→
   427→              {/* Features */}
   428→              {allFeatures.map((feature, idx) => {
   429→                const isZebra = idx % 2 === 1;
   430→                return (
   431→                  <tr
   432→                    key={feature}
   433→                    className={isZebra ? "bg-muted/20" : "bg-card"}
   434→                  >
   435→                    <td
   436→                      className={`p-4 text-sm text-foreground sticky left-0 border-b border-border/30 ${
   437→                        isZebra ? "bg-muted/20" : "bg-card"
   438→                      }`}
   439→                    >
   440→                      {feature}
   441→                    </td>
   442→                    {offers.map((offer) => {
   443→                      const hasFeature = offer.features.includes(feature);
   444→                      return (
   445→                        <td
   446→                          key={offer.id}
   447→                          className="p-4 text-center border-b border-border/30"
   448→                        >
   449→                          {hasFeature ? (
   450→                            <div className="flex items-center justify-center">
   451→                              <div className="flex h-6 w-6 items-center justify-center rounded-full bg-green-100 dark:bg-green-900/30">
   452→                                <Check className="size-3.5 text-green-600 dark:text-green-400" />

   454→                            </div>
   455→                          ) : (
   456→                            <div className="flex items-center justify-center">
   457→                              <div className="flex h-6 w-6 items-center justify-center rounded-full bg-red-100 dark:bg-red-900/30">
   458→                                <X className="size-3.5 text-red-500 dark:text-red-400" />
   459→                              </div>
   460→                            </div>
   461→                          )}
   462→                        </td>
   463→                      );
   464→                    })}
   465→                  </tr>
   466→                );
   467→              })}
   468→
   469→              {/* Conditions */}
   470→              {offers.some((o) => o.conditions) && (
   471→                <tr className="bg-muted/20">
   472→                  <td className="p-4 text-sm font-medium text-foreground sticky left-0 bg-muted/20">
   473→                    Conditions
   474→                  </td>
   475→                  {offers.map((offer) => (
   476→                    <td
   477→                      key={offer.id}
   478→                      className="p-4 text-center text-xs text-muted-foreground max-w-[200px]"
   479→                    >
   480→                      <span className="line-clamp-2">{offer.conditions || "—"}</span>
   481→                    </td>
   482→                  ))}
   483→                </tr>
   484→              )}
   485→            </tbody>
   486→          </table>
   487→        </div>
   488→      </DialogContent>
   489→    </Dialog>
   490→  );
   491→}

   493→/* ──────── Offer Card ──────── */
   494→
   495→function OfferCard({

   497→  priceMode,
   498→  onViewDetails,
   499→  onRequestQuote,
   500→  onAddToCompare,
   501→  isCompared,
   502→}: {

   504→  priceMode: "annual" | "monthly";
   505→  onViewDetails: (offer: InsurerOffer) => void;
   506→  onRequestQuote: (offer: InsurerOffer) => void;
   507→  onAddToCompare: (offer: InsurerOffer) => void;
   508→  isCompared: boolean;
   509→}) {
   510→  const [expanded, setExpanded] = useState(false);
   511→  const visibleFeatures = offer.features.slice(0, 4);
   512→  const hiddenFeatures = offer.features.slice(4);
   513→
   514→  const mainPrice =
   515→    priceMode === "annual" ? offer.annualPrice : offer.monthlyPrice;
   516→  const secondaryPrice =
   517→    priceMode === "annual" ? offer.monthlyPrice : offer.annualPrice;
   518→  const mainLabel = priceMode === "annual" ? "/an" : "/mois";
   519→  const secondaryLabel = priceMode === "annual" ? "/mois" : "/an";
   520→
   521→  return (
   522→    <motion.div
   523→      initial={{ opacity: 0, y: 16 }}
   524→      animate={{ opacity: 1, y: 0 }}
   525→      exit={{ opacity: 0, y: -10 }}
   526→      transition={{ duration: 0.3 }}
   527→      layout
   528→    >
   529→      <div className="bg-card rounded-xl p-5 shadow-sm hover:shadow-lg transition-shadow duration-300 flex flex-col gap-4 h-full">
   530→        {/* Top section: Insurer name + coverage badge */}
   531→        <div className="flex items-start justify-between gap-3">
   532→          <div className="flex flex-col gap-1">
   533→            <h3 className="text-lg font-bold text-foreground leading-tight">
   534→              {offer.insurerName}
   535→            </h3>
   536→            <p className="text-sm text-muted-foreground">{offer.name}</p>
   537→          </div>
   538→          <Badge
   539→            variant="outline"
   540→            className={
   541→              coverageBadgeStyle(offer.coverageType) +
   542→              " rounded-full px-3 py-0.5 text-xs font-semibold shrink-0"
   543→            }
   544→          >
   545→            {offer.coverageType}
   546→          </Badge>
   547→        </div>
   548→
   549→        {/* Star rating */}
   550→        <StarRating rating={offer.insurerRating} />
   551→
   552→        <Separator />
   553→
   554→        {/* Guarantees included */}

   561→              <li key={idx} className="flex items-start gap-2 text-sm">
   562→                <CheckCircle2 className="size-4 text-green-600 mt-0.5 shrink-0" />
   563→                <span className="text-foreground/90">{feature}</span>
   564→              </li>
   565→            ))}
   566→          </ul>
   567→          {hiddenFeatures.length > 0 && (
   568→            <div className="mt-1.5">
   569→              <button
   570→                onClick={() => setExpanded(!expanded)}
   571→                className="text-sm text-secondary hover:text-primary font-medium flex items-center gap-1 transition-colors"
   572→              >

   828→export function ResultsPage() {

   830→    comparisonResults,
   831→    sortBy,

   833→    selectedOffer,

   842→    setOffersToCompare,
   843→    comparisonModalOpen,
   844→    setComparisonModalOpen,
   845→  } = useAppStore();
   846→  const { toast } = useToast();
   847→
   848→  /* ── local filter state ── */
   849→  const [uncheckedInsurers, setUncheckedInsurers] = useState<Set<string>>(
   850→    new Set()
   851→  );
   852→  const [coverageFilter, setCoverageFilter] = useState<string>("all");
   853→  const [budgetMax, setBudgetMax] = useState<number>(300000);
   854→  const [priceMode, setPriceMode] = useState<"annual" | "monthly">("annual");
   855→  const [mobileFiltersOpen, setMobileFiltersOpen] = useState(false);
   856→
   857→  /* ── derived data ── */
   858→  const uniqueInsurers = useMemo(() => {
   859→    const names = [...new Set(comparisonResults.map((o) => o.insurerName))];
   860→    return names.sort();
   861→  }, [comparisonResults]);
   862→
   863→  const effectiveChecked = useMemo(() => {
   864→    const all = new Set(comparisonResults.map((o) => o.insurerName));
   865→    for (const name of uncheckedInsurers) {
   866→      all.delete(name);
   867→    }
   868→    return all;
   869→  }, [comparisonResults, uncheckedInsurers]);
   870→
   871→  const toggleInsurer = useCallback((name: string) => {
   872→    setUncheckedInsurers((prev) => {
   873→      const next = new Set(prev);
   874→      if (next.has(name)) {
   875→        next.delete(name);
   876→      } else {
   877→        next.add(name);
   878→      }
   879→      return next;
   880→    });
   881→  }, []);
   882→
   883→  const filteredAndSorted = useMemo(() => {
   884→    let results = [...comparisonResults];
   885→
   886→    if (coverageFilter !== "all") {
   887→      results = results.filter((o) => o.coverageType === coverageFilter);
   888→    }
   889→
   890→    if (effectiveChecked.size > 0) {
   891→      results = results.filter((o) => effectiveChecked.has(o.insurerName));
   892→    }
   893→
   894→    results = results.filter((o) => o.monthlyPrice <= budgetMax);
   895→
   896→    switch (sortBy) {
   897→      case "price_asc":
   898→        results.sort((a, b) => a.monthlyPrice - b.monthlyPrice);
   899→        break;
   900→      case "price_desc":
   901→        results.sort((a, b) => b.monthlyPrice - a.monthlyPrice);
   902→        break;
   903→      case "rating_desc":
   904→        results.sort((a, b) => b.insurerRating - a.insurerRating);
   905→        break;
   906→      case "name_asc":
   907→        results.sort((a, b) => a.name.localeCompare(b.name, "fr"));
   908→        break;
   909→    }
   910→
   911→    return results;
   912→  }, [comparisonResults, sortBy, coverageFilter, effectiveChecked, budgetMax]);
   913→
   914→  /* ── comparison handlers ── */
   915→  const isOfferCompared = (offerId: string) =>
   916→    offersToCompare.some((o) => o.id === offerId);
   917→
   918→  const handleToggleCompare = useCallback(
   919→    (offer: InsurerOffer) => {
   920→      const alreadyCompared = offersToCompare.some((o) => o.id === offer.id);
   921→      if (alreadyCompared) {
   922→        setOffersToCompare(offersToCompare.filter((o) => o.id !== offer.id));
   923→      } else {
   924→        if (offersToCompare.length >= MAX_COMPARE) {
   925→          toast({
   926→            title: "Limite atteinte",
   927→            description: `Vous pouvez comparer au maximum ${MAX_COMPARE} offres à la fois.`,
   928→            variant: "destructive",
   929→          });
   930→          return;
   931→        }
   932→        setOffersToCompare([...offersToCompare, offer]);
   933→      }
   934→    },
   935→    [offersToCompare, setOffersToCompare, toast]
   936→  );
   937→
   938→  const handleRemoveCompare = useCallback(
   939→    (id: string) => {
   940→      setOffersToCompare(offersToCompare.filter((o) => o.id !== id));
   941→    },
   942→    [offersToCompare, setOffersToCompare]
   943→  );
   944→
   945→  const handleClearCompare = useCallback(() => {
   946→    setOffersToCompare([]);
   947→  }, [setOffersToCompare]);
   948→
   949→  const handleOpenComparison = useCallback(() => {

   979→  const handleViewDetails = (offer: InsurerOffer) => {

   981→  };
   982→
   983→  const handleToggleAllInsurers = useCallback(() => {

   985→      setUncheckedInsurers(new Set(uniqueInsurers));
   986→    } else {
   987→      setUncheckedInsurers(new Set());
   988→    }
   989→  }, [uncheckedInsurers, uniqueInsurers]);
   990→
   991→  const resetFilters = () => {

  1025→      <section className="max-w-7xl mx-auto px-4 sm:px-6 py-6 sm:py-10">
