    97→    97→
    98→    98→function formatLastLogin(dateStr?: string | null): string {
    99→    99→  if (!dateStr) return "—";
   100→   100→  const d = new Date(dateStr);
   101→   101→  const now = new Date();
   102→   102→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   103→   103→  const yesterday = new Date(today);
   104→   104→  yesterday.setDate(yesterday.getDate() - 1);
   105→   105→  const target = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   106→   106→  if (target.getTime() === today.getTime()) return "Aujourd'hui";
   107→   107→  if (target.getTime() === yesterday.getTime()) return "Hier";
   108→   108→  return d.toLocaleDateString("fr-FR");
   109→   109→}
