   537→/* ══════════════════════════ MAIN ══════════════════════════ */
   538→
   539→export function ResultsPage() {
   540→  const {
   541→    comparisonResults,
   542→    sortBy,
   543→    setSortBy,
   544→    selectedInsurerFilter,
   545→    setSelectedInsurerFilter,
   546→    selectedOffer,
   547→    setSelectedOffer,
   548→    setView,
   549→    userQuotes,
   550→    setUserQuotes,
   551→    personalInfo,
   552→    vehicleInfo,
   553→    coverageNeeds,
   554→  } = useAppStore();
   555→  const { toast } = useToast();
   556→
   557→  /* ── local filter state ── */
   558→  // `uncheckedInsurers` is a blacklist: empty set = all insurers selected
   559→  const [uncheckedInsurers, setUncheckedInsurers] = useState<Set<string>>(
   560→    new Set()
   561→  );
   562→  const [coverageFilter, setCoverageFilter] = useState<string>("all");
   563→  const [budgetMax, setBudgetMax] = useState<number>(300000);
   564→  const [priceMode, setPriceMode] = useState<"annual" | "monthly">("annual");
   565→  const [mobileFiltersOpen, setMobileFiltersOpen] = useState(false);
   566→
   567→  /* ── derived data ── */
   568→  const uniqueInsurers = useMemo(() => {
   569→    const names = [...new Set(comparisonResults.map((o) => o.insurerName))];
   570→    return names.sort();
   571→  }, [comparisonResults]);
   572→
   573→  // Compute effective checked insurers: all minus unchecked
   574→  const effectiveChecked = useMemo(() => {
   575→    const all = new Set(comparisonResults.map((o) => o.insurerName));
   576→    for (const name of uncheckedInsurers) {
   577→      all.delete(name);
   578→    }
   579→    return all;
   580→  }, [comparisonResults, uncheckedInsurers]);
   581→
   582→  const toggleInsurer = useCallback((name: string) => {
   583→    setUncheckedInsurers((prev) => {
   584→      const next = new Set(prev);
   585→      if (next.has(name)) {
   586→        next.delete(name);
   587→      } else {
   588→        next.add(name);
   589→      }
   590→      return next;
   591→    });
   592→  }, []);
   593→
   594→  const filteredAndSorted = useMemo(() => {
   595→    let results = [...comparisonResults];
   596→
   597→    // Coverage filter
   598→    if (coverageFilter !== "all") {
   599→      results = results.filter((o) => o.coverageType === coverageFilter);
   600→    }
   601→
   602→    // Insurer filter: only show checked insurers
   603→    if (effectiveChecked.size > 0) {
   604→      results = results.filter((o) => effectiveChecked.has(o.insurerName));
   605→    }
   606→
   607→    // Budget filter (monthly price)
   608→    results = results.filter((o) => o.monthlyPrice <= budgetMax);
   609→
   610→    // Sort
   611→    switch (sortBy) {
   612→      case "price_asc":
   613→        results.sort((a, b) => a.monthlyPrice - b.monthlyPrice);
   614→        break;
   615→      case "price_desc":
   616→        results.sort((a, b) => b.monthlyPrice - a.monthlyPrice);
   617→        break;
   618→      case "rating_desc":
   619→        results.sort((a, b) => b.insurerRating - a.insurerRating);
   620→        break;
   621→      case "name_asc":
   622→        results.sort((a, b) => a.name.localeCompare(b.name, "fr"));
   623→        break;
   624→    }
   625→
   626→    return results;
   627→  }, [comparisonResults, sortBy, coverageFilter, effectiveChecked, budgetMax]);
   628→
   629→  /* ── handlers ── */
   630→  const handleRequestQuote = (offer: InsurerOffer) => {
   631→    toast({
   632→      title: "Devis enregistré",
   633→      description: `Votre demande de devis pour ${offer.name} a été enregistrée.`,
   634→    });
   635→
   636→    const quote: import("@/types").QuoteRecord = {
   637→      id: crypto.randomUUID(),
   638→      reference: `DEV-${Date.now().toString(36).toUpperCase()}`,
   639→      status: "pending",
   640→      personalInfo: { ...personalInfo },
   641→      vehicleInfo: { ...vehicleInfo },
   642→      coverageNeeds: { ...coverageNeeds },
   643→      proposedPrice: offer.annualPrice,
   644→      finalPrice: null,
   645→      insurerName: offer.insurerName,
   646→      offerName: offer.name,
   647→      createdAt: new Date().toISOString(),
   648→    };
   649→
   650→    setUserQuotes([quote, ...userQuotes]);
   651→  };
   652→
   653→  const handleViewDetails = (offer: InsurerOffer) => {
   654→    setSelectedOffer(offer);
   655→  };
   656→
   657→  const handleToggleAllInsurers = useCallback(() => {
   658→    if (uncheckedInsurers.size === 0) {
   659→      // All checked → uncheck all
   660→      setUncheckedInsurers(new Set(uniqueInsurers));
   661→    } else {
   662→      // Some unchecked → check all
   663→      setUncheckedInsurers(new Set());
   664→    }
   665→  }, [uncheckedInsurers, uniqueInsurers]);
   666→
   667→  const resetFilters = () => {
   668→    setCoverageFilter("all");
   669→    setUncheckedInsurers(new Set());
   670→    setBudgetMax(300000);
   671→    setSortBy("price_asc");
   672→    setSelectedInsurerFilter("all");
   673→  };
   674→
   675→  const activeFilterCount = [
   676→    coverageFilter !== "all",
   677→    uncheckedInsurers.size > 0,
   678→    budgetMax < 300000,
   679→  ].filter(Boolean).length;
   680→
   681→  /* ── empty state (no comparison results at all) ── */
   682→  if (comparisonResults.length === 0) {
   683→    return <EmptyResultsState onGoBack={() => setView("compare")} />;
   684→  }
   685→
   686→  /* ── main render ── */
   687→  return (
   688→    <section className="max-w-7xl mx-auto px-4 sm:px-6 py-6 sm:py-10">
   689→      {/* ── Top Bar ── */}
   690→      <div className="flex flex-col sm:flex-row sm:items-center gap-4 mb-6">
   691→        {/* Back button */}
   692→        <Button
   693→          variant="ghost"
   694→          size="sm"
   695→          className="-ml-2 text-muted-foreground hover:text-foreground shrink-0"
   696→          onClick={() => setView("compare")}
   697→        >
   698→          <ArrowLeft className="size-4 mr-1.5" />
   699→          <span className="hidden sm:inline">Retour au formulaire</span>
   700→          <span className="sm:hidden">Retour</span>
   701→        </Button>
   702→
   703→        {/* Title */}
   704→        <h1 className="text-xl sm:text-2xl font-bold text-foreground flex-1">
   705→          {filteredAndSorted.length} offre
   706→          {filteredAndSorted.length !== 1 ? "s" : ""} trouvée
   707→          {filteredAndSorted.length !== 1 ? "s" : ""}
   708→        </h1>
   709→
   710→        {/* Price mode toggle */}
   711→        <div className="flex items-center bg-muted/30 rounded-full p-0.5 shrink-0 self-start sm:self-auto">
   712→          <button
   713→            onClick={() => setPriceMode("annual")}
   714→            className={`px-4 py-1.5 rounded-full text-sm font-medium transition-all duration-200 ${
   715→              priceMode === "annual"
   716→                ? "bg-primary text-primary-foreground shadow-sm"
   717→                : "text-muted-foreground hover:text-foreground"
   718→            }`}
   719→          >
   720→            Par an
   721→          </button>
   722→          <button
   723→            onClick={() => setPriceMode("monthly")}
   724→            className={`px-4 py-1.5 rounded-full text-sm font-medium transition-all duration-200 ${
   725→              priceMode === "monthly"
   726→                ? "bg-primary text-primary-foreground shadow-sm"
   727→                : "text-muted-foreground hover:text-foreground"
   728→            }`}
   729→          >
   730→            Par mois
   731→          </button>
   732→        </div>
   733→      </div>
   734→
   735→      {/* ── Mobile Filter Toggle ── */}
   736→      <div className="lg:hidden mb-4">
   737→        <Button
   738→          variant="outline"
   739→          className="w-full justify-between rounded-xl border-border"
   740→          onClick={() => setMobileFiltersOpen(!mobileFiltersOpen)}
   741→        >
   742→          <div className="flex items-center gap-2">
   743→            <Filter className="size-4" />
   744→            <span className="font-medium">Filtres</span>
   745→            {activeFilterCount > 0 && (
   746→              <Badge className="bg-primary text-primary-foreground rounded-full px-1.5 text-xs min-w-5 h-5 flex items-center justify-center">
   747→                {activeFilterCount}
   748→              </Badge>
   749→            )}
   750→          </div>
   751→          <ChevronDown
   752→            className={`size-4 text-muted-foreground transition-transform duration-200 ${
   753→              mobileFiltersOpen ? "rotate-180" : ""
   754→            }`}
   755→          />
   756→        </Button>
   757→      </div>
   758→
   759→      {/* ── Mobile Filters (collapsible) ── */}
   760→      <AnimatePresence>
   761→        {mobileFiltersOpen && (
   762→          <motion.div
   763→            initial={{ height: 0, opacity: 0 }}
   764→            animate={{ height: "auto", opacity: 1 }}
   765→            exit={{ height: 0, opacity: 0 }}
   766→            transition={{ duration: 0.25 }}
   767→            className="overflow-hidden lg:hidden mb-6"
   768→          >
   769→            <FiltersSidebar
   770→              coverageFilter={coverageFilter}
   771→              setCoverageFilter={setCoverageFilter}
   772→              uncheckedInsurers={uncheckedInsurers}
   773→              toggleInsurer={toggleInsurer}
   774→              onToggleAllInsurers={handleToggleAllInsurers}
   775→              uniqueInsurers={uniqueInsurers}
   776→              budgetMax={budgetMax}
   777→              setBudgetMax={setBudgetMax}
   778→              onReset={resetFilters}
   779→              totalOffers={comparisonResults.length}
   780→            />
   781→          </motion.div>
   782→        )}
   783→      </AnimatePresence>
   784→
   785→      {/* ── Two-column layout ── */}
   786→      <div className="flex gap-6 items-start">
   787→        {/* Desktop Sidebar */}
   788→        <div className="hidden lg:block w-72 shrink-0">
   789→          <FiltersSidebar
   790→            coverageFilter={coverageFilter}
   791→            setCoverageFilter={setCoverageFilter}
   792→            uncheckedInsurers={uncheckedInsurers}
   793→            toggleInsurer={toggleInsurer}
   794→            onToggleAllInsurers={handleToggleAllInsurers}
   795→            uniqueInsurers={uniqueInsurers}
   796→            budgetMax={budgetMax}
   797→            setBudgetMax={setBudgetMax}
   798→            onReset={resetFilters}
   799→            totalOffers={comparisonResults.length}
   800→          />
   801→        </div>
   802→
   803→        {/* Main Content */}
   804→        <main className="flex-1 min-w-0 space-y-4">
   805→          {filteredAndSorted.length > 0 ? (
   806→            <AnimatePresence mode="popLayout">
   807→              {filteredAndSorted.map((offer) => (
   808→                <OfferCard
   809→                  key={offer.id}
   810→                  offer={offer}
   811→                  priceMode={priceMode}
   812→                  onViewDetails={handleViewDetails}
   813→                  onRequestQuote={handleRequestQuote}
   814→                />
   815→              ))}
   816→            </AnimatePresence>
   817→          ) : (
   818→            /* Filtered empty state */
   819→            <div className="text-center py-16">
   820→              <div className="rounded-full bg-muted/40 p-6 mb-4 inline-block">
   821→                <SearchX className="size-10 text-muted-foreground" />
   822→              </div>
   823→              <h3 className="text-lg font-semibold mb-2">
   824→                Aucune offre ne correspond à vos critères
   825→              </h3>
   826→              <p className="text-muted-foreground mb-6 text-sm">
   827→                Essayez d&rsquo;élargir vos filtres ou de modifier le budget.
   828→              </p>
   829→              <Button
   830→                variant="outline"
   831→                className="rounded-full"
   832→                onClick={resetFilters}
   833→              >
   834→                <RotateCcw className="size-4 mr-2" />
   835→                Réinitialiser les filtres
   836→              </Button>
   837→            </div>
   838→          )}
   839→        </main>
   840→      </div>
   841→
   842→      {/* Detail dialog */}
   843→      <OfferDetailDialog
   844→        offer={selectedOffer}
   845→        open={selectedOffer !== null}
   846→        onClose={() => setSelectedOffer(null)}
   847→        onRequestQuote={handleRequestQuote}
   848→      />
   849→    </section>
   850→  );
   851→}
