   409→// ── ASSUREURS TAB ───────────────────────────────────────────────
   410→// ══════════════════════════════════════════════════════════════════
   411→function InsurersTab() {
   412→  const { toast } = useToast();
   413→  const [insurers, setInsurers] = useState<Insurer[]>([]);
   414→  const [loading, setLoading] = useState(true);
   415→  const [search, setSearch] = useState("");
   416→  const [dialogOpen, setDialogOpen] = useState(false);
   417→  const [editing, setEditing] = useState<Insurer | null>(null);
   418→  const [saving, setSaving] = useState(false);
   419→
   420→  // Form state
   421→  const [form, setForm] = useState({
   422→    name: "",
   423→    description: "",
   424→    phone: "",
   425→    email: "",
   426→    website: "",
   427→    rating: 3,
   428→    isVerified: false,
   429→    isActive: true,
   430→  });
   431→
   432→  const resetForm = () =>
   433→    setForm({
   434→      name: "",
   435→      description: "",
   436→      phone: "",
   437→      email: "",
   438→      website: "",
   439→      rating: 3,
   440→      isVerified: false,
   441→      isActive: true,
   442→    });
   443→
   444→  const fetchInsurers = useCallback(async () => {
   445→    setLoading(true);
   446→    try {
   447→      const params = new URLSearchParams();
   448→      if (search) params.set("search", search);
   449→      const res = await fetch(`/api/admin/insurers?${params.toString()}`);
   450→      if (res.ok) setInsurers(await res.json());
   451→    } catch {
   452→      // silent
   453→    } finally {
   454→      setLoading(false);
   455→    }
   456→  }, [search]);
   457→
   458→  useEffect(() => {
   459→    fetchInsurers();
   460→  }, [fetchInsurers]);
   461→
   462→  const openCreate = () => {
   463→    setEditing(null);
   464→    resetForm();
   465→    setDialogOpen(true);
   466→  };
   467→
   468→  const openEdit = (item: Insurer) => {
   469→    setEditing(item);
   470→    setForm({
   471→      name: item.name,
   472→      description: item.description ?? "",
   473→      phone: item.phone ?? "",
   474→      email: item.email ?? "",
   475→      website: item.website ?? "",
   476→      rating: item.rating,
   477→      isVerified: item.isVerified,
   478→      isActive: item.isActive,
   479→    });
   480→    setDialogOpen(true);
   481→  };
   482→
   483→  const handleSave = async () => {
   484→    if (!form.name.trim()) {
   485→      toast({ title: "Le nom est requis", variant: "destructive" });
   486→      return;
   487→    }
   488→    setSaving(true);
   489→    try {
   490→      const url = editing
   491→        ? `/api/admin/insurers/${editing.id}`
   492→        : "/api/admin/insurers";
   493→      const method = editing ? "PUT" : "POST";
   494→      const res = await fetch(url, {
   495→        method,
   496→        headers: { "Content-Type": "application/json" },
   497→        body: JSON.stringify(form),
   498→      });
   499→      if (res.ok) {
   500→        toast({
   501→          title: editing
   502→            ? "Assureur mis à jour"
   503→            : "Assureur créé avec succès",
   504→        });
   505→        setDialogOpen(false);
   506→        fetchInsurers();
   507→      } else {
   508→        toast({ title: "Erreur lors de l&apos;enregistrement", variant: "destructive" });
   509→      }
   510→    } catch {
   511→      toast({ title: "Erreur réseau", variant: "destructive" });
   512→    } finally {
   513→      setSaving(false);
   514→    }
   515→  };
   516→
   517→  const handleDelete = async (id: string) => {
   518→    if (!window.confirm("Supprimer cet assureur ?")) return;
   519→    try {
   520→      const res = await fetch(`/api/admin/insurers/${id}`, { method: "DELETE" });
   521→      if (res.ok) {
   522→        toast({ title: "Assureur supprimé" });
   523→        fetchInsurers();
   524→      }
   525→    } catch {
   526→      toast({ title: "Erreur réseau", variant: "destructive" });
   527→    }
   528→  };
   529→
   530→  return (
   531→    <div className="space-y-6">
   532→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   533→        <div>
   534→          <h1 className="text-2xl font-bold tracking-tight">Assureurs</h1>
   535→          <p className="text-muted-foreground text-sm mt-1">
   536→            Gérer les compagnies d&apos;assurance partenaires
   537→          </p>
   538→        </div>
   539→        <Button onClick={openCreate} size="sm">
   540→          <Plus className="size-4 mr-2" />
   541→          Ajouter un assureur
   542→        </Button>
   543→      </div>
   544→
   545→      {/* Search */}
   546→      <div className="relative max-w-sm">
   547→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   548→        <Input
   549→          placeholder="Rechercher un assureur..."
   550→          value={search}
   551→          onChange={(e) => setSearch(e.target.value)}
   552→          className="pl-9"
   553→        />
   554→      </div>
   555→
   556→      {/* Content */}
   557→      {loading ? (
   558→        <div className="space-y-3">
   559→          {Array.from({ length: 4 }).map((_, i) => (
   560→            <Skeleton key={i} className="h-14 w-full rounded-lg" />
   561→          ))}
   562→        </div>
   563→      ) : !insurers.length ? (
   564→        <div className="bg-card border rounded-xl p-8 text-center text-muted-foreground">
   565→          Aucun assureur trouvé.
   566→        </div>
   567→      ) : (
   568→        <>
   569→          {/* Desktop table */}
   570→          <div className="hidden md:block bg-card border rounded-xl overflow-hidden">
   571→            <Table>
   572→              <TableHeader>
   573→                <TableRow>
   574→                  <TableHead>Nom</TableHead>
   575→                  <TableHead>Téléphone</TableHead>
   576→                  <TableHead>Email</TableHead>
   577→                  <TableHead>Note</TableHead>
   578→                  <TableHead>Vérifié</TableHead>
   579→                  <TableHead>Statut</TableHead>
   580→                  <TableHead className="text-right">Actions</TableHead>
   581→                </TableRow>
   582→              </TableHeader>
   583→              <TableBody>
   584→                {insurers.map((ins) => (
   585→                  <TableRow key={ins.id}>
   586→                    <TableCell className="font-medium">{ins.name}</TableCell>
   587→                    <TableCell>{ins.phone ?? "—"}</TableCell>
   588→                    <TableCell>{ins.email ?? "—"}</TableCell>
   589→                    <TableCell>
   590→                      <div className="flex items-center gap-1">
   591→                        <Star className="size-3.5 fill-primary text-primary" />
   592→                        <span className="text-sm">{ins.rating}</span>
   593→                      </div>
   594→                    </TableCell>
   595→                    <TableCell>
   596→                      {ins.isVerified ? (
   597→                        <Badge
   598→                          variant="secondary"
   599→                          className="bg-green-100 text-green-700 border-0"
   600→                        >
   601→                          Vérifié
   602→                        </Badge>
   603→                      ) : (
   604→                        <Badge variant="outline">Non vérifié</Badge>
   605→                      )}
   606→                    </TableCell>
   607→                    <TableCell>
   608→                      <Badge
   609→                        variant={ins.isActive ? "default" : "outline"}
   610→                        className={
   611→                          ins.isActive
   612→                            ? "bg-green-100 text-green-700 border-0"
   613→                            : "bg-muted text-muted-foreground"
   614→                        }
   615→                      >
   616→                        {ins.isActive ? "Actif" : "Inactif"}
   617→                      </Badge>
   618→                    </TableCell>
   619→                    <TableCell className="text-right">
