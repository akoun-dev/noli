    35→// ── Types ────────────────────────────────────────────────────────
    36→interface RateCondition { condition: string; rate: number; }
    37→
    38→interface Guarantee {
    39→  id: string; name: string; slug: string; description: string | null;
    40→  icon: string | null; category: string; calcMethod: string;
    41→  fixedPrice: number | null; rate: number | null;
    42→  rateConditions: RateCondition[] | null;
    43→  capital: Record<string, string> | null;
    44→  franchise: Record<string, unknown> | null;
    45→  matrixConfig: Record<string, unknown> | null;
    46→  sortOrder: number; isActive: boolean; createdAt: string;
    47→}
    48→
    49→const fmtPrice = (n: number) => new Intl.NumberFormat("fr-FR").format(n) + " FCFA";
    50→const fmtDate = (d: string) => (d ? new Date(d).toLocaleDateString("fr-FR") : "—");
    51→
    52→const calcColors: Record<string, string> = {
    53→  FREE: "bg-emerald-100 text-emerald-800",
    54→  FIXED_AMOUNT: "bg-blue-100 text-blue-800",
    55→  VARIABLE_BASED: "bg-orange-100 text-orange-800",
    56→  MATRIX_BASED: "bg-purple-100 text-purple-800",
    57→};
    58→const calcLabels: Record<string, string> = {
    59→  FREE: "Gratuit", FIXED_AMOUNT: "Montant fixe", VARIABLE_BASED: "Taux variable", MATRIX_BASED: "Matrice",
    60→};
    61→
    62→function slugify(text: string): string {
    63→  return text.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)+/g, "");
    64→}
    65→
    66→// ── Empty form ───────────────────────────────────────────────────
    67→const emptyGuarantee = {
    68→  name: "", slug: "", description: "", icon: "", category: "garantie",
    69→  calcMethod: "FIXED_AMOUNT" as string, fixedPrice: null as number | null,
    70→  rate: null as number | null, rateConditionsStr: "[]", capitalStr: "",
    71→  franchiseStr: "", matrixConfigStr: "", sortOrder: 0, isActive: true,
    72→};
    73→
    74→// ── Helpers to display parsed values ─────────────────────────────
    75→function displayRateOrPrice(g: Guarantee): string {
    76→  if (g.calcMethod === "FREE") return "Gratuit";
    77→  if (g.calcMethod === "FIXED_AMOUNT") return g.fixedPrice ? fmtPrice(g.fixedPrice) : "—";
    78→  if (g.rateConditions && g.rateConditions.length > 0) {
    79→    return g.rateConditions.map((rc) => `${rc.condition}: ${rc.rate}%`).join(" / ");
    80→  }
    81→  if (g.rate != null) return `${g.rate} %`;
    82→  return "—";
    83→}
    84→
    85→function displayCapital(g: Guarantee): string {
    86→  if (!g.capital) return "—";
    87→  return Object.entries(g.capital).map(([k, v]) => `${k}: ${v}`).join("\n");
    88→}
    89→
    90→function displayFranchise(g: Guarantee): string {
    91→  if (!g.franchise) return "—";
    92→  if (g.franchise.description) return g.franchise.description as string;
    93→  const parts: string[] = [];
    94→  if (g.franchise.percent) parts.push(`${g.franchise.percent}%`);
    95→  if (g.franchise.min) parts.push(`min ${fmtPrice(g.franchise.min as number)}`);
    96→  return parts.join(", ") || "—";
    97→}

   428→                      ) : (

   430→                          <div className="flex items-center justify-between">
   431→                            <Label className="text-xs">Conditions de taux</Label>
   432→                            <Button type="button" variant="outline" size="sm" className="h-7 text-xs" onClick={addRateCondition}><Plus className="h-3 w-3 mr-1" />Ajouter</Button>
   433→                          </div>
   434→                          {rateConditions.map((rc, idx) => (
   435→                            <div key={idx} className="flex items-center gap-2">
   436→                              <Input placeholder="Ex: VN ≤ 25 000 000 FCFA" value={rc.condition} onChange={(e) => updateRateCondition(idx, "condition", e.target.value)} className="flex-1 h-9 text-sm" />
   437→                              <div className="relative w-24">
   438→                                <Input type="number" step="0.01" placeholder="Taux" value={rc.rate || ""} onChange={(e) => updateRateCondition(idx, "rate", parseFloat(e.target.value) || 0)} className="h-9 text-sm pr-7" />
   439→                                <span className="absolute right-2 top-1/2 -translate-y-1/2 text-xs text-muted-foreground">%</span>
   440→                              </div>
   441→                              <Button type="button" variant="ghost" size="icon" className="h-9 w-9 text-destructive" onClick={() => removeRateCondition(idx)}><Minus className="h-4 w-4" /></Button>
   442→                            </div>
   443→                          ))}
   444→                        </div>
   445→                      )}
   446→                    </div>
   447→                  )}
   448→
   449→                  {form.calcMethod === "MATRIX_BASED" && (
   450→                    <div className="grid gap-2">
   451→                      <Label>Configuration matrice (JSON)</Label>
   452→                      <Textarea value={form.matrixConfigStr} onChange={(e) => setForm({ ...form, matrixConfigStr: e.target.value })} rows={4} className="font-mono text-xs" placeholder='{"type": "grille_cv_carburant", ...}' />
   453→                      <p className="text-xs text-muted-foreground">Configuration de la grille tarifaire (puissance fiscale, formules, etc.).</p>
   454→                    </div>
   455→                  )}
   456→                </div>
   457→
   458→                <div className="border-t" />
   459→
   460→                {/* ── Section 3: Capital ───────────────────────────── */}
   461→                <div className="space-y-3">
   462→                  <h3 className="font-semibold text-sm text-muted-foreground uppercase tracking-wide">Capital couvert</h3>
   463→                  <div className="grid gap-2">
   464→                    <Label>Capital (JSON)</Label>
   465→                    <Textarea value={form.capitalStr} onChange={(e) => setForm({ ...form, capitalStr: e.target.value })} rows={3} className="font-mono text-xs" placeholder='{"corporel": "7 000 000 000 FCFA", "materiel": "500 000 000 FCFA"}' />
   466→                  </div>
   467→                  {form.capitalStr && (() => {
   468→                    try {
   469→                      const parsed = JSON.parse(form.capitalStr);
   470→                      return (
   471→                        <div className="rounded-lg bg-muted/50 p-3">
   472→                          <p className="text-xs font-medium mb-1 text-muted-foreground">Aperçu :</p>
   473→                          <div className="space-y-1">
   474→                            {Object.entries(parsed).map(([k, v]) => (
   475→                              <p key={k} className="text-sm"><span className="font-medium">{k} :</span> {String(v)}</p>
   476→                            ))}
   477→                          </div>
   478→                        </div>
   479→                      );
   480→                    } catch { return <p className="text-xs text-destructive">JSON invalide</p>; }
   481→                  })()}
   482→                </div>
   483→
   484→                <div className="border-t" />
   485→
   486→                {/* ── Section 4: Franchise ────────────────────────── */}
   487→                <div className="space-y-3">
   488→                  <h3 className="font-semibold text-sm text-muted-foreground uppercase tracking-wide">Franchise</h3>
   489→                  <div className="grid gap-2">
   490→                    <Label>Franchise (JSON)</Label>
   491→                    <Textarea value={form.franchiseStr} onChange={(e) => setForm({ ...form, franchiseStr: e.target.value })} rows={3} className="font-mono text-xs" placeholder='{"percent": 10, "min": 255000, "description": "10% du sinistre, minimum 255 000 FCFA"}' />
   492→                  </div>
   493→                  {form.franchiseStr && (() => {
   494→                    try {
   495→                      const parsed = JSON.parse(form.franchiseStr);
   496→                      return (
   497→                        <div className="rounded-lg bg-muted/50 p-3">
   498→                          <p className="text-xs font-medium mb-1 text-muted-foreground">Aperçu :</p>
   499→                          <div className="space-y-1">
   500→                            {Object.entries(parsed).map(([k, v]) => (
   501→                              <p key={k} className="text-sm">
   502→                                <span className="font-medium">{k} :</span>{" "}
   503→                                {k === "min" ? fmtPrice(v as number) : String(v)}
   504→                              </p>
   505→                            ))}
   506→                          </div>
   507→                        </div>
   508→                      );
   509→                    } catch { return <p className="text-xs text-destructive">JSON invalide</p>; }
   510→                  })()}
   511→                </div>
   512→              </div>
   513→            </ScrollArea>
   514→            <DialogFooter className="pt-2 border-t">
   515→              <Button variant="outline" onClick={() => setFormOpen(false)}>Annuler</Button>
   516→              <Button onClick={handleSave} disabled={saving} className="bg-[#B9E54D] text-black hover:bg-[#a5d044]">
   517→                {saving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}{editing ? "Modifier" : "Créer"}
   518→              </Button>
   519→            </DialogFooter>
   520→          </DialogContent>
   521→        </Dialog>
   522→
   523→        {/* ── Delete Dialog ────────────────────────────────────────── */}
   524→        <AlertDialog open={!!deleteId} onOpenChange={() => setDeleteId(null)}>
   525→          <AlertDialogContent>
   526→            <AlertDialogHeader><AlertDialogTitle>Supprimer cette garantie ?</AlertDialogTitle><AlertDialogDescription>Les liaisons avec les assureurs et offres seront également supprimées.</AlertDialogDescription></AlertDialogHeader>
   527→            <AlertDialogFooter>
   528→              <AlertDialogCancel>Annuler</AlertDialogCancel>
   529→              <AlertDialogAction onClick={handleDelete} disabled={deleting} className="bg-destructive text-destructive-foreground">{deleting && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}Supprimer</AlertDialogAction>
   530→            </AlertDialogFooter>
   531→          </AlertDialogContent>
   532→        </AlertDialog>
   533→      </div>
   534→    </TooltipProvider>
   535→  );
   536→}
