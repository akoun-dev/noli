    51→    51→

    53→    53→  const { toast } = useToast();

    55→    55→  const [insurers, setInsurers] = useState<InsurerMini[]>([]);
    56→    56→  const [loading, setLoading] = useState(true);
    57→    57→  const [search, setSearch] = useState("");
    58→    58→  const [filterInsurer, setFilterInsurer] = useState("");
    59→    59→  const [filterType, setFilterType] = useState("");
    60→    60→  const [formOpen, setFormOpen] = useState(false);
    61→    61→  const [editing, setEditing] = useState<Offer | null>(null);
    62→    62→  const [form, setForm] = useState(empty);
    63→    63→  const [saving, setSaving] = useState(false);
    64→    64→  const [deleteId, setDeleteId] = useState<string | null>(null);
    65→    65→
    66→    66→  // Detail state
    67→    67→  const [detailOpen, setDetailOpen] = useState(false);
    68→    68→  const [selected, setSelected] = useState<Offer | null>(null);
    69→    69→  const [detailLoading, setDetailLoading] = useState(false);
    70→    70→  const [detailCoverages, setDetailCoverages] = useState<CoverageMini[]>([]);
    71→    71→  const [detailQuotesCount, setDetailQuotesCount] = useState(0);
    72→    72→
    73→    73→  // Form: coverages for selected insurer (guarantee checkboxes)

    94→    94→      setFormCoverages([]);

    96→    96→    }

    99→    99→      .then((r) => { if (r.ok) return r.json(); return []; })
   100→   100→      .then((data) => setFormCoverages(data))
   101→   101→      .catch(() => setFormCoverages([]))

   170→   170→        method: "PUT", headers: { "Content-Type": "application/json" },
   171→   171→        body: JSON.stringify({ isActive: !o.isActive }),
   172→   172→      });
   173→   173→      fetchData();
   174→   174→    } catch { /* silent */ }
   175→   175→  };
   176→   176→
   177→   177→  if (loading) return <div className="space-y-4">{Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12 rounded-lg" />)}</div>;
   178→   178→
   179→   179→  // Group coverages by category for detail view
   180→   180→  const groupedCoverages = detailCoverages.reduce<Record<string, CoverageMini[]>>((acc, c) => {
   181→   181→    const key = c.category?.name || "Autre";
   182→   182→    (acc[key] ??= []).push(c);
   183→   183→    return acc;
   184→   184→  }, {});
   185→   185→
   186→   186→  const detailStatCards = [
   187→   187→    { label: "Garanties", value: detailCoverages.length, icon: Shield, color: "text-orange-600 bg-orange-50" },
   188→   188→    { label: "Devis", value: detailQuotesCount, icon: FileText, color: "text-blue-600 bg-blue-50" },
   189→   189→    { label: "Assureur", value: selected?.insurer?.code || "—", icon: Building2, color: "text-purple-600 bg-purple-50" },
   190→   190→  ];
   191→   191→
   192→   192→  return (
   193→   193→    <div className="space-y-4">
   194→   194→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   195→   195→        <h1 className="text-2xl font-bold">Offres</h1>

   197→   197→          <div className="relative flex-1 sm:w-56">

   199→   199→            <Input placeholder="Rechercher..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-9" />

   201→   201→          <Select value={filterInsurer || "all"} onValueChange={(v) => setFilterInsurer(v === "all" ? "" : v)}>

   204→   204→          </Select>
   205→   205→          <Select value={filterType || "all"} onValueChange={(v) => setFilterType(v === "all" ? "" : v)}>
   206→   206→            <SelectTrigger className="w-full sm:w-40"><SelectValue placeholder="Type" /></SelectTrigger>
   207→   207→            <SelectContent><SelectItem value="all">Tous</SelectItem><SelectItem value="basic">Tiers Simple</SelectItem><SelectItem value="third_party_plus">Tiers+</SelectItem><SelectItem value="all_risks">Tous Risques</SelectItem></SelectContent>
   208→   208→          </Select>
   209→   209→          <Button onClick={openCreate} className="bg-[#B9E54D] text-black hover:bg-[#a5d044]"><Plus className="h-4 w-4 mr-1" /> Ajouter</Button>
   210→   210→        </div>
   211→   211→      </div>
   212→   212→
   213→   213→      {/* Desktop table */}
   214→   214→      <div className="hidden md:block rounded-xl border bg-card overflow-hidden">
   215→   215→        <Table>
   216→   216→          <TableHeader>
   217→   217→            <TableRow className="bg-muted/50">
   218→   218→              <TableHead>Nom</TableHead><TableHead>Assureur</TableHead>
   219→   219→              <TableHead className="text-right">Prix min</TableHead><TableHead className="text-right">Prix max</TableHead>
   220→   220→              <TableHead className="text-center">Franchise</TableHead><TableHead>Catégorie</TableHead>
   221→   221→              <TableHead className="text-center">Statut</TableHead><TableHead className="w-12" />
   222→   222→            </TableRow>
   223→   223→          </TableHeader>
   224→   224→          <TableBody>
   225→   225→            {offers.map((o) => (

   232→   232→                <TableCell className="text-sm">{o.category?.name || "—"}</TableCell>

   236→   236→                    <DropdownMenuContent align="end">
   237→   237→                      <DropdownMenuItem onClick={() => openDetail(o)}><Eye className="h-4 w-4 mr-2" /> Voir</DropdownMenuItem>

   239→   239→                      <DropdownMenuItem className="text-destructive" onClick={() => setDeleteId(o.id)}><Trash2 className="h-4 w-4 mr-2" /> Supprimer</DropdownMenuItem>
   240→   240→                    </DropdownMenuContent>
   241→   241→                  </DropdownMenu>
   242→   242→                </TableCell>
   243→   243→              </TableRow>
   244→   244→            ))}
   245→   245→          </TableBody>
   246→   246→        </Table>

   248→   248→      </div>
   249→   249→
   250→   250→      {/* Mobile cards */}
   251→   251→      <div className="md:hidden space-y-3">{offers.map((o) => (

   253→   253→          <div className="flex items-start justify-between mb-1">
   254→   254→            <div><p className="font-semibold">{o.name}</p><p className="text-sm text-muted-foreground">{o.insurer?.name}</p></div>
   255→   255→          </div>
   256→   256→          <div className="flex items-center gap-3 mt-2 text-sm">

   260→   260→          <div className="flex gap-2 mt-3">
   261→   261→            <Button variant="outline" size="sm" className="flex-1" onClick={(e) => { e.stopPropagation(); openDetail(o); }}><Eye className="h-3.5 w-3.5 mr-1" /> Détails</Button>

   263→   263→            <Button variant="outline" size="sm" className="text-destructive" onClick={(e) => { e.stopPropagation(); setDeleteId(o.id); }}><Trash2 className="h-3.5 w-3.5" /></Button>
   264→   264→          </div>
   265→   265→        </Card>
   266→   266→      ))}</div>
   267→   267→
   268→   268→      {/* Create/Edit Dialog */}
   269→   269→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   270→   270→        <DialogContent className="sm:max-w-xl max-h-[90vh] overflow-y-auto">
   271→   271→          <DialogHeader>
   272→   272→            <DialogTitle>{editing ? "Modifier l'offre" : "Nouvelle offre"}</DialogTitle>
   273→   273→            <DialogDescription>Configurez les détails de l'offre.</DialogDescription>
   274→   274→          </DialogHeader>
   275→   275→          <div className="grid gap-4 py-4">
   276→   276→            <div className="w-full grid gap-2">
   277→   277→              <Label>Assureur *</Label>
   278→   278→              <Select value={form.insurerId} onValueChange={(v) => setForm({ ...form, insurerId: v })}>
   279→   279→                <SelectTrigger className="w-full"><SelectValue placeholder="Sélectionner..." /></SelectTrigger>
   280→   280→                <SelectContent>{insurers.map((i) => <SelectItem key={i.id} value={i.id}>{i.name}</SelectItem>)}</SelectContent>
   281→   281→              </Select>
   282→   282→            </div>
   283→   283→
   284→   284→            <div className="w-full grid gap-2">
   285→   285→              <Label>Nom *</Label>

   288→   288→            <div className="w-full grid gap-2">

   292→   292→            <div className="w-full grid gap-2">

   303→   303→              )}
   304→   304→              {!formCoveragesLoading && formCoverages.length > 0 && (() => {
   305→   305→                const grouped = formCoverages.reduce<Record<string, CoverageMini[]>>((acc, c) => {
   306→   306→                  const key = c.category?.name || "Autre";
   307→   307→                  (acc[key] ??= []).push(c);
   308→   308→                  return acc;
   309→   309→                }, {});
   310→   310→                const selectedFeatures: string[] = (() => { try { return JSON.parse(form.features); } catch { return []; } })();
   311→   311→                return (
   312→   312→                  <div className="border rounded-lg max-h-64 overflow-y-auto p-3 space-y-3">
   313→   313→                    {Object.entries(grouped).sort(([a], [b]) => a.localeCompare(b)).map(([cat, covs]) => (

   315→   315→                        <p className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-1.5">{cat}</p>

   334→   334→                        </div>
   335→   335→                      </div>
   336→   336→                    ))}
   337→   337→                  </div>
   338→   338→                );
   339→   339→              })()}
   340→   340→              {!form.insurerId && (
